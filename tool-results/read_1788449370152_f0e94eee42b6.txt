   102→function greetingForHour(): string {
   103→  const h = new Date().getHours();
   104→  if (h < 5) return "Burning the midnight oil";
   105→  if (h < 12) return "Good morning";
   106→  if (h < 17) return "Good afternoon";
   107→  return "Good evening";
   108→}

   110→function fmtRel(iso: string): string {
   111→  const diff = Date.now() - new Date(iso).getTime();
   112→  const mins = Math.floor(diff / 60_000);
   113→  if (mins < 1) return "just now";
   114→  if (mins < 60) return `${mins}m ago`;
   115→  const hrs = Math.floor(mins / 60);
   116→  if (hrs < 24) return `${hrs}h ago`;
   117→  const days = Math.floor(hrs / 24);
   118→  if (days < 7) return `${days}d ago`;
   119→  return new Date(iso).toLocaleDateString("en-IN", { day: "numeric", month: "short" });
   120→}

   126→// Month-over-month delta chip: ▲/▼ with % — colored by whether the move is good.
   127→function Delta({ pct, goodUp = true }: { pct: number | null; goodUp?: boolean }) {
   128→  if (pct === null) {
   129→    return <span className="rounded-full bg-muted px-1.5 py-0.5 text-[10px] font-semibold text-muted-foreground">new</span>;
   130→  }
   131→  if (pct === 0) {
   132→    return (
   133→      <span className="inline-flex items-center gap-0.5 rounded-full bg-muted px-1.5 py-0.5 text-[10px] font-semibold text-muted-foreground">
   134→        <Minus className="h-2.5 w-2.5" aria-hidden />0%
   135→      </span>
   136→    );
   137→  }
   138→  const up = pct > 0;
   139→  const good = up === goodUp;
   140→  const Icon = up ? TrendingUp : TrendingDown;
   141→  return (
   142→    <span
   143→      className={cn(
   144→        "inline-flex items-center gap-0.5 rounded-full px-1.5 py-0.5 text-[10px] font-bold tabular-nums",
   145→        good
   146→          ? "bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-300"
   147→          : "bg-red-100 text-red-700 dark:bg-red-950 dark:text-red-300"
   148→      )}
   149→    >
   150→      <Icon className="h-2.5 w-2.5" aria-hidden />
   151→      {Math.abs(pct)}%
   152→    </span>
   153→  );
   154→}
   155→
   156→function pctOf(cur: number, prev: number): number | null {
   157→  if (prev === 0) return cur === 0 ? 0 : null;
   158→  return Math.round(((cur - prev) / Math.abs(prev)) * 100);
   159→}

   161→function monthShort(month: string): string {
   162→  const [y, m] = month.split("-").map(Number);
   163→  return new Date(y, m - 1, 1).toLocaleDateString("en-IN", { month: "short" });
   164→}
