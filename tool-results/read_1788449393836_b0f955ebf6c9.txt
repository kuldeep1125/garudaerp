    64→function rangeDates(r: RangeKey): { from: string; to: string } {
    65→  const now = new Date();
    66→  const f = (d: Date) => `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;
    67→  if (r === "today") { const t = f(now); return { from: t, to: t }; }
    68→  if (r === "yesterday") return { from: f(new Date(now.getFullYear(), now.getMonth(), now.getDate() - 1)), to: f(new Date(now.getFullYear(), now.getMonth(), now.getDate() - 1)) };
    69→  if (r === "week") return { from: f(new Date(now.getFullYear(), now.getMonth(), now.getDate() - 6)), to: f(now) };
    70→  if (r === "lastweek") return { from: f(new Date(now.getFullYear(), now.getMonth(), now.getDate() - 13)), to: f(new Date(now.getFullYear(), now.getMonth(), now.getDate() - 7)) };
    71→  if (r === "lastmonth") return { from: f(new Date(now.getFullYear(), now.getMonth() - 1, 1)), to: f(new Date(now.getFullYear(), now.getMonth(), 0)) };
    72→  return { from: f(new Date(now.getFullYear(), now.getMonth(), 1)), to: f(new Date(now.getFullYear(), now.getMonth() + 1, 0)) };
    73→}

    95→function ExpenseFormDialog({ open, onOpenChange, expense, vehicles, onDone }: {
    96→  open: boolean;
    97→  onOpenChange: (v: boolean) => void;
    98→  expense: ExpenseRec | null;
    99→  vehicles: VehicleRec[];
   100→  onDone: () => void;
   101→}) {
   102→  const editing = Boolean(expense);
   103→  const [form, setForm] = useState({
   104→    date: todayStr(), business: "MANPOWER", categoryId: "", amount: "", method: "Cash",
   105→    description: "", vehicleId: "", notes: "",
   106→  });
   107→  const [categories, setCategories] = useState<ExpenseCategoryRec[]>([]);
   108→  const { mutate, saving } = useMutation();
   109→
   110→  // Reset form each time the dialog opens (render-time state adjustment).
   111→  const [prevOpen, setPrevOpen] = useState(open);
   112→  if (open !== prevOpen) {
   113→    setPrevOpen(open);
   114→    if (open) {
   115→      setForm({
   116→        date: expense?.date?.slice(0, 10) ?? todayStr(),
   117→        business: expense?.business ?? "MANPOWER",
   118→        categoryId: expense?.categoryId ?? "",
   119→        amount: expense ? String(expense.amount) : "",
   120→        method: expense?.method ?? "Cash",
   121→        description: expense?.description ?? "",
   122→        vehicleId: expense?.vehicleId ?? "",
   123→        notes: expense?.notes ?? "",
   124→      });
   125→    }
   126→  }
   127→
   128→  useEffect(() => {
   129→    if (!open) return;
   130→    let cancelled = false;
   131→    api.get<{ items: ExpenseCategoryRec[] }>("/api/expense-categories" + qs({ business: form.business || undefined }))
   132→      .then((d) => { if (!cancelled) setCategories(d.items ?? []); })
   133→      .catch(() => { if (!cancelled) setCategories([]); });
   134→    return () => { cancelled = true; };
   135→  }, [open, form.business]);
   136→
   137→  const set = (k: keyof typeof form) => (v: string) => setForm((f) => ({ ...f, [k]: v }));
   138→
   139→  const submit = async () => {
   140→    const amt = parseAmount(form.amount);
   141→    if (amt <= 0) { toast.error("Enter a valid amount"); return; }
   142→    if (!form.date) { toast.error("Date is required"); return; }
   143→    const body = {
   144→      date: form.date,
   145→      business: form.business,
   146→      categoryId: form.categoryId || undefined,
   147→      amount: amt,
   148→      method: form.method || undefined,
   149→      description: form.description || undefined,
   150→      notes: form.notes || undefined,
   151→      vehicleId: form.business === "TRANSPORT" && form.vehicleId ? form.vehicleId : undefined,
   152→    };
   153→    const res = editing
   154→      ? await mutate(() => api.put(`/api/expenses/${expense!.id}`, body), "Expense updated", () => ({ module: "EXPENSE", recordId: expense!.id, onUndo: onDone }))
   155→      : await mutate(() => api.post("/api/expenses", body), "Expense recorded", (data) => ({ module: "EXPENSE", recordId: (data as { id: string }).id, onUndo: onDone }));
   156→    if (res.ok) { onOpenChange(false); onDone(); }
   157→  };
   158→
   159→  return (
   160→    <Dialog open={open} onOpenChange={onOpenChange}>
   161→      <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-lg">
   162→        <DialogHeader>
   163→          <DialogTitle>{editing ? "Edit Expense" : "Add Expense"}</DialogTitle>
   164→          <DialogDescription>Expenses are tracked separately for each business.</DialogDescription>
   165→        </DialogHeader>
   166→        <div className="grid gap-3 sm:grid-cols-2">
   167→          <Field label="Date" required>
   168→            <Input type="date" value={form.date} onChange={(e) => set("date")(e.target.value)} className="h-10" />
   169→          </Field>
   170→          <Field label="Business" required>
   171→            <SelectInput
   172→              value={form.business}
   173→              onChange={(v) => setForm((f) => ({ ...f, business: v, categoryId: "", vehicleId: "" }))}
   174→              options={[{ label: "Manpower", value: "MANPOWER" }, { label: "Transport", value: "TRANSPORT" }]}
   175→            />
   176→          </Field>
   177→          <Field label="Category">
   178→            <SelectInput
   179→              value={form.categoryId}
   180→              onChange={set("categoryId")}
   181→              placeholder="Select category…"
   182→              options={categories.map((c) => ({ label: c.name, value: c.id }))}
   183→            />
   184→          </Field>
   185→          <Field label="Amount (₹)" required>
   186→            <MoneyInput value={form.amount} onChange={set("amount")} min={1} className="h-10" />
   187→          </Field>
   188→          <Field label="Method">
   189→            <SelectInput value={form.method} onChange={set("method")} options={METHOD_OPTIONS} />
   190→          </Field>
   191→          {form.business === "TRANSPORT" && (
   192→            <Field label="Vehicle">
   193→              <SelectInput
   194→                value={form.vehicleId}
   195→                onChange={set("vehicleId")}
   196→                placeholder="No specific vehicle"
   197→                options={[{ label: "No specific vehicle", value: "" }, ...vehicles.map((v) => ({ label: v.name, value: v.id }))]}
   198→              />
   199→            </Field>
   200→          )}
   201→          <Field label="Description" className="sm:col-span-2">
   202→            <Input value={form.description} onChange={(e) => set("description")(e.target.value)} className="h-10" placeholder="What was this expense for?" />
   203→          </Field>
   204→          <Field label="Notes" className="sm:col-span-2">
   205→            <Textarea value={form.notes} onChange={(e) => set("notes")(e.target.value)} rows={2} placeholder="Optional" />
   206→          </Field>
   207→        </div>
   208→        <DialogFooter className="gap-2">
   209→          <Button variant="outline" className="min-h-10 flex-1 sm:flex-none" onClick={() => onOpenChange(false)}>Cancel</Button>
   210→          <Button className="min-h-10 flex-1 sm:flex-none" onClick={submit} disabled={saving}>
   211→            {saving ? "Saving…" : editing ? "Save changes" : "Add expense"}
   212→          </Button>
   213→        </DialogFooter>
   214→      </DialogContent>
   215→    </Dialog>
   216→  );
   217→}
