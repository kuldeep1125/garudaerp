    57→function BackupCard() {
    58→  const { lang } = useLang();
    59→  const [exporting, setExporting] = useState(false);
    60→  const [csvOpen, setCsvOpen] = useState(false);
    61→
    62→  const exportBackup = async () => {
    63→    setExporting(true);
    64→    try {
    65→      const backup = await api.get<BackupResp>("/api/settings/backup");
    66→      const total = Object.values(backup.counts).reduce((s, n) => s + n, 0);
    67→      downloadJSON(`bizhub-backup-${todayStr()}.json`, backup);
    68→      toast.success(
    69→        t(lang, "settings.backupDone")
    70→          .replace("{total}", total.toLocaleString("en-IN"))
    71→          .replace("{collections}", String(Object.keys(backup.counts).length))
    72→      );
    73→    } catch (e) {
    74→      toast.error(errMessage(e));
    75→    } finally {
    76→      setExporting(false);
    77→    }
    78→  };
    79→
    80→  return (
    81→    <Card>
    82→      <CardHeader>
    83→        <CardTitle className="flex items-center gap-2 text-base">
    84→          <Database className="h-4 w-4 text-primary" aria-hidden />
    85→          {t(lang, "settings.backupTitle")}
    86→        </CardTitle>
    87→        <CardDescription>{t(lang, "settings.backupDesc")}</CardDescription>
    88→      </CardHeader>
    89→      <CardContent className="space-y-3">
    90→        <div className="flex flex-wrap gap-2">
    91→          <Button variant="outline" className="min-h-10 flex-1 gap-2" onClick={() => void exportBackup()} disabled={exporting}>
    92→            {exporting ? (
    93→              <Loader2 className="h-4 w-4 animate-spin" aria-hidden />
    94→            ) : (
    95→              <Download className="h-4 w-4" aria-hidden />
    96→            )}
    97→            {exporting ? t(lang, "settings.backupWorking") : t(lang, "settings.backupAction")}
    98→          </Button>
    99→          <Button variant="outline" className="min-h-10 flex-1 gap-2" onClick={() => setCsvOpen(true)}>
   100→            <Upload className="h-4 w-4" aria-hidden />

   104→        </div>
   105→        <p className="text-xs leading-relaxed text-muted-foreground">{t(lang, "settings.backupNote")}</p>
   106→        <p className="text-xs leading-relaxed text-muted-foreground">

   119→// Data integrity — zero-mismatch health checks (GET/POST /api/settings/integrity)
   120→// ---------------------------------------------------------------------------
   121→
   122→interface IntegrityResp {
   123→  ok: boolean;
   124→  checks: { id: string; label: string; detail: string; count: number; status: "ok" | "warn" }[];
   125→  totals: { trips: number; settlements: number };

   127→
   128→function IntegrityCard() {
   129→  const { data, loading, error, reload } = useAsync<IntegrityResp>(() => api.get("/api/settings/integrity"), []);
   130→  const [repairing, setRepairing] = useState(false);
   131→
   132→  const repair = async () => {
   133→    setRepairing(true);
   134→    try {
   135→      const r = await api.post<{ repaired: { properties: number; cancelledCleaned: number } }>("/api/settings/integrity");
   136→      toast.success(`Repaired — allocations re-checked across ${r.repaired.properties} property(ies)`);
   137→      void reload();
   138→    } catch (e) {
   139→      toast.error(errMessage(e));
   140→    } finally {
   141→      setRepairing(false);
   142→    }
   143→  };
   144→
   145→  const issues = data?.checks.reduce((s, c) => s + c.count, 0) ?? 0;
   146→
   147→  return (
   148→    <Card>
   149→      <CardHeader>
   150→        <CardTitle className="flex items-center gap-2 text-base">
   151→          <ShieldCheck className="h-4 w-4 text-primary" aria-hidden />
   152→          Data integrity
   153→        </CardTitle>
   154→        <CardDescription>
   155→          Continuous zero-mismatch audit — derived numbers are re-checked against their source records.
   156→        </CardDescription>
   157→      </CardHeader>
   158→      <CardContent className="space-y-3">
   159→        {loading ? (
   160→          <div className="space-y-2" aria-busy="true">
   161→            <Skeleton className="h-9 w-full" />
   162→            <Skeleton className="h-9 w-full" />
   163→            <Skeleton className="h-9 w-full" />
   164→          </div>
   165→        ) : error ? (
   166→          <div className="flex items-center justify-between gap-2">
   167→            <p className="text-xs text-red-600 dark:text-red-400">{error}</p>
   168→            <Button variant="outline" size="sm" className="h-8" onClick={() => void reload()}>Retry</Button>
   169→          </div>
   170→        ) : data ? (
   171→          <>
   172→            <div
   173→              className={cn(
   174→                "flex items-center gap-2 rounded-lg border px-3 py-2 text-xs font-medium",
   175→                data.ok
   176→                  ? "border-emerald-200 bg-emerald-50/70 text-emerald-800 dark:border-emerald-900 dark:bg-emerald-950/30 dark:text-emerald-300"
   177→                  : "border-amber-300 bg-amber-50/70 text-amber-800 dark:border-amber-900 dark:bg-amber-950/30 dark:text-amber-300"
   178→              )}
   179→              role="status"
   180→            >
   181→              {data.ok ? <Check className="h-4 w-4 shrink-0" aria-hidden /> : <AlertTriangle className="h-4 w-4 shrink-0" aria-hidden />}
   182→              {data.ok
   183→                ? "All checks passed — books are consistent"
   184→                : `${issues} issue${issues === 1 ? "" : "s"} found — auto-repair can fix allocation drift`}
   185→            </div>
   186→            <ul className="space-y-1.5">
   187→              {data.checks.map((c) => (
   188→                <li key={c.id} className="flex items-center justify-between gap-2 rounded-lg border px-3 py-2">
   189→                  <div className="min-w-0">
   190→                    <p className="text-xs font-semibold">{c.label}</p>
   191→                    <p className="truncate text-[10px] text-muted-foreground">{c.detail}</p>
   192→                  </div>
   193→                  <span
   194→                    className={cn(
   195→                      "shrink-0 rounded-full px-2 py-0.5 text-[10px] font-bold",
   196→                      c.count === 0
   197→                        ? "bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-300"
   198→                        : "bg-red-100 text-red-700 dark:bg-red-950 dark:text-red-300"
   199→                    )}
   200→                  >
   201→                    {c.count === 0 ? "OK" : c.count}
   202→                  </span>
   203→                </li>
   204→              ))}
   205→            </ul>
   206→            <div className="flex gap-2">
   207→              <Button variant="outline" size="sm" className="h-9 gap-1.5" onClick={() => void reload()} disabled={loading}>
   208→                <RefreshCw className="h-3.5 w-3.5" aria-hidden />
   209→                Re-run checks
   210→              </Button>
   211→              {issues > 0 && (
   212→                <Button size="sm" className="h-9 gap-1.5" onClick={() => void repair()} disabled={repairing}>
   213→                  {repairing ? <Loader2 className="h-3.5 w-3.5 animate-spin" aria-hidden /> : <ShieldCheck className="h-3.5 w-3.5" aria-hidden />}
   214→                  Auto-repair
   215→                </Button>
   216→              )}
   217→            </div>
   218→          </>
   219→        ) : null}
   220→      </CardContent>
   221→    </Card>
   222→  );
   223→}

   229→function DangerZoneCard() {

   241→      toast.error(errMessage(e));

   247→  return (

   253→        </CardTitle>
   254→        <CardDescription>
   255→          Start over: wipe every employee, property, vehicle, trip, payment and expense. Your login and

   258→      </CardHeader>
   259→      <CardContent>
   260→        <AlertDialog>
   261→          <AlertDialogTrigger asChild>
   262→            <Button variant="destructive" className="min-h-10 gap-1.5">

   277→              value={confirmText}
   278→              onChange={(e) => setConfirmText(e.target.value)}

   293→                {resetting ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <AlertTriangle className="h-4 w-4" aria-hidden />}
   294→                Yes, wipe everything
   295→              </AlertDialogAction>
   296→            </AlertDialogFooter>
   297→          </AlertDialogContent>
   298→        </AlertDialog>
   299→      </CardContent>
   300→    </Card>

   376→  const [form, setForm] = useState<SettingsForm | null>(null);

   378→
   379→  // Derived defaults from server data; user edits take over via `form`.
   380→  const biz = data?.business;
   381→  const derived: SettingsForm | null = biz
   382→    ? {
   383→        name: biz.name ?? "",

   391→
   392→  const dirty = Boolean(form && derived && JSON.stringify(form) !== JSON.stringify(derived));

   403→      () => api.put("/api/settings", {

   405→        address: current.address || undefined,
   406→        contact: current.contact || undefined,
   407→        gstin: current.gstin || undefined,
   408→        logoText: current.logoText || undefined,
   409→      }),
   410→      "Settings saved"
   411→    );

   436→              <div className="space-y-2.5" aria-busy="true">
