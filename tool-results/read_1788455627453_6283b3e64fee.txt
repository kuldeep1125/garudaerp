    58→function attClass(n: number): string {
    59→  if (n <= 0) return ATT_HEAT[0];
    60→  if (n === 1) return ATT_HEAT[1];
    61→  if (n === 2) return ATT_HEAT[2];
    62→  return ATT_HEAT[3];

    65→const WD_LETTERS = ["S", "M", "T", "W", "T", "F", "S"];
    66→
    67→function monthLabel(month: string): string {
    68→  const [y, m] = month.split("-").map(Number);
    69→  return new Date(y, (m ?? 1) - 1, 1).toLocaleDateString("en-IN", { month: "long", year: "numeric" });
    70→}
    71→
    72→function shiftMonth(month: string, delta: number): string {
    73→  const [y, m] = month.split("-").map(Number);
    74→  const d = new Date(y, (m ?? 1) - 1 + delta, 1);
    75→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`;
    76→}
    77→
    78→function isWeekend(dateStr: string): boolean {
    79→  const [y, m, d] = dateStr.split("-").map(Number);
    80→  const wd = new Date(y, (m ?? 1) - 1, d ?? 1).getDay();
    81→  return wd === 0 || wd === 6;
    82→}
    83→
    84→function monthBounds(offset = 0): { from: string; to: string } {
    85→  const now = new Date();
    86→  const first = new Date(now.getFullYear(), now.getMonth() + offset, 1);
    87→  const last = new Date(now.getFullYear(), now.getMonth() + offset + 1, 0);
    88→  const f = (d: Date) => `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;
    89→  return { from: f(first), to: f(last) };
    90→}

   459→      </Card>
   460→
   461→      {/* Filters */}
   462→      <Card>
   463→        <CardContent className="space-y-3 p-3 sm:p-4">
   464→          <RangeSelector
   465→            value={(rangeKey ?? "today") as RangeKey}
   466→            onChange={onRange}
   467→          />
   468→          <div className="grid gap-2.5 sm:grid-cols-2 lg:grid-cols-4">
   469→            <div>
   470→              <Label className="mb-1.5 block text-[10px] uppercase tracking-wide text-muted-foreground">Date</Label>
   471→              <Input type="date" value={date} onChange={(e) => onDateChange(e.target.value)} className="h-10" aria-label="Filter by date" />
   472→            </div>
   473→            <div>
   474→              <Label className="mb-1.5 block text-[10px] uppercase tracking-wide text-muted-foreground">Property</Label>
   475→              <SelectInput
   476→                value={propertyId}
   477→                onChange={setPropertyId}
   478→                options={[{ label: "All properties", value: "" }, ...properties.map((p) => ({ label: p.name, value: p.id }))]}
   479→                placeholder="All properties"
   480→              />
   481→            </div>
   482→            <div>
   483→              <Label className="mb-1.5 block text-[10px] uppercase tracking-wide text-muted-foreground">Employee</Label>
   484→              <SelectInput
   485→                value={employeeId}
   486→                onChange={setEmployeeId}
   487→                options={[{ label: "All employees", value: "" }, ...employees.map((e) => ({ label: `${e.fullName} (${e.code})`, value: e.id }))]}
   488→                placeholder="All employees"
   489→              />
   490→            </div>
   491→            <div>
   492→              <Label className="mb-1.5 block text-[10px] uppercase tracking-wide text-muted-foreground">Shift</Label>
   493→              <SelectInput value={shift} onChange={setShift} options={SHIFT_FILTER_OPTIONS} placeholder="All shifts" />
   494→            </div>
   495→          </div>
   496→        </CardContent>
   497→      </Card>
   498→
   499→      {/* Totals */}
   500→      <div className="grid grid-cols-2 gap-2.5 sm:grid-cols-4">
   501→        {loading ? (
   502→          Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-20 rounded-xl" />)
   503→        ) : (
   504→          <>
   505→            <Card className="border-border/70 shadow-sm"><CardContent className="p-3 sm:p-4">
   506→              <p className="text-[11px] sm:text-xs text-muted-foreground">Deployments</p>
   507→              <p className="mt-1 text-lg sm:text-xl font-bold tabular-nums">{totals?.count ?? rows.length}</p>
   508→            </CardContent></Card>
   509→            <Card className="border-border/70 shadow-sm"><CardContent className="p-3 sm:p-4">
   510→              <p className="text-[11px] sm:text-xs text-muted-foreground">Billing</p>
   511→              <p className="mt-1 text-lg sm:text-xl font-bold tabular-nums">{formatINR(totals?.billing ?? 0, { compact: true })}</p>
   512→            </CardContent></Card>
   513→            <Card className="border-border/70 shadow-sm"><CardContent className="p-3 sm:p-4">
   514→              <p className="text-[11px] sm:text-xs text-muted-foreground">Payout</p>
   515→              <p className="mt-1 text-lg sm:text-xl font-bold tabular-nums">{formatINR(totals?.payout ?? 0, { compact: true })}</p>
   516→            </CardContent></Card>
   517→            <Card className="border-border/70 shadow-sm"><CardContent className="p-3 sm:p-4">
   518→              <p className="text-[11px] sm:text-xs text-muted-foreground">Margin</p>
   519→              <p className={cn("mt-1 text-lg sm:text-xl font-bold tabular-nums", (totals?.margin ?? 0) >= 0 ? "text-emerald-600 dark:text-emerald-400" : "text-red-600 dark:text-red-400")}>
   520→                {formatINR(totals?.margin ?? 0, { compact: true })}
   521→              </p>
   522→            </CardContent></Card>
   523→          </>
   524→        )}
   525→      </div>
   526→
   527→      {error && (
   528→        <div className="rounded-xl border border-red-200 bg-red-50/60 p-4 text-center dark:border-red-900 dark:bg-red-950/30">
   529→          <p className="text-xs text-red-700 dark:text-red-300">{error}</p>
   530→          <Button variant="outline" size="sm" className="mt-2 h-8" onClick={() => void load()}>Retry</Button>
   531→        </div>
   532→      )}
   533→
   534→      {!error && (
   535→        <Card>
   536→          <CardContent className="p-3 sm:p-4">
   537→            <DataTable
   538→              columns={columns}
   539→              rows={rows}
   540→              rowKey={(r) => r.id}
   541→              onRowClick={openDetail}
   542→              exportName="deployments"
   543→              stickyFirstCol
   544→              loading={loading}
   545→              emptyIcon={CalendarCheck}
   546→              emptyTitle="No deployments for this filter"
   547→              emptyDescription="Use the Deploy Employees button to create today's roster."
   548→              selectKey={(r) => r.id}
   549→              selectedIds={selectedIds}
   550→              onSelectedChange={setSelectedIds}
   551→              bulkBar={(ids) => (
   552→                <>
   553→                  <span className="px-1.5 text-xs font-semibold tabular-nums whitespace-nowrap">
   554→                    {ids.length} selected
   555→                  </span>
   556→                  <Button
   557→                    variant="outline"
   558→                    size="sm"
   559→                    className="h-8 gap-1.5 text-xs"
   560→                    onClick={() =>
   561→                      downloadCsv(
   562→                        "deployments-selected",
   563→                        columns,
   564→                        rows.filter((r) => ids.includes(r.id))
   565→                      )
   566→                    }
   567→                    aria-label={`Export ${ids.length} selected deployments as CSV`}
   568→                  >
   569→                    <Download className="h-3.5 w-3.5" aria-hidden />Export CSV
   570→                  </Button>
   571→                  <Button
   572→                    variant="destructive"
   573→                    size="sm"
   574→                    className="h-8 gap-1.5 text-xs"
   575→                    onClick={() => setBulkConfirm(true)}
   576→                    aria-label={`Remove ${ids.length} selected deployments`}
   577→                  >
   578→                    <Trash2 className="h-3.5 w-3.5" aria-hidden />Remove
   579→                  </Button>
   580→                </>
   581→              )}
   582→            />
   583→          </CardContent>
   584→        </Card>
   585→      )}
   586→
   587→      <DeployWizard open={wizardOpen} onOpenChange={setWizardOpen} defaultDate={date || undefined} defaultPropertyId={propertyId || undefined} onDone={load} />
   588→
   589→      {/* Detail dialog */}
   590→      <Dialog open={Boolean(detail)} onOpenChange={(v) => !v && setDetail(null)}>
   591→        <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-lg">
   592→          {detail && (
   593→            <>
   594→              <DialogHeader>
   595→                <DialogTitle className="flex flex-wrap items-center gap-2">
   596→                  {detail.employeeName}
   597→                  <StatusBadge status={detail.paidStatus ?? "UNPAID"} />
   598→                </DialogTitle>
   599→                <DialogDescription>
   600→                  {fmtDay(detail.date)} · {detail.propertyName} · {detail.shift === "FULL" ? "Full (day + night)" : detail.shift} shift
   601→                </DialogDescription>
   602→              </DialogHeader>
   603→
   604→              <div className="grid grid-cols-3 gap-2 text-center">
   605→                <div className="rounded-xl bg-muted/60 p-2.5">
   606→                  <p className="text-[10px] uppercase text-muted-foreground">Billing</p>
   607→                  <p className="text-sm font-bold tabular-nums">{formatINR(detail.billingAmount ?? 0)}</p>
   608→                  <p className="text-[10px] text-muted-foreground tabular-nums">{formatINR(detail.billingRate ?? 0)} × {detailUnits}</p>
   609→                </div>
   610→                <div className="rounded-xl bg-muted/60 p-2.5">
   611→                  <p className="text-[10px] uppercase text-muted-foreground">Payout</p>
   612→                  <p className="text-sm font-bold tabular-nums">{formatINR(detail.payoutAmount ?? 0)}</p>
   613→                  <p className="text-[10px] text-muted-foreground tabular-nums">{formatINR(detail.payoutRate ?? 0)} × {detailUnits}</p>
   614→                </div>
   615→                <div className="rounded-xl bg-muted/60 p-2.5">
   616→                  <p className="text-[10px] uppercase text-muted-foreground">Margin</p>
   617→                  <p className={cn("text-sm font-bold tabular-nums", (detail.billingAmount ?? 0) - (detail.payoutAmount ?? 0) >= 0 ? "text-emerald-600 dark:text-emerald-400" : "text-red-600 dark:text-red-400")}>
   618→                    {formatINR((detail.billingAmount ?? 0) - (detail.payoutAmount ?? 0))}
   619→                  </p>
   620→                  <p className="text-[10px] text-muted-foreground">this record</p>
   621→                </div>
   622→              </div>
   623→
   624→              {(detail.adjustmentAmount ?? 0) !== 0 && (
   625→                <p className="rounded-lg bg-amber-50 px-3 py-2 text-xs text-amber-800 dark:bg-amber-950/40 dark:text-amber-300">
   626→                  Adjustment {formatINR(detail.adjustmentAmount ?? 0)}{detail.adjustmentNote ? ` — ${detail.adjustmentNote}` : ""}
   627→                </p>
   628→              )}
   629→              {detail.notes && <p className="text-xs text-muted-foreground">Notes: {detail.notes}</p>}
   630→
   631→              <Separator />
   632→
   633→              {editRates ? (
   634→                <div className="space-y-3">
   635→                  <div className="space-y-1.5">
   636→                    <Label className="text-xs">Shift</Label>
   637→                    <div className="flex rounded-lg border bg-muted/50 p-1" role="group" aria-label="Shift">
   638→                      {SHIFT_OPTIONS.map((s) => (
   639→                        <button
   640→                          key={s.value}
   641→                          type="button"
   642→                          className={cn(
   643→                            "min-h-9 flex-1 rounded-md text-xs font-medium transition-colors",
   644→                            rateForm.shift === s.value ? "bg-primary text-primary-foreground shadow-sm" : "text-muted-foreground hover:text-foreground"
   645→                          )}
   646→                          aria-pressed={rateForm.shift === s.value}
   647→                          onClick={() => setRateForm((f) => ({ ...f, shift: s.value }))}
   648→                        >
   649→                          {s.label}
   650→                        </button>
   651→                      ))}
   652→                    </div>
   653→                  </div>
   654→                  <div className="grid gap-3 sm:grid-cols-2">
   655→                    <div className="space-y-1.5">
   656→                      <Label className="text-xs">Billing rate (₹/shift)</Label>
   657→                      <MoneyInput className="h-9" value={rateForm.billingRate} onChange={(v) => setRateForm((f) => ({ ...f, billingRate: v }))} />
   658→                    </div>
   659→                    <div className="space-y-1.5">
   660→                      <Label className="text-xs">Payout rate (₹/shift)</Label>
   661→                      <MoneyInput className="h-9" value={rateForm.payoutRate} onChange={(v) => setRateForm((f) => ({ ...f, payoutRate: v }))} />
   662→                    </div>
   663→                    <div className="space-y-1.5">
   664→                      <Label className="text-xs">Adjustment amount (± ₹)</Label>
   665→                      <MoneyInput className="h-9" value={rateForm.adjustmentAmount} onChange={(v) => setRateForm((f) => ({ ...f, adjustmentAmount: v }))} placeholder="e.g. -100 or 250" />
   666→                    </div>
   667→                    <div className="space-y-1.5">
   668→                      <Label className="text-xs">Adjustment note</Label>
   669→                      <Input className="h-9" value={rateForm.adjustmentNote} onChange={(e) => setRateForm((f) => ({ ...f, adjustmentNote: e.target.value }))} />
   670→                    </div>
   671→                    <div className="space-y-1.5 sm:col-span-2">
   672→                      <Label className="text-xs">Notes</Label>
   673→                      <Input className="h-9" value={rateForm.notes} onChange={(e) => setRateForm((f) => ({ ...f, notes: e.target.value }))} />
   674→                    </div>
   675→                  </div>
   676→                  <p className="text-[11px] text-muted-foreground">Amounts recompute as rate × shift units (Full = 2) + adjustment.</p>
   677→                  <div className="flex gap-2">
   678→                    <Button variant="outline" className="min-h-10 flex-1" onClick={() => setEditRates(false)}>Cancel</Button>
   679→                    <Button className="min-h-10 flex-1" onClick={saveRates} disabled={saving}>{saving ? "Saving…" : "Save"}</Button>
   680→                  </div>
   681→                </div>
   682→              ) : (
   683→                <div className="space-y-2.5">
   684→                  <div className="flex items-center justify-between gap-2">
   685→                    <p className="text-[11px] text-muted-foreground">
   686→                      By {detail.createdByName ?? "—"} · {fmtDateTime(detail.createdAt)}
   687→                    </p>
   688→                    <div className="flex gap-1.5">
   689→                      <Button variant="outline" size="sm" className="h-8 gap-1.5" onClick={() => setEditRates(true)}>
   690→                        <Pencil className="h-3 w-3" aria-hidden />Edit
   691→                      </Button>
   692→                      <Button variant="outline" size="sm" className="h-8 gap-1.5 border-red-200 text-red-600 hover:bg-red-50 dark:border-red-900 dark:text-red-400 dark:hover:bg-red-950" onClick={() => setConfirmDelete(true)}>
   693→                        <Trash2 className="h-3 w-3" aria-hidden />Remove
   694→                      </Button>
   695→                    </div>
   696→                  </div>
   697→                  <p className="rounded-lg bg-muted/60 px-3 py-2 text-[11px] text-muted-foreground">
   698→                    Removed by mistake? The toast Undo (or Audit Log) restores it exactly — amounts, allocation and totals included.
   699→                  </p>
   700→                </div>
   701→              )}
   702→            </>
   703→          )}
   704→        </DialogContent>
   705→      </Dialog>
   706→
   707→      {/* Delete confirmation */}
   708→      <AlertDialog open={confirmDelete} onOpenChange={setConfirmDelete}>
   709→        <AlertDialogContent>
   710→          <AlertDialogHeader>
   711→            <AlertDialogTitle>Remove this deployment?</AlertDialogTitle>
   712→            <AlertDialogDescription>
   713→              {detail && `${detail.employeeName} — ${fmtDay(detail.date)} · ${detail.propertyName} · ${detail.shift} shift (${formatINR(detail.billingAmount ?? 0)} billing).`} The entry is snapshotted, so Undo restores it with zero mismatch.
   714→            </AlertDialogDescription>
   715→          </AlertDialogHeader>
   716→          <AlertDialogFooter>
   717→            <AlertDialogCancel className="min-h-10">Cancel</AlertDialogCancel>
   718→            <AlertDialogAction className="min-h-10 bg-red-600 text-white hover:bg-red-700" onClick={() => void deleteDeployment()}>
   719→              {saving ? "Removing…" : "Remove entry"}
   720→            </AlertDialogAction>
   721→          </AlertDialogFooter>
   722→        </AlertDialogContent>
   723→      </AlertDialog>
   724→
   725→      {/* Bulk remove confirmation */}
   726→      <AlertDialog open={bulkConfirm} onOpenChange={setBulkConfirm}>
   727→        <AlertDialogContent>
   728→          <AlertDialogHeader>
   729→            <AlertDialogTitle>Remove {selectedIds.size} selected deployment{selectedIds.size === 1 ? "" : "s"}?</AlertDialogTitle>
   730→            <AlertDialogDescription>
   731→              Every removed row is fully snapshotted — a single Undo action after removal restores the whole batch, including FIFO payment allocation.
   732→            </AlertDialogDescription>
   733→          </AlertDialogHeader>
   734→          <AlertDialogFooter>
   735→            <AlertDialogCancel className="min-h-10">Cancel</AlertDialogCancel>
   736→            <AlertDialogAction
   737→              className="min-h-10 bg-red-600 text-white hover:bg-red-700"
   738→              onClick={(e) => { e.preventDefault(); void bulkDelete(); }}
   739→            >
   740→              Remove {selectedIds.size}
   741→            </AlertDialogAction>
   742→          </AlertDialogFooter>
   743→        </AlertDialogContent>
   744→      </AlertDialog>
   745→    </div>
   746→  );
   747→}
