    41→    41→// Media-query rail detection for md–lg widths (tablets / narrow laptops):
    42→    42→// the sidebar becomes an icons-only rail there, while <md hides it entirely
    43→    43→// and lg+ shows the full collapsible sidebar. useSyncExternalStore keeps this
    44→    44→// SSR-safe (server snapshot = false, so hydration never mismatches).
    45→    45→const RAIL_QUERY = "(min-width: 768px) and (max-width: 1023.98px)";
    46→    46→function useIsRailWidth(): boolean {
    47→    47→  return useSyncExternalStore(
    48→    48→    (onStoreChange) => {
    49→    49→      const mq = window.matchMedia(RAIL_QUERY);
    50→    50→      mq.addEventListener("change", onStoreChange);
    51→    51→      return () => mq.removeEventListener("change", onStoreChange);
    52→    52→    },
    53→    53→    () => window.matchMedia(RAIL_QUERY).matches,
    54→    54→    () => false
    55→    55→  );
    56→    56→}

   183→   183→function TopBar({ onOpenMore, onOpenPalette, onOpenShortcuts }: { onOpenMore: () => void; onOpenPalette: () => void; onOpenShortcuts: () => void }) {
   184→   184→  const { owner, logout } = useAuth();
   185→   185→  const { navigate } = useNav();
   186→   186→  const { lang, setLang } = useLang();
   187→   187→  const [notifCount, setNotifCount] = useState(0);
   188→   188→  const [pulse, setPulse] = useState(false);
   189→   189→  const [undoOpen, setUndoOpen] = useState(false);
   190→   190→  const [notifRefresh, setNotifRefresh] = useState(0);
   191→   191→  const { canInstall, install } = usePwaInstall();
   192→   192→  const prevKeysRef = useRef<Set<string> | null>(null);
   193→   193→  const firstLoadRef = useRef(true);
   194→   194→  const pulseTimerRef = useRef<number | null>(null);
   195→   195→
   196→   196→  useEffect(() => {
   197→   197→    let alive = true;
   198→   198→    const load = async () => {
   199→   199→      try {
   200→   200→        const items = await import("@/lib/api-client").then(({ api }) =>
   201→   201→          api.get<AppNotification[]>("/api/notifications")
   202→   202→        );
   203→   203→        if (!alive || !Array.isArray(items)) return;
   204→   204→        // Respect muted categories (localStorage) — the bell only counts alerts
   205→   205→        // the owner hasn't muted from the notifications view.
   206→   206→        const muted = readMutedGroups();
   207→   207→        const visible = muted.length > 0 ? items.filter((n) => !muted.includes(n.severity)) : items;
   208→   208→        setNotifCount(visible.length);
   209→   209→        const prevKeys = prevKeysRef.current;
   210→   210→        prevKeysRef.current = new Set(items.map((n) => n.key));
   211→   211→        // Announce only genuine arrivals (skip initial load), when the tab is
   212→   212→        // visible and the alert isn't muted.
   213→   213→        if (!firstLoadRef.current && prevKeys && document.visibilityState === "visible") {
   214→   214→          const fresh = visible.filter((n) => !prevKeys.has(n.key));
   215→   215→          if (fresh.length > 0) {
   216→   216→            const rank = { CRITICAL: 0, WARNING: 1, INFO: 2 } as const;
   217→   217→            const headline =
   218→   218→              [...fresh].sort((a, b) => rank[a.severity] - rank[b.severity])[0] ?? fresh[0];
   219→   219→            toast(headline.title, {
   220→   220→              description:
   221→   221→                fresh.length > 1
   222→   222→                  ? `${headline.message} — and ${fresh.length - 1} more new alert${fresh.length === 2 ? "" : "s"}.`
   223→   223→                  : headline.message,
   224→   224→              icon: SEVERITY_ICON[headline.severity] ?? SEVERITY_ICON.INFO,
   225→   225→              action: { label: "View", onClick: () => navigate("notifications") },
   226→   226→            });
   227→   227→            // Bell ripple — brief ping so the eye is drawn to the new alert.
   228→   228→            setPulse(true);
   229→   229→            if (pulseTimerRef.current) window.clearTimeout(pulseTimerRef.current);
   230→   230→            pulseTimerRef.current = window.setTimeout(() => setPulse(false), 2400);
   231→   231→          }
   232→   232→        }
   233→   233→        firstLoadRef.current = false;
   234→   234→      } catch { /* ignore */ }
   235→   235→    };
   236→   236→    load();
   237→   237→    const t = setInterval(load, 60_000);
   238→   238→    // Re-check immediately when the user returns to the tab.
   239→   239→    const onVis = () => { if (document.visibilityState === "visible") void load(); };
   240→   240→    // Mute toggles + dismissals (this tab via custom event, other tabs via
   241→   241→    // storage) re-count instantly.
   242→   242→    const onMuteChange = () => { void load(); };
   243→   243→    document.addEventListener("visibilitychange", onVis);
   244→   244→    window.addEventListener(NOTIF_MUTED_EVENT, onMuteChange);
   245→   245→    window.addEventListener(NOTIF_CHANGED_EVENT, onMuteChange);
   246→   246→    window.addEventListener("storage", onMuteChange);
   247→   247→    return () => {
   248→   248→      alive = false; clearInterval(t);
   249→   249→      document.removeEventListener("visibilitychange", onVis);
   250→   250→      window.removeEventListener(NOTIF_MUTED_EVENT, onMuteChange);
   251→   251→      window.removeEventListener(NOTIF_CHANGED_EVENT, onMuteChange);
   252→   252→      window.removeEventListener("storage", onMuteChange);
   253→   253→      if (pulseTimerRef.current) window.clearTimeout(pulseTimerRef.current);
   254→   254→    };
   255→   255→  }, [navigate, notifRefresh]);
   256→   256→
   257→   257→  const initials = (owner?.name ?? "?").slice(0, 2).toUpperCase();
   258→   258→
   259→   259→  return (
   260→   260→    <>
   261→   261→    <header className="sticky top-0 z-40 border-b bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/80">
   262→   262→      <div className="mx-auto flex h-14 max-w-[1600px] items-center gap-2 px-3 sm:px-5">
   263→   263→        <button className="flex items-center gap-2 md:hidden" onClick={() => navigate("dashboard")} aria-label="BizHub home">
   264→   264→          <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-emerald-600 text-white font-bold">
   265→   265→            <span className="text-xs">B</span>
   266→   266→          </div>
   267→   267→          <span className="text-sm font-bold">BizHub</span>
   268→   268→        </button>
   269→   269→
   270→   270→        <Button variant="ghost" size="icon" className="hidden lg:inline-flex h-9 w-9 lg:hidden" aria-hidden tabIndex={-1} />
   271→   271→
   272→   272→        {/* Global search trigger — opens the command palette (Ctrl K) */}
   273→   273→        <button
   274→   274→          onClick={onOpenPalette}
   275→   275→          className="ml-0 hidden sm:flex h-9 min-w-0 flex-1 max-w-md items-center gap-2 rounded-xl border bg-muted/40 px-3 text-sm text-muted-foreground hover:bg-muted hover:border-primary/40 transition-colors"
   276→   276→          aria-label="Open command palette"
   277→   277→        >
   278→   278→          <Search className="h-4 w-4 shrink-0" />
   279→   279→          <span className="min-w-0 truncate whitespace-nowrap hidden md:inline">{t(lang, "topbar.search")}</span>
   280→   280→          <span className="md:hidden whitespace-nowrap">{t(lang, "topbar.searchShort")}</span>

   284→   284→        <div className="ml-auto flex items-center gap-0.5 sm:gap-1">

   286→   286→            <Search className="h-5 w-5" />
   287→   287→          </Button>
   288→   288→          <Button
   289→   289→            variant="ghost"

   487→   487→export function AppShell({ children }: { children: React.ReactNode }) {
   488→   488→  const { navigate, view } = useNav();
   489→   489→  const [collapsed, setCollapsed] = useState(false);
   490→   490→  // md–lg widths force the icons-only rail (no toggle there); lg+ keeps the
   491→   491→  // full sidebar with the manual collapse button.
   492→   492→  const railMode = useIsRailWidth();
   493→   493→  const railActive = railMode || collapsed;
   494→   494→  const [moreOpen, setMoreOpen] = useState(false);
   495→   495→  const [paletteOpen, setPaletteOpen] = useState(false);
   496→   496→  const [shortcutsOpen, setShortcutsOpen] = useState(false);
   497→   497→
   498→   498→  useEffect(() => {
   499→   499→    // "G then key" chord: pendingG holds the timeout id while waiting for the
   500→   500→    // second key (1s window). Cleared by typing anywhere else.
   501→   501→    let pendingG: number | null = null;
   502→   502→    const clearPendingG = () => {
   503→   503→      if (pendingG !== null) window.clearTimeout(pendingG);
   504→   504→      pendingG = null;
   505→   505→    };
   506→   506→    const GO_CHORDS: Record<string, string> = {
   507→   507→      d: "dashboard", e: "employees", p: "properties", t: "trips",
   508→   508→    };
   509→   509→    const isTypingTarget = (el: EventTarget | null): boolean => {
   510→   510→      const t = el as HTMLElement | null;
   511→   511→      return Boolean(
   512→   512→        t &&
   513→   513→        (t.tagName === "INPUT" || t.tagName === "TEXTAREA" || t.tagName === "SELECT" || t.isContentEditable)
   514→   514→      );
   515→   515→    };
   516→   516→    const onKey = (e: KeyboardEvent) => {
   517→   517→      if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === "k") {
   518→   518→        e.preventDefault();
   519→   519→        clearPendingG();
   520→   520→        setPaletteOpen((o) => !o);
   521→   521→        return;
   522→   522→      }
   523→   523→      // Palette is open → its input owns the keyboard; ignore global chords.
   524→   524→      if (paletteOpen) return;
   525→   525→      if (e.altKey || e.ctrlKey || e.metaKey) return;
   526→   526→      // Typing in a field always cancels any pending chord.
   527→   527→      if (isTypingTarget(e.target)) {
   528→   528→        clearPendingG();
   529→   529→        return;
   530→   530→      }
   531→   531→      if (pendingG !== null) {
   532→   532→        const target = GO_CHORDS[e.key.toLowerCase()];
   533→   533→        clearPendingG();
   534→   534→        if (target) {
   535→   535→          e.preventDefault();
   536→   536→          navigate(target);
   537→   537→        }
   538→   538→        return;
   539→   539→      }
   540→   540→      if (e.key === "g" || e.key === "G") {
   541→   541→        pendingG = window.setTimeout(clearPendingG, 1000);
   542→   542→        return;
   543→   543→      }
   544→   544→      if (e.key === "?") {
   545→   545→        e.preventDefault();
   546→   546→        setShortcutsOpen(true);
   547→   547→      }
   548→   548→    };
   549→   549→    window.addEventListener("keydown", onKey);
   550→   550→    return () => {
   551→   551→      window.removeEventListener("keydown", onKey);
   552→   552→      clearPendingG();
   553→   553→    };
   554→   554→  }, [navigate, paletteOpen]);
   555→   555→
   556→   556→  const currentView = getView(view);
   557→   557→  const isWide = ["dashboard", "manpower", "transport"].includes(view);
   558→   558→
   559→   559→  return (
   560→   560→    <div className="flex min-h-screen bg-muted/30">
   561→   561→      {/* Desktop sidebar — icons-only rail at md–lg, full (collapsible) at lg+ */}
   562→   562→      <aside className={cn("sticky top-0 hidden h-screen shrink-0 border-r bg-background transition-all md:block", railActive ? "w-[68px]" : "w-60")}>
   563→   563→        <SideNav collapsed={railActive} onToggle={railMode ? undefined : () => setCollapsed((c) => !c)} />
   564→   564→      </aside>
   565→   565→
   566→   566→      <div className="flex min-w-0 flex-1 flex-col">
   567→   567→        <TopBar onOpenMore={() => setMoreOpen(true)} onOpenPalette={() => setPaletteOpen(true)} onOpenShortcuts={() => setShortcutsOpen(true)} />
   568→   568→        <BusinessBanner />
   569→   569→        <main
   570→   570→          id="main-scroll"
   571→   571→          className={cn("mx-auto w-full max-w-[1600px] flex-1 px-3 pb-24 pt-3 sm:px-5 sm:pt-4 md:pb-10", !isWide && "max-w-[1400px]")}
   572→   572→          aria-label={currentView?.label ?? "Content"}
   573→   573→        >
   574→   574→          {/* Keyed by view → remounts on navigation and plays the enter animation */}
   575→   575→          <div key={view} className="view-enter">
   576→   576→            {children}
   577→   577→          </div>
   578→   578→        </main>
   579→   579→        <footer className="mt-auto hidden border-t bg-background py-3 md:block">
   580→   580→          <div className="mx-auto flex max-w-[1600px] items-center justify-between px-5 text-[11px] text-muted-foreground">
   581→   581→            <p>BizHub — one control center for Manpower & Transport businesses.</p>
   582→   582→            <p className="tabular-nums">{new Date().getFullYear()} · All financial actions are audited.</p>
   583→   583→          </div>
   584→   584→        </footer>
   585→   585→      </div>
   586→   586→
   587→   587→      {/* Mobile bottom nav */}
   588→   588→      <BottomNav onOpenMore={() => setMoreOpen(true)} />
   589→   589→
   590→   590→      {/* Global command palette (Ctrl/Cmd+K, search buttons) */}
   591→   591→      <CommandPalette open={paletteOpen} onOpenChange={setPaletteOpen} />
   592→   592→
   593→   593→      {/* Keyboard shortcuts overlay (? key or owner menu) */}
   594→   594→      <ShortcutsDialog open={shortcutsOpen} onOpenChange={setShortcutsOpen} />
   595→   595→
   596→   596→      {/* Mobile "More" sheet with full nav */}
   597→   597→      <Sheet open={moreOpen} onOpenChange={setMoreOpen}>
   598→   598→        <SheetTrigger className="hidden" aria-hidden />
   599→   599→        <SheetContent side="left" className="w-[280px] p-0">
   600→   600→          <SheetHeader className="sr-only">
   601→   601→            <SheetTitle>All navigation</SheetTitle>
   602→   602→          </SheetHeader>
   603→   603→          <SideNav onNavigate={() => setMoreOpen(false)} />
   604→   604→        </SheetContent>
   605→   605→      </Sheet>
   606→   606→    </div>
   607→   607→  );
