    48→interface AttendanceCell { date: string; shifts: number }
    49→interface AttendanceRow {
    50→  employeeId: string; name: string; code: string; role?: string | null;
    51→  total: number; workedDays: number; cells: AttendanceCell[];

    55→const ATT_HEAT = [

    62→function attClass(n: number): string {
    63→  if (n <= 0) return ATT_HEAT[0];
    64→  if (n === 1) return ATT_HEAT[1];
    65→  if (n === 2) return ATT_HEAT[2];
    66→  return ATT_HEAT[3];
    67→}
    68→
    69→const WD_LETTERS = ["S", "M", "T", "W", "T", "F", "S"];
    70→
    71→function monthLabel(month: string): string {
    72→  const [y, m] = month.split("-").map(Number);
    73→  return new Date(y, (m ?? 1) - 1, 1).toLocaleDateString("en-IN", { month: "long", year: "numeric" });
    74→}
    75→
    76→function shiftMonth(month: string, delta: number): string {
    77→  const [y, m] = month.split("-").map(Number);
    78→  const d = new Date(y, (m ?? 1) - 1 + delta, 1);
    79→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`;
    80→}
    81→
    82→function isWeekend(dateStr: string): boolean {
    83→  const [y, m, d] = dateStr.split("-").map(Number);
    84→  const wd = new Date(y, (m ?? 1) - 1, d ?? 1).getDay();
    85→  return wd === 0 || wd === 6;
    86→}
    87→
    88→function monthBounds(offset = 0): { from: string; to: string } {
    89→  const now = new Date();
    90→  const first = new Date(now.getFullYear(), now.getMonth() + offset, 1);
    91→  const last = new Date(now.getFullYear(), now.getMonth() + offset + 1, 0);
    92→  const f = (d: Date) => `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;
    93→  return { from: f(first), to: f(last) };
    94→}

   451→      </Card>
   452→
   453→      {/* Filters */}
   454→      <Card>
   455→        <CardContent className="space-y-3 p-3 sm:p-4">
   456→          <RangeSelector
   457→            value={(rangeKey ?? "today") as RangeKey}
   458→            onChange={onRange}
   459→          />
   460→          <div className="grid gap-2.5 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-5">
   461→            <div>
   462→              <Label className="mb-1.5 block text-[10px] uppercase tracking-wide text-muted-foreground">Date</Label>
   463→              <Input type="date" value={date} onChange={(e) => onDateChange(e.target.value)} className="h-10" aria-label="Filter by date" />
   464→            </div>
   465→            <div>
   466→              <Label className="mb-1.5 block text-[10px] uppercase tracking-wide text-muted-foreground">Property</Label>
   467→              <SelectInput
   468→                value={propertyId}
   469→                onChange={(v) => { setPropertyId(v); setSelectedIds([]); }}
   470→                options={[{ label: "All properties", value: "" }, ...properties.map((p) => ({ label: p.name, value: p.id }))]}
   471→                placeholder="All properties"
   472→              />
   473→            </div>
   474→            <div>
   475→              <Label className="mb-1.5 block text-[10px] uppercase tracking-wide text-muted-foreground">Employee</Label>
   476→              <SelectInput
   477→                value={employeeId}
   478→                onChange={(v) => { setEmployeeId(v); setSelectedIds([]); }}
   479→                options={[{ label: "All employees", value: "" }, ...employees.map((e) => ({ label: `${e.fullName} (${e.code})`, value: e.id }))]}
   480→                placeholder="All employees"
   481→              />
   482→            </div>
   483→            <div>
   484→              <Label className="mb-1.5 block text-[10px] uppercase tracking-wide text-muted-foreground">Shift</Label>
   485→              <SelectInput value={shift} onChange={(v) => { setShift(v); setSelectedIds([]); }} options={SHIFT_OPTIONS} placeholder="All shifts" />
   486→            </div>
   487→            <div>
   488→              <Label className="mb-1.5 block text-[10px] uppercase tracking-wide text-muted-foreground">Status</Label>
   489→              <SelectInput value={status} onChange={(v) => { setStatus(v); setSelectedIds([]); }} options={STATUS_OPTIONS} placeholder="All statuses" />
   490→            </div>
   491→          </div>
   492→        </CardContent>
   493→      </Card>
   494→
   495→      {/* Totals */}
   496→      <div className="grid grid-cols-2 gap-2.5 sm:grid-cols-4">
   497→        {loading ? (
   498→          Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-20 rounded-xl" />)
   499→        ) : (
   500→          <>
   501→            <Card className="border-border/70 shadow-sm"><CardContent className="p-3 sm:p-4">
   502→              <p className="text-[11px] sm:text-xs text-muted-foreground">Deployments</p>
   503→              <p className="mt-1 text-lg sm:text-xl font-bold tabular-nums">{totals?.count ?? rows.length}</p>
   504→            </CardContent></Card>
   505→            <Card className="border-border/70 shadow-sm"><CardContent className="p-3 sm:p-4">
   506→              <p className="text-[11px] sm:text-xs text-muted-foreground">Billing</p>
   507→              <p className="mt-1 text-lg sm:text-xl font-bold tabular-nums">{formatINR(totals?.billing ?? 0, { compact: true })}</p>
   508→            </CardContent></Card>
   509→            <Card className="border-border/70 shadow-sm"><CardContent className="p-3 sm:p-4">
   510→              <p className="text-[11px] sm:text-xs text-muted-foreground">Payout</p>
   511→              <p className="mt-1 text-lg sm:text-xl font-bold tabular-nums">{formatINR(totals?.payout ?? 0, { compact: true })}</p>
   512→            </CardContent></Card>
   513→            <Card className="border-border/70 shadow-sm"><CardContent className="p-3 sm:p-4">
   514→              <p className="text-[11px] sm:text-xs text-muted-foreground">Margin</p>
   515→              <p className={cn("mt-1 text-lg sm:text-xl font-bold tabular-nums", (totals?.margin ?? 0) >= 0 ? "text-emerald-600 dark:text-emerald-400" : "text-red-600 dark:text-red-400")}>
   516→                {formatINR(totals?.margin ?? 0, { compact: true })}
   517→              </p>
   518→            </CardContent></Card>
   519→          </>
   520→        )}
   521→      </div>
   522→
   523→      {error && (
   524→        <div className="rounded-xl border border-red-200 bg-red-50/60 p-4 text-center dark:border-red-900 dark:bg-red-950/30">
   525→          <p className="text-xs text-red-700 dark:text-red-300">{error}</p>
   526→          <Button variant="outline" size="sm" className="mt-2 h-8" onClick={() => void load()}>Retry</Button>
   527→        </div>
   528→      )}
   529→
   530→      {!error && (
   531→        <Card>
   532→          <CardContent className="p-3 sm:p-4">
   533→            <div className="mb-2.5 flex items-center justify-between">
   534→              <button
   535→                type="button"
   536→                className="flex items-center gap-1.5 rounded-lg px-2 py-1.5 text-xs font-medium text-muted-foreground transition-colors hover:bg-muted hover:text-foreground"
   537→                onClick={toggleAll}
   538→              >
   539→                {allChecked ? <CheckSquare className="h-4 w-4 text-primary" aria-hidden /> : <Square className="h-4 w-4" aria-hidden />}
   540→                Select all {selectedIds.length > 0 && <span className="font-bold text-foreground">({selectedIds.length})</span>}
   541→              </button>
   542→              <p className="text-xs text-muted-foreground">Tap a row for details & rate edits</p>
   543→            </div>
   544→            <DataTable
   545→              columns={columns}
   546→              rows={rows}
   547→              rowKey={(r) => r.id}
   548→              onRowClick={openDetail}
   549→              exportName="deployments"
   550→              loading={loading}
   551→              emptyIcon={CalendarCheck}
   552→              emptyTitle="No deployments for this filter"
   553→              emptyDescription="Use the Deploy Employees button to create today's roster."
   554→            />
   555→          </CardContent>
   556→        </Card>
   557→      )}
   558→
   559→      {/* Bulk action bar */}
   560→      {selectedIds.length > 0 && (
   561→        <div className={cn(
   562→          "fixed inset-x-3 bottom-20 z-40 md:inset-x-auto md:left-1/2 md:bottom-6 md:-translate-x-1/2",
   563→          "rounded-2xl border bg-card/95 p-2 shadow-xl backdrop-blur supports-[backdrop-filter]:bg-card/80"
   564→        )} role="toolbar" aria-label="Bulk actions">
   565→          <div className="flex items-center gap-1.5 overflow-x-auto no-scrollbar">
   566→            <span className="shrink-0 px-1.5 text-xs font-semibold tabular-nums">{selectedIds.length} selected</span>
   567→            <Separator orientation="vertical" className="h-6 shrink-0" />
   568→            <Button size="sm" variant="outline" className="h-9 shrink-0 gap-1.5" onClick={() => void bulkStatus("COMPLETED")}>
   569→              <CheckCheck className="h-3.5 w-3.5 text-emerald-600" aria-hidden />Complete
   570→            </Button>
   571→            <Button size="sm" variant="outline" className="h-9 shrink-0 gap-1.5" onClick={() => void bulkStatus("CONFIRMED")}>
   572→              <IndianRupee className="h-3.5 w-3.5 text-teal-600" aria-hidden />Confirm
   573→            </Button>
   574→            <Button size="sm" variant="outline" className="h-9 shrink-0 gap-1.5" onClick={() => void bulkStatus("NO_SHOW")}>
   575→              <UserX className="h-3.5 w-3.5 text-red-600" aria-hidden />No-Show
   576→            </Button>
   577→            <Button size="sm" variant="ghost" className="h-9 shrink-0" onClick={() => setSelectedIds([])}>Clear</Button>
   578→          </div>
   579→        </div>
   580→      )}
   581→
   582→      <DeployWizard open={wizardOpen} onOpenChange={setWizardOpen} defaultDate={date || undefined} defaultPropertyId={propertyId || undefined} onDone={load} />
   583→
   584→      {/* Detail dialog */}
   585→      <Dialog open={Boolean(detail)} onOpenChange={(v) => !v && setDetail(null)}>
   586→        <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-lg">
   587→          {detail && (
   588→            <>
   589→              <DialogHeader>
   590→                <DialogTitle className="flex flex-wrap items-center gap-2">
   591→                  {detail.employeeName}
   592→                  <StatusBadge status={detail.status} />
   593→                  <StatusBadge status={detail.paidStatus ?? "UNPAID"} />
   594→                </DialogTitle>
   595→                <DialogDescription>
   596→                  {fmtDay(detail.date)} · {detail.propertyName} · {detail.shift} shift{detail.workCategory ? ` · ${detail.workCategory}` : ""}
   597→                </DialogDescription>
   598→              </DialogHeader>
   599→
   600→              <div className="grid grid-cols-3 gap-2 text-center">
   601→                <div className="rounded-xl bg-muted/60 p-2.5">
   602→                  <p className="text-[10px] uppercase text-muted-foreground">Billing</p>
   603→                  <p className="text-sm font-bold tabular-nums">{formatINR(detail.billingAmount ?? 0)}</p>
   604→                  <p className="text-[10px] text-muted-foreground tabular-nums">@ {formatINR(detail.billingRate ?? 0)}</p>
   605→                </div>
   606→                <div className="rounded-xl bg-muted/60 p-2.5">
   607→                  <p className="text-[10px] uppercase text-muted-foreground">Payout</p>
   608→                  <p className="text-sm font-bold tabular-nums">{formatINR(detail.payoutAmount ?? 0)}</p>
   609→                  <p className="text-[10px] text-muted-foreground tabular-nums">@ {formatINR(detail.payoutRate ?? 0)}</p>
   610→                </div>
   611→                <div className="rounded-xl bg-muted/60 p-2.5">
   612→                  <p className="text-[10px] uppercase text-muted-foreground">Margin</p>
   613→                  <p className={cn("text-sm font-bold tabular-nums", (detail.billingAmount ?? 0) - (detail.payoutAmount ?? 0) >= 0 ? "text-emerald-600 dark:text-emerald-400" : "text-red-600 dark:text-red-400")}>
   614→                    {formatINR((detail.billingAmount ?? 0) - (detail.payoutAmount ?? 0))}
   615→                  </p>
   616→                  <p className="text-[10px] text-muted-foreground">per shift</p>
   617→                </div>
   618→              </div>
   619→
   620→              {(detail.adjustmentAmount ?? 0) !== 0 && (
   621→                <p className="rounded-lg bg-amber-50 px-3 py-2 text-xs text-amber-800 dark:bg-amber-950/40 dark:text-amber-300">
   622→                  Adjustment {formatINR(detail.adjustmentAmount ?? 0)}{detail.adjustmentNote ? ` — ${detail.adjustmentNote}` : ""}
   623→                </p>
   624→              )}
   625→              {detail.notes && <p className="text-xs text-muted-foreground">Notes: {detail.notes}</p>}
   626→
   627→              <div>
   628→                <p className="mb-1.5 text-[10px] uppercase tracking-wide text-muted-foreground">Change status</p>
   629→                <div className="flex flex-wrap gap-1.5">
   630→                  {["CONFIRMED", "COMPLETED", "PARTIAL", "NO_SHOW", "CANCELLED"].filter((s) => s !== detail.status).map((s) => (
   631→                    <Button
   632→                      key={s}
   633→                      size="sm"
   634→                      variant="outline"
   635→                      className="h-8 gap-1"
   636→                      disabled={saving}
   637→                      onClick={() => void changeStatus(detail, s)}
   638→                    >
   639→                      {s === "CANCELLED" && <XCircle className="h-3 w-3 text-red-500" aria-hidden />}
   640→                      {s === "NO_SHOW" && <UserX className="h-3 w-3 text-red-500" aria-hidden />}
   641→                      {s === "PARTIAL" && <MinusCircle className="h-3 w-3 text-amber-500" aria-hidden />}
   642→                      {s.replace("_", " ").charAt(0) + s.replace("_", " ").slice(1).toLowerCase()}
   643→                    </Button>
   644→                  ))}
   645→                </div>
   646→              </div>
   647→
   648→              <Separator />
   649→
   650→              {editRates ? (
   651→                <div className="space-y-3">
   652→                  <div className="grid gap-3 sm:grid-cols-2">
   653→                    <div className="space-y-1.5">
   654→                      <Label className="text-xs">Billing rate (₹)</Label>
   655→                      <Input type="number" inputMode="numeric" className="h-9" value={rateForm.billingRate} onChange={(e) => setRateForm((f) => ({ ...f, billingRate: e.target.value }))} />
   656→                    </div>
   657→                    <div className="space-y-1.5">
   658→                      <Label className="text-xs">Payout rate (₹)</Label>
   659→                      <Input type="number" inputMode="numeric" className="h-9" value={rateForm.payoutRate} onChange={(e) => setRateForm((f) => ({ ...f, payoutRate: e.target.value }))} />
   660→                    </div>
   661→                    <div className="space-y-1.5">
   662→                      <Label className="text-xs">Adjustment amount (± ₹)</Label>
   663→                      <Input type="number" inputMode="numeric" className="h-9" value={rateForm.adjustmentAmount} onChange={(e) => setRateForm((f) => ({ ...f, adjustmentAmount: e.target.value }))} placeholder="e.g. -100 or 250" />
   664→                    </div>
   665→                    <div className="space-y-1.5">
   666→                      <Label className="text-xs">Adjustment note</Label>
   667→                      <Input className="h-9" value={rateForm.adjustmentNote} onChange={(e) => setRateForm((f) => ({ ...f, adjustmentNote: e.target.value }))} />
   668→                    </div>
   669→                    <div className="space-y-1.5 sm:col-span-2">
   670→                      <Label className="text-xs">Notes</Label>
   671→                      <Input className="h-9" value={rateForm.notes} onChange={(e) => setRateForm((f) => ({ ...f, notes: e.target.value }))} />
   672→                    </div>
   673→                  </div>
   674→                  <div className="flex gap-2">
   675→                    <Button variant="outline" className="min-h-10 flex-1" onClick={() => setEditRates(false)}>Cancel</Button>
   676→                    <Button className="min-h-10 flex-1" onClick={saveRates} disabled={saving}>{saving ? "Saving…" : "Save"}</Button>
   677→                  </div>
   678→                </div>
   679→              ) : (
   680→                <div className="flex items-center justify-between gap-2">
   681→                  <p className="text-[11px] text-muted-foreground">
   682→                    By {detail.createdByName ?? "—"} · {fmtDateTime(detail.createdAt)}
   683→                  </p>
   684→                  <Button variant="outline" size="sm" className="h-8 gap-1.5" onClick={() => setEditRates(true)}>
   685→                    <Pencil className="h-3 w-3" aria-hidden />Edit rates
   686→                  </Button>
   687→                </div>
   688→              )}
   689→            </>
   690→          )}
   691→        </DialogContent>
   692→      </Dialog>
   693→    </div>
   694→  );
   695→}
