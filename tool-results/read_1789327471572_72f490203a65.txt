    44→function daysUntil(dateStr?: string | null): number | null {
    45→  if (!dateStr) return null;
    46→  const d = new Date(dateStr.length === 10 ? `${dateStr}T00:00:00` : dateStr);
    47→  if (Number.isNaN(d.getTime())) return null;
    48→  const today = new Date();
    49→  today.setHours(0, 0, 0, 0);
    50→  return Math.round((d.getTime() - today.getTime()) / 86400000);
    51→}
    52→
    53→function tripTotal(t: TripRec): number {
    54→  return t.finalAmount ?? (t.agreedAmount + (t.extraCharges ?? 0));
    55→}
    56→
    57→function expiryCls(dateStr?: string | null): string {
    58→  const d = daysUntil(dateStr);
    59→  if (d === null) return "";
    60→  if (d < 0) return "text-red-600 dark:text-red-400 font-semibold";
    61→  if (d <= 30) return "text-amber-600 dark:text-amber-400 font-semibold";
    62→  return "";
    63→}
    64→
    65→function expiryHint(dateStr?: string | null): string {
    66→  const d = daysUntil(dateStr);
    67→  if (d === null) return "";
    68→  if (d < 0) return ` (expired ${Math.abs(d)}d ago)`;
    69→  if (d <= 30) return ` (in ${d}d)`;
    70→  return "";
    71→}

    77→function VehicleExpenseDialog({ open, onOpenChange, vehicleId, onDone }: {
    78→  open: boolean;
    79→  onOpenChange: (v: boolean) => void;
    80→  vehicleId: string;
    81→  onDone: () => void;
    82→}) {
    83→  const [form, setForm] = useState({ date: todayStr(), categoryId: "", amount: "", method: "Cash", description: "", notes: "" });
    84→  const [categories, setCategories] = useState<ExpenseCategoryRec[]>([]);
    85→  const { mutate, saving } = useMutation();
    86→
    87→  const [prevOpen, setPrevOpen] = useState(open);
    88→  if (open !== prevOpen) {
    89→    setPrevOpen(open);
    90→    if (open) setForm({ date: todayStr(), categoryId: "", amount: "", method: "Cash", description: "", notes: "" });
    91→  }
    92→
    93→  useEffect(() => {
    94→    if (!open) return;
    95→    let cancelled = false;
    96→    api.get<{ items: ExpenseCategoryRec[] }>("/api/expense-categories" + qs({ business: "TRANSPORT" }))
    97→      .then((d) => { if (!cancelled) setCategories(d.items ?? []); })
    98→      .catch(() => { if (!cancelled) setCategories([]); });
    99→    return () => { cancelled = true; };
   100→  }, [open]);
   101→
   102→  const submit = async () => {
   103→    const amt = parseAmount(form.amount);
   104→    if (amt <= 0) { toast.error("Enter a valid amount"); return; }
   105→    if (!form.date) { toast.error("Date is required"); return; }
   106→    const res = await mutate(
   107→      () => api.post("/api/expenses", {
   108→        date: form.date,
   109→        business: "TRANSPORT",
   110→        vehicleId: vehicleId || undefined,
   111→        categoryId: form.categoryId || undefined,
   112→        amount: amt,
   113→        method: form.method || undefined,
   114→        description: form.description || undefined,
   115→        notes: form.notes || undefined,
   116→      }),
   117→      "Expense recorded"
   118→    );
   119→    if (res.ok) { onOpenChange(false); onDone(); }
   120→  };
   121→
   122→  return (
   123→    <Dialog open={open} onOpenChange={onOpenChange}>
   124→      <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-md">
   125→        <DialogHeader>
   126→          <DialogTitle>Add Vehicle Expense</DialogTitle>
   127→          <DialogDescription>Recorded under the Transport business for this vehicle.</DialogDescription>
   128→        </DialogHeader>
   129→        <div className="grid gap-3 sm:grid-cols-2">
   130→          <Field label="Date" required>
   131→            <Input type="date" value={form.date} onChange={(e) => setForm((f) => ({ ...f, date: e.target.value }))} className="h-10" />
   132→          </Field>
   133→          <Field label="Category">
   134→            <SelectInput
   135→              value={form.categoryId}
   136→              onChange={(v) => setForm((f) => ({ ...f, categoryId: v }))}
   137→              placeholder="Select category…"
   138→              options={categories.map((c) => ({ label: c.name, value: c.id }))}
   139→            />
   140→          </Field>
   141→          <Field label="Amount (₹)" required>
   142→            <MoneyInput value={form.amount} onChange={(v) => setForm((f) => ({ ...f, amount: v }))} min={1} className="h-10" />
   143→          </Field>
   144→          <Field label="Method">
   145→            <SelectInput value={form.method} onChange={(v) => setForm((f) => ({ ...f, method: v }))} options={METHOD_OPTIONS} />
   146→          </Field>
   147→          <Field label="Description" className="sm:col-span-2">
   148→            <Input value={form.description} onChange={(e) => setForm((f) => ({ ...f, description: e.target.value }))} className="h-10" placeholder="Fuel, toll, repair…" />
   149→          </Field>
   150→          <Field label="Notes" className="sm:col-span-2">
   151→            <Textarea value={form.notes} onChange={(e) => setForm((f) => ({ ...f, notes: e.target.value }))} rows={2} placeholder="Optional" />
   152→          </Field>
   153→        </div>
   154→        <DialogFooter className="gap-2">
   155→          <Button variant="outline" className="min-h-10 flex-1 sm:flex-none" onClick={() => onOpenChange(false)}>Cancel</Button>
   156→          <Button className="min-h-10 flex-1 sm:flex-none" onClick={submit} disabled={saving}>{saving ? "Saving…" : "Add expense"}</Button>
   157→        </DialogFooter>
   158→      </DialogContent>
   159→    </Dialog>
   160→  );
   161→}

   167→function AddMaintenanceDialog({ open, onOpenChange, vehicleId, onDone }: {
   168→  open: boolean;
   169→  onOpenChange: (v: boolean) => void;
   170→  vehicleId: string;
   171→  onDone: () => void;
   172→}) {
   173→  const [form, setForm] = useState({ date: todayStr(), type: "", description: "", cost: "", nextDueDate: "" });
   174→  const { mutate, saving } = useMutation();
   175→
   176→  const [prevOpen, setPrevOpen] = useState(open);
   177→  if (open !== prevOpen) {
   178→    setPrevOpen(open);
   179→    if (open) setForm({ date: todayStr(), type: "", description: "", cost: "", nextDueDate: "" });
   180→  }
   181→
   182→  const submit = async () => {
   183→    if (!form.type.trim()) { toast.error("Maintenance type is required"); return; }
   184→    if (!form.date) { toast.error("Date is required"); return; }

   186→      () => api.post("/api/maintenance", {

   194→      "Maintenance scheduled"
   195→    );

   200→    <Dialog open={open} onOpenChange={onOpenChange}>
   201→      <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-md">

   431→        </TabsList>
   432→
   433→        {/* Trips */}
   434→        <TabsContent value="trips" className="mt-3">
   435→          <Card>
   436→            <CardContent className="p-3 sm:p-4">
   437→              <DataTable
   438→                columns={tripColumns}
   439→                rows={trips}
   440→                rowKey={(r) => r.id}
   441→                onRowClick={() => navigate("trips", { vehicleId: v.id })}
   442→                emptyIcon={Route}
   443→                emptyTitle="No trips for this vehicle"
   444→                emptyDescription="Create a rental from the Trips & Rentals screen."
   445→              />
   446→            </CardContent>
   447→          </Card>
   448→        </TabsContent>

   450→        {/* Expenses */}

   457→                </Button>
   458→              </div>
   459→              <DataTable
   460→                columns={expenseColumns}
   461→                rows={expenses}
   462→                rowKey={(r) => r.id}
   463→                emptyIcon={Receipt}
   464→                emptyTitle="No expenses recorded"

   466→              />
   467→            </CardContent>
   468→          </Card>
   469→        </TabsContent>
   470→
   471→        {/* EMI */}
   472→        <TabsContent value="emi" className="mt-3">
   473→          <Card>
   474→            <CardContent className="p-3 sm:p-4">

   498→                      </div>

   500→                        <span className="text-sm font-semibold tabular-nums">{formatINR(e.amount)}</span>
   501→                        <StatusBadge status={e.status} />

   505→                          </Button>
   506→                        )}
   507→                      </div>
   508→                    </div>
   509→                  ))}

   512→            </CardContent>
   513→          </Card>
   514→        </TabsContent>
   515→
   516→        {/* Maintenance */}
   517→        <TabsContent value="maintenance" className="mt-3">

   519→            <CardContent className="p-3 sm:p-4">
   520→              <div className="mb-3 flex justify-end">
   521→                <Button size="sm" variant="outline" className="h-9 gap-1.5" onClick={() => setMaintenanceOpen(true)}>

   523→                </Button>
   524→              </div>
   525→
   526→              {maintenance.length === 0 ? (
   527→                <p className="py-8 text-center text-xs text-muted-foreground">No maintenance records yet.</p>
   528→              ) : (
   529→                <div className="max-h-96 space-y-2 overflow-y-auto pr-0.5">
   530→                  {maintenance.map((m) => (
   531→                    <div key={m.id} className="rounded-xl border p-3">
   532→                      <div className="flex flex-wrap items-start justify-between gap-2">
   533→                        <div className="min-w-0">
   534→                          <p className="text-sm font-medium">{m.type}{(m.cost ?? 0) > 0 && <span className="ml-2 font-semibold tabular-nums">{formatINR(m.cost ?? 0)}</span>}</p>
   535→                          <p className="mt-0.5 text-[11px] text-muted-foreground">
   536→                            {fmtDay(m.date)}{m.nextDueDate ? ` · next due ${fmtDay(m.nextDueDate)}` : ""}
   537→                            {m.createdByName ? ` · by ${m.createdByName}` : ""}
   538→                          </p>
   539→                          {m.description && <p className="mt-1 line-clamp-2 text-xs text-muted-foreground">{m.description}</p>}
   540→                        </div>
   541→                        <div className="flex shrink-0 items-center gap-2">
   542→                          <StatusBadge status={m.status} />
   543→                          {m.status === "SCHEDULED" && (
   544→                            <Button size="sm" variant="outline" className="h-8 text-xs" disabled={saving} onClick={() => void markMaintenanceDone(m)}>
   545→                              Mark Done
   546→                            </Button>
   547→                          )}
   548→                        </div>
   549→                      </div>
   550→                    </div>
   551→                  ))}
   552→                </div>
   553→              )}
   554→            </CardContent>
   555→          </Card>
   556→        </TabsContent>
   557→      </Tabs>
   558→
   559→      <VehicleExpenseDialog open={expenseOpen} onOpenChange={setExpenseOpen} vehicleId={v.id} onDone={() => void reload()} />
   560→      <AddMaintenanceDialog open={maintenanceOpen} onOpenChange={setMaintenanceOpen} vehicleId={v.id} onDone={() => void reload()} />
   561→    </div>
   562→  );
   563→}
