   271→   271→/** Fetch-on-mount hook with loading/error/reload. Toasts load failures. */
   272→   272→export function useAsync<T>(loader: () => Promise<T>, deps: React.DependencyList) {
   273→   273→  const [data, setData] = useState<T | null>(null);
   274→   274→  const [loading, setLoading] = useState(true);
   275→   275→  const [error, setError] = useState<string | null>(null);
   276→   276→  const load = useCallback(async () => {
   277→   277→    setLoading(true);
   278→   278→    try {
   279→   279→      const d = await loader();
   280→   280→      setData(d);
   281→   281→      setError(null);
   282→   282→    } catch (e) {
   283→   283→      const msg = errMessage(e);
   284→   284→      setError(msg);
   285→   285→      toast.error(msg);
   286→   286→    } finally {
   287→   287→      setLoading(false);
   288→   288→    }
   289→   289→  }, deps);
   290→   290→  useEffect(() => { void load(); }, [load]);
   291→   291→  return { data, loading, error, reload: load, setData };
   292→   292→}

   299→   299→  recordId?: string;

   305→   305→  module?: string;
   306→   306→  recordId?: string;

   308→   308→}): Promise<boolean> {
   309→   309→  try {
   310→   310→    await api.post("/api/undo", {
   311→   311→      ...(spec.auditLogId ? { auditLogId: spec.auditLogId } : { module: spec.module, recordId: spec.recordId }),
   312→   312→    });
   313→   313→    toast.success("Entry reversed — everything is back to how it was", { icon: "↩️" });
   314→   314→    spec.onUndo?.();
   315→   315→    return true;
   316→   316→  } catch (e) {

   322→   322→export function useMutation() {
   323→   323→  const [saving, setSaving] = useState(false);
   324→   324→  const mutate = useCallback(async (
   325→   325→    fn: () => Promise<unknown>,
   326→   326→    success?: string,
   327→   327→    undo?: (data: unknown) => UndoSpec | null | undefined
   328→   328→  ) => {
   329→   329→    setSaving(true);
   330→   330→    try {
   331→   331→      const data = await fn();
   332→   332→      if (success) {
   333→   333→        const spec = undo?.(data);
   334→   334→        if (spec?.module && spec.recordId) {
   335→   335→          toast.success(success, {
   336→   336→            duration: 8000,
   337→   337→            action: { label: "Undo", onClick: () => void undoRequest(spec) },
   338→   338→          });
   339→   339→        } else {
   340→   340→          toast.success(success);
   341→   341→        }
   342→   342→      }
   343→   343→      return { ok: true as const, data };
   344→   344→    } catch (e) {
   345→   345→      toast.error(errMessage(e));
   346→   346→      return { ok: false as const, error: e };
   347→   347→    } finally {
   348→   348→      setSaving(false);
   349→   349→    }
   350→   350→  }, []);
   351→   351→  return { mutate, saving };
   352→   352→}
