    51→    51→function AddAdjustmentDialog({ open, onOpenChange, employeeId, onDone }: {
    52→    52→  open: boolean; onOpenChange: (v: boolean) => void; employeeId: string; onDone: () => void;
    53→    53→}) {
    54→    54→  const [type, setType] = useState("BONUS");
    55→    55→  const [date, setDate] = useState(todayStr());
    56→    56→  const [amount, setAmount] = useState("");
    57→    57→  const [reason, setReason] = useState("");
    58→    58→  const { mutate, saving } = useMutation();
    59→    59→
    60→    60→  // Reset form each time the dialog opens (render-time state adjustment).
    61→    61→  const [prevOpen, setPrevOpen] = useState(open);
    62→    62→  if (open !== prevOpen) {
    63→    63→    setPrevOpen(open);
    64→    64→    if (open) { setType("BONUS"); setDate(todayStr()); setAmount(""); setReason(""); }
    65→    65→  }
    66→    66→
    67→    67→  const submit = async () => {
    68→    68→    const amt = parseAmount(amount);
    69→    69→    if (amt <= 0) { toast.error("Enter a valid amount"); return; }
    70→    70→    const signed = type === "DEDUCTION" || type === "PENALTY" ? -amt : amt;
    71→    71→    const res = await mutate(
    72→    72→      () => api.post("/api/adjustments", { employeeId, date, type, amount: signed, reason: reason || undefined }),
    73→    73→      "Adjustment recorded",
    74→    74→      (data) => ({ module: "ADJUSTMENT", recordId: (data as { id: string }).id, onUndo: onDone })
    75→    75→    );
    76→    76→    if (res.ok) { onOpenChange(false); onDone(); }
    77→    77→  };
    78→    78→
    79→    79→  return (
    80→    80→    <Dialog open={open} onOpenChange={onOpenChange}>
    81→    81→      <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-md">
    82→    82→        <DialogHeader>
    83→    83→          <DialogTitle>Add Adjustment</DialogTitle>
    84→    84→          <DialogDescription>Bonus/overtime add to payout; deduction/penalty subtract in settlements.</DialogDescription>
    85→    85→        </DialogHeader>
    86→    86→        <div className="grid gap-3 sm:grid-cols-2">
    87→    87→          <Field label="Type" required>
    88→    88→            <SelectInput value={type} onChange={setType} options={ADJ_TYPES} />
    89→    89→          </Field>
    90→    90→          <Field label="Amount (₹)" required>
    91→    91→            <MoneyInput value={amount} onChange={setAmount} min={1} className="h-10" />
    92→    92→          </Field>
    93→    93→          <Field label="Date" required>
    94→    94→            <Input type="date" value={date} onChange={(e) => setDate(e.target.value)} className="h-10" />
    95→    95→          </Field>
    96→    96→          <Field label="Reason">
    97→    97→            <Input value={reason} onChange={(e) => setReason(e.target.value)} className="h-10" placeholder="Extra shift, late mark…" />
    98→    98→          </Field>
    99→    99→        </div>
   100→   100→        <DialogFooter className="gap-2">
   101→   101→          <Button variant="outline" className="min-h-10 flex-1 sm:flex-none" onClick={() => onOpenChange(false)}>Cancel</Button>
   102→   102→          <Button className="min-h-10 flex-1 sm:flex-none" onClick={submit} disabled={saving}>{saving ? "Saving…" : "Add adjustment"}</Button>
   103→   103→        </DialogFooter>
   104→   104→      </DialogContent>
   105→   105→    </Dialog>
   106→   106→  );
   107→   107→}
