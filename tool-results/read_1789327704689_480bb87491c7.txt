    71→function EmailDeltaPill({ pct, goodUp }: { pct: number | null; goodUp: boolean }) {
    72→  if (pct === null) {
    73→    return <span className="rounded-full bg-primary/10 px-1.5 py-0.5 text-[9px] font-bold uppercase tracking-wide text-primary">new</span>;
    74→  }
    75→  if (pct === 0) {
    76→    return <span className="inline-flex items-center gap-0.5 rounded-full bg-muted px-1.5 py-0.5 text-[9px] font-semibold text-muted-foreground"><Minus className="h-2.5 w-2.5" aria-hidden />0%</span>;
    77→  }
    78→  const up = pct > 0;
    79→  const good = up === goodUp;
    80→  const Icon = up ? TrendingUp : TrendingDown;
    81→  return (
    82→    <span
    83→      className={cn(
    84→        "inline-flex items-center gap-0.5 rounded-full px-1.5 py-0.5 text-[9px] font-bold tabular-nums",
    85→        good
    86→          ? "bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-300"
    87→          : "bg-red-100 text-red-700 dark:bg-red-950 dark:text-red-300"
    88→      )}
    89→    >
    90→      <Icon className="h-2.5 w-2.5" aria-hidden />{up ? "+" : ""}{pct}%
    91→    </span>
    92→  );
    93→}

   259→function MyAccountCard() {
   260→  const { owner: me } = useAuth();
   261→  const [current, setCurrent] = useState("");
   262→  const [next, setNext] = useState("");
   263→  const [confirm, setConfirm] = useState("");
   264→  const { mutate, saving } = useMutation();
   265→
   266→  const submit = async () => {
   267→    if (!current || !next) {
   268→      toast.error("Fill in current and new password");
   269→      return;
   270→    }
   271→    if (next.length < 6) {
   272→      toast.error("New password must be at least 6 characters");
   273→      return;
   274→    }
   275→    if (next !== confirm) {
   276→      toast.error("New password and confirmation don't match");
   277→      return;
   278→    }
   279→    const res = await mutate(
   280→      () => api.post("/api/auth/change-password", { currentPassword: current, newPassword: next }),
   281→      "Password updated"
   282→    );
   283→    if (res.ok) {
   284→      setCurrent("");
   285→      setNext("");
   286→      setConfirm("");
   287→    }
   288→  };
   289→
   290→  return (
   291→    <Card className="border-primary/30">
   292→      <CardContent className="p-4">
   293→        <div className="flex flex-col gap-4 sm:flex-row sm:items-start">
   294→          <div className="flex min-w-0 flex-1 items-start gap-3">
   295→            <InitialAvatar name={me?.name} className="h-11 w-11 text-base" />
   296→            <div className="min-w-0">
   297→              <div className="flex flex-wrap items-center gap-1.5">
   298→                <p className="truncate text-sm font-bold">{me?.name ?? "My account"}</p>
   299→                <span className="inline-flex items-center gap-1 rounded-full bg-emerald-100 px-2 py-0.5 text-[10px] font-semibold text-emerald-700 dark:bg-emerald-950 dark:text-emerald-300">
   300→                  <ShieldCheck className="h-3 w-3" aria-hidden />You
   301→                </span>
   302→              </div>
   303→              <p className="truncate text-xs text-muted-foreground">@{me?.username ?? "—"}{me?.mobile ? ` · ${me.mobile}` : ""}</p>
   304→            </div>
   305→          </div>
   306→
   307→          <div className="w-full space-y-2 sm:max-w-sm">
   308→            <p className="text-xs font-semibold text-foreground">Change my password</p>
   309→            <Input
   310→              type="password"
   311→              value={current}
   312→              onChange={(e) => setCurrent(e.target.value)}
   313→              placeholder="Current password"
   314→              autoComplete="current-password"
   315→              className="h-10"
   316→            />
   317→            <div className="grid grid-cols-1 gap-2 sm:grid-cols-2">
   318→              <Input
   319→                type="password"
   320→                value={next}
   321→                onChange={(e) => setNext(e.target.value)}
   322→                placeholder="New password (min 6)"
   323→                autoComplete="new-password"
   324→                className="h-10"
   325→              />
   326→              <Input
   327→                type="password"
   328→                value={confirm}
   329→                onChange={(e) => setConfirm(e.target.value)}
   330→                placeholder="Confirm new password"
   331→                autoComplete="new-password"
   332→                className="h-10"
   333→              />
   334→            </div>
   335→            <Button className="min-h-10 w-full sm:w-auto" onClick={submit} disabled={saving}>
   336→              {saving ? "Updating…" : "Update password"}
   337→            </Button>
   338→          </div>
   339→        </div>
   340→      </CardContent>
   341→    </Card>
   342→  );
   343→}

   433→function OwnerFormDialog({ open, onOpenChange, target, onDone }: {
   434→  open: boolean;
   435→  onOpenChange: (v: boolean) => void;
   436→  target: OwnerRow | null; // null = add
   437→  onDone: () => void;
   438→}) {
   439→  const [form, setForm] = useState({ name: "", username: "", mobile: "", password: "" });
   440→  const { mutate, saving } = useMutation();
   441→
   442→  const [prevOpen, setPrevOpen] = useState(open);
   443→  if (open !== prevOpen) {
   444→    setPrevOpen(open);
   445→    if (open) {
   446→      setForm(target
   447→        ? { name: target.name, username: target.username, mobile: target.mobile ?? "", password: "" }
   448→        : { name: "", username: "", mobile: "", password: "" });
   449→    }
   450→  }
   451→
   452→  const set = (k: keyof typeof form) => (v: string) => setForm((f) => ({ ...f, [k]: v }));
   453→
   454→  const submit = async () => {
   455→    if (!form.name.trim()) {
   456→      toast.error("Name is required");
   457→      return;
   458→    }
   459→    if (!form.username.trim()) {
   460→      toast.error("Username is required");
   461→      return;
   462→    }
   463→    if (!target && form.password && form.password.length < 6) {
   464→      toast.error("Password must be at least 6 characters");
   465→      return;
   466→    }
   467→    const res = target
   468→      ? await mutate(
   469→          () => api.put(`/api/owners/${target.id}`, { name: form.name.trim(), mobile: form.mobile || undefined }),
   470→          "Owner updated"
   471→        )
   472→      : await mutate(
   473→          () => api.post("/api/owners", {
   474→            name: form.name.trim(),
   475→            username: form.username.trim(),
   476→            mobile: form.mobile || undefined,
   477→            password: form.password || undefined,
   478→          }),
   479→          "Owner added"
   480→        );
   481→    if (res.ok) {
   482→      onOpenChange(false);
   483→      onDone();
   484→    }
   485→  };
   486→
   487→  return (
   488→    <Dialog open={open} onOpenChange={onOpenChange}>
   489→      <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-md">
   490→        <DialogHeader>
   491→          <DialogTitle>{target ? "Edit owner" : "Add owner"}</DialogTitle>
   492→          <DialogDescription>
   493→            {target ? "Update the owner's display details." : "New owners can log in and manage the business."}
   494→          </DialogDescription>
   495→        </DialogHeader>
   496→        <div className="grid gap-3 sm:grid-cols-2">
   497→          <Field label="Name" required>
   498→            <Input value={form.name} onChange={(e) => set("name")(e.target.value)} className="h-10" placeholder="e.g. Rahul Sharma" />
   499→          </Field>
   500→          <Field label="Mobile" hint="Optional contact number">
   501→            <Input value={form.mobile} onChange={(e) => set("mobile")(e.target.value)} inputMode="tel" className="h-10" />
   502→          </Field>
   503→          <Field
   504→            label="Username"
   505→            required
   506→            className="sm:col-span-2"
   507→            hint={target ? "Username can't be changed" : "Used to log in — can't be changed later"}
   508→          >
   509→            <Input
   510→              value={form.username}
   511→              onChange={(e) => set("username")(e.target.value)}
   512→              className="h-10"
   513→              placeholder="e.g. rahul"
   514→              disabled={Boolean(target)}
   515→            />
   516→          </Field>
   517→          {!target && (
   518→            <Field label="Password" className="sm:col-span-2" hint="Leave blank to use the default: owner123">
   519→              <Input
   520→                type="password"
   521→                value={form.password}
   522→                onChange={(e) => set("password")(e.target.value)}
   523→                autoComplete="new-password"
   524→                className="h-10"
   525→                placeholder="Default: owner123"
   526→              />
   527→            </Field>
   528→          )}
   529→        </div>
   530→        <DialogFooter className="gap-2">
   531→          <Button variant="outline" className="min-h-10 flex-1 sm:flex-none" onClick={() => onOpenChange(false)}>Cancel</Button>
   532→          <Button className="min-h-10 flex-1 sm:flex-none" onClick={submit} disabled={saving}>
   533→            {saving ? "Saving…" : target ? "Save changes" : "Add owner"}
   534→          </Button>
   535→        </DialogFooter>
   536→      </DialogContent>
   537→    </Dialog>
   538→  );
   539→}

   541→// ---------------------------------------------------------------------------
   542→// Reset password dialog
   543→// ---------------------------------------------------------------------------
   544→
   545→function ResetPasswordDialog({ open, onOpenChange, target }: {
   546→  open: boolean;
   547→  onOpenChange: (v: boolean) => void;
   548→  target: OwnerRow | null;
   549→}) {
   550→  const [password, setPassword] = useState("");
   551→  const { mutate, saving } = useMutation();
   552→
   553→  const [prevOpen, setPrevOpen] = useState(open);
   554→  if (open !== prevOpen) {
   555→    setPrevOpen(open);
   556→    if (open) setPassword("");
   557→  }
   558→
   559→  const submit = async () => {
   560→    if (!target) return;
   561→    if (password.length < 6) {
   562→      toast.error("Password must be at least 6 characters");
   563→      return;
   564→    }
   565→    const res = await mutate(
   566→      () => api.put(`/api/owners/${target.id}`, { password }),
   567→      `Password reset for ${target.name}`
   568→    );
   569→    if (res.ok) {
   570→      setPassword("");
   571→      onOpenChange(false);
   572→    }
   573→  };
   574→
   575→  return (
   576→    <Dialog open={open} onOpenChange={onOpenChange}>
   577→      <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-sm">
   578→        <DialogHeader>
   579→          <DialogTitle>Reset password</DialogTitle>
   580→          <DialogDescription>
   581→            Set a new password for <span className="font-medium">@{target?.username ?? ""}</span>. Minimum 6 characters.
   582→          </DialogDescription>
   583→        </DialogHeader>
   584→        <Field label="New password" required>
   585→          <Input
   586→            type="password"
   587→            value={password}
   588→            onChange={(e) => setPassword(e.target.value)}
   589→            autoComplete="new-password"
   590→            className="h-10"
   591→            placeholder="Min 6 characters"
   592→          />
   593→        </Field>
   594→        <DialogFooter className="gap-2">
   595→          <Button variant="outline" className="min-h-10 flex-1 sm:flex-none" onClick={() => onOpenChange(false)}>Cancel</Button>
   596→          <Button className="min-h-10 flex-1 sm:flex-none" onClick={submit} disabled={saving}>
   597→            {saving ? "Saving…" : "Reset password"}
   598→          </Button>
   599→        </DialogFooter>
   600→      </DialogContent>
   601→    </Dialog>
   602→  );
   603→}

   609→export default function OwnersView(_props: ViewProps) {
   610→  const { owner: me } = useAuth();
   611→  const { data, loading, error, reload } = useAsync<ListResp<OwnerRow>>(
   612→    () => api.get<ListResp<OwnerRow>>("/api/owners"),
   613→    []
   614→  );
   615→
   616→  const [addOpen, setAddOpen] = useState(false);
   617→  const [editTarget, setEditTarget] = useState<OwnerRow | null>(null);
   618→  const [resetTarget, setResetTarget] = useState<OwnerRow | null>(null);
   619→  const [deactivateTarget, setDeactivateTarget] = useState<OwnerRow | null>(null);
   620→  const [emailTarget, setEmailTarget] = useState<OwnerRow | null>(null);
   621→  const { mutate, saving } = useMutation();
   622→
   623→  const owners = data?.items ?? [];
   624→
   625→  const activate = async (row: OwnerRow) => {
   626→    const res = await mutate(() => api.put(`/api/owners/${row.id}`, { isActive: true }), `${row.name} activated`);
   627→    if (res.ok) reload();
   628→  };
   629→
   630→  const deactivate = async () => {
   631→    if (!deactivateTarget) return;
   632→    const row = deactivateTarget;
   633→    setDeactivateTarget(null);
   634→    const res = await mutate(() => api.put(`/api/owners/${row.id}`, { isActive: false }), `${row.name} deactivated`);
   635→    if (res.ok) reload();
   636→  };
   637→
   638→  const onToggleActive = (row: OwnerRow, next: boolean) => {
   639→    if (next) void activate(row);
   640→    else setDeactivateTarget(row);
   641→  };
   642→
   643→  return (
   644→    <div className="space-y-4">
   645→      <PageHeader
   646→        title="Owners"
   647→        subtitle={`${owners.length} team account${owners.length === 1 ? "" : "s"}`}
   648→        icon={Crown}
   649→        actions={
   650→          <Button size="sm" className="h-9 gap-1.5" onClick={() => setAddOpen(true)}>
   651→            <Plus className="h-4 w-4" aria-hidden />Add Owner
   652→          </Button>
   653→        }
   654→      />
   655→
   656→      <MyAccountCard />
   657→
   658→      {error && (
   659→        <div className="rounded-xl border border-red-200 bg-red-50/60 p-4 text-center dark:border-red-900 dark:bg-red-950/30">
   660→          <p className="text-xs text-red-700 dark:text-red-300">{error}</p>
   661→          <Button variant="outline" size="sm" className="mt-2 h-8" onClick={() => void reload()}>Retry</Button>
   662→        </div>
   663→      )}
   664→
   665→      {loading && !error && (
   666→        <div className="grid gap-3 sm:grid-cols-2 xl:grid-cols-3">
   667→          {Array.from({ length: 6 }).map((_, i) => (
   668→            <Skeleton key={i} className="h-36 rounded-xl" />
   669→          ))}
   670→        </div>
   671→      )}
   672→
   673→      {!loading && !error && owners.length === 0 && (
   674→        <EmptyState
   675→          icon={UserPlus}
   676→          title="No owners yet"
   677→          description="Add team members so they can log in and help manage the business."
   678→          action={{ label: "Add Owner", onClick: () => setAddOpen(true) }}
   679→        />
   680→      )}
   681→
   682→      {!loading && !error && owners.length > 0 && (
   683→        <div className="grid min-w-0 gap-3 sm:grid-cols-2 xl:grid-cols-3 [&>*]:min-w-0">
   684→          {owners.map((o) => (
   685→            <OwnerCard
   686→              key={o.id}
   687→              owner={o}
   688→              isSelf={o.username === me?.username}
   689→              saving={saving}
   690→              onEdit={() => setEditTarget(o)}
   691→              onReset={() => setResetTarget(o)}
   692→              onToggleActive={(next) => onToggleActive(o, next)}
   693→              onEmail={() => setEmailTarget(o)}
   694→            />
   695→          ))}
   696→        </div>
   697→      )}
   698→
   699→      {/* Deactivate confirmation */}
   700→      <AlertDialog open={Boolean(deactivateTarget)} onOpenChange={(v) => !v && setDeactivateTarget(null)}>
   701→        <AlertDialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-md">
   702→          <AlertDialogHeader>
   703→            <AlertDialogTitle>Deactivate @{deactivateTarget?.username ?? ""}?</AlertDialogTitle>
   704→            <AlertDialogDescription>
   705→              {deactivateTarget?.name} will lose access immediately. You can reactivate the account anytime with the
   706→              switch on their card.
   707→            </AlertDialogDescription>
   708→          </AlertDialogHeader>
   709→          <AlertDialogFooter className="gap-2">
   710→            <AlertDialogCancel className="min-h-10 flex-1 sm:flex-none">Cancel</AlertDialogCancel>
   711→            <AlertDialogAction
   712→              className="min-h-10 flex-1 bg-red-600 text-white hover:bg-red-700 sm:flex-none"
   713→              onClick={(e) => {
   714→                e.preventDefault();
   715→                void deactivate();
   716→              }}
   717→            >
   718→              Deactivate
   719→            </AlertDialogAction>
   720→          </AlertDialogFooter>
   721→        </AlertDialogContent>
   722→      </AlertDialog>
   723→
   724→      <OwnerFormDialog
   725→        open={addOpen || Boolean(editTarget)}
   726→        onOpenChange={(v) => {
   727→          if (!v) {
   728→            setAddOpen(false);
   729→            setEditTarget(null);
   730→          }
   731→        }}
   732→        target={editTarget}
   733→        onDone={reload}
   734→      />
   735→
   736→      <ResetPasswordDialog
   737→        open={Boolean(resetTarget)}
   738→        onOpenChange={(v) => !v && setResetTarget(null)}
   739→        target={resetTarget}
   740→      />
   741→
   742→      {emailTarget && (
   743→        <MonthlyEmailDialog target={emailTarget} onClose={() => setEmailTarget(null)} />
   744→      )}
   745→    </div>
   746→  );
