    92→function rangeDates(r: RangeKey): { from: string; to: string } {
    93→  const now = new Date();
    94→  const f = (d: Date) => `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;
    95→  if (r === "today") { const t = f(now); return { from: t, to: t }; }
    96→  if (r === "yesterday") return { from: f(new Date(now.getFullYear(), now.getMonth(), now.getDate() - 1)), to: f(new Date(now.getFullYear(), now.getMonth(), now.getDate() - 1)) };
    97→  if (r === "week") return { from: f(new Date(now.getFullYear(), now.getMonth(), now.getDate() - 6)), to: f(now) };
    98→  if (r === "lastweek") return { from: f(new Date(now.getFullYear(), now.getMonth(), now.getDate() - 13)), to: f(new Date(now.getFullYear(), now.getMonth(), now.getDate() - 7)) };
    99→  if (r === "lastmonth") return { from: f(new Date(now.getFullYear(), now.getMonth() - 1, 1)), to: f(new Date(now.getFullYear(), now.getMonth(), 0)) };
   100→  return { from: f(new Date(now.getFullYear(), now.getMonth(), 1)), to: f(new Date(now.getFullYear(), now.getMonth() + 1, 0)) };
   101→}

   141→function ExpenseFormDialog({ open, onOpenChange, expense, vehicles, onDone, preset }: {
   142→  open: boolean;
   143→  onOpenChange: (v: boolean) => void;
   144→  expense: ExpenseRec | null;
   145→  vehicles: VehicleRec[];
   146→  onDone: () => void;
   147→  /** Optional pre-fill for quick actions (e.g. "Add deposit" opens the form with the Owner Contribution category). */
   148→  preset?: { categoryName?: string; spentById?: string } | null;
   149→}) {
   150→  const editing = Boolean(expense);
   151→  const [form, setForm] = useState({
   152→    date: todayStr(), business: "MANPOWER", categoryId: "", amount: "", method: "Cash",
   153→    description: "", vehicleId: "", notes: "", reason: "", spentById: "",
   154→  });
   155→  const [categories, setCategories] = useState<ExpenseCategoryRec[]>([]);
   156→  const [owners, setOwners] = useState<OwnerRec[]>([]);
   157→  const [meId, setMeId] = useState("");
   158→  const [newCatOpen, setNewCatOpen] = useState(false);
   159→  const [newCatName, setNewCatName] = useState("");
   160→  const { mutate, saving } = useMutation();
   161→
   162→  // Reset form each time the dialog opens (render-time state adjustment).
   163→  const [prevOpen, setPrevOpen] = useState(open);
   164→  if (open !== prevOpen) {
   165→    setPrevOpen(open);
   166→    if (open) {
   167→      setNewCatOpen(false);
   168→      setNewCatName("");
   169→      setForm({
   170→        date: expense?.date?.slice(0, 10) ?? todayStr(),
   171→        business: expense?.business ?? "MANPOWER",
   172→        categoryId: expense?.categoryId ?? "",
   173→        amount: expense ? String(expense.amount) : "",
   174→        method: expense?.method ?? "Cash",
   175→        description: expense?.description ?? "",
   176→        vehicleId: expense?.vehicleId ?? "",
   177→        notes: expense?.notes ?? "",
   178→        reason: expense?.reason ?? "",
   179→        spentById: expense ? (expense.isCommon ? "COMMON" : (expense.spentById ?? "")) : (preset?.spentById || ""),
   180→      });
   181→    }
   182→  }
   183→
   184→  useEffect(() => {
   185→    if (!open) return;
   186→    let cancelled = false;
   187→    api.get<{ items: ExpenseCategoryRec[] }>("/api/expense-categories" + qs({ business: form.business || undefined }))
   188→      .then((d) => { if (!cancelled) setCategories(d.items ?? []); })
   189→      .catch(() => { if (!cancelled) setCategories([]); });
   190→    return () => { cancelled = true; };
   191→  }, [open, form.business]);
   192→
   193→  // Owners (who is taking out the money) + the acting user (default attribution).
   194→  useEffect(() => {
   195→    if (!open) return;
   196→    let cancelled = false;
   197→    api.get<{ items: OwnerRec[] }>("/api/owners")
   198→      .then((d) => { if (!cancelled) setOwners(d.items ?? []); })
   199→      .catch(() => { /* non-fatal */ });
   200→    api.get<OwnerRec>("/api/auth/me")
   201→      .then((d) => { if (!cancelled) setMeId(d?.id ?? ""); })
   202→      .catch(() => { /* non-fatal */ });
   203→    return () => { cancelled = true; };
   204→  }, [open]);
   205→
   206→  // Quick-action preset: resolve the pre-filled category by NAME (render-time
   207→  // derivation — no setState inside an effect). Only for brand-new expenses.
   208→  const presetCategoryId = useMemo(() => {
   209→    if (editing || !preset?.categoryName) return "";
   210→    return categories.find((c) => c.name.toUpperCase() === preset.categoryName!.toUpperCase())?.id ?? "";
   211→  }, [editing, preset, categories]);
   212→
   213→  const set = (k: keyof typeof form) => (v: string) => setForm((f) => ({ ...f, [k]: v }));
   214→
   215→  const createCategory = async () => {
   216→    const name = newCatName.trim();
   217→    if (!name) { toast.error("Enter a category name"); return; }
   218→    const res = await mutate(
   219→      () => api.post<ExpenseCategoryRec>("/api/expense-categories", { name, business: form.business }),
   220→      `Category "${name}" created`
   221→    );
   222→    if (res.ok && res.data) {
   223→      setCategories((cs) => [...cs, res.data as ExpenseCategoryRec]);
   224→      setForm((f) => ({ ...f, categoryId: (res.data as ExpenseCategoryRec).id }));
   225→      setNewCatOpen(false);
   226→      setNewCatName("");
   227→    }
   228→  };
   229→
   230→  const submit = async () => {
   231→    const amt = parseAmount(form.amount);
   232→    if (amt <= 0) { toast.error("Enter a valid amount"); return; }
   233→    if (!form.date) { toast.error("Date is required"); return; }
   234→    const body = {
   235→      date: form.date,
   236→      business: form.business,
   237→      categoryId: form.categoryId || presetCategoryId || undefined,
   238→      amount: amt,
   239→      method: form.method || undefined,
   240→      description: form.description || undefined,
   241→      notes: form.notes || undefined,
   242→      reason: form.reason || undefined,
   243→      // "" → attribute to the acting user (API default); "COMMON" → shared by all owners.
   244→      spentById: form.spentById || meId || undefined,
   245→      vehicleId: form.business === "TRANSPORT" && form.vehicleId ? form.vehicleId : undefined,
   246→    };
   247→    const res = editing
   248→      ? await mutate(() => api.put(`/api/expenses/${expense!.id}`, body), "Expense updated", () => ({ module: "EXPENSE", recordId: expense!.id, onUndo: onDone }))
   249→      : await mutate(() => api.post("/api/expenses", body), "Expense recorded", (data) => ({ module: "EXPENSE", recordId: (data as { id: string }).id, onUndo: onDone }));
   250→    if (res.ok) { onOpenChange(false); onDone(); }
   251→  };
   252→
   253→  const selectedOwnerLabel = form.spentById === "COMMON"
   254→    ? "Common — all owners"
   255→    : (owners.find((o) => o.id === form.spentById)?.name ?? "You");
   256→
   257→  return (
   258→    <Dialog open={open} onOpenChange={onOpenChange}>
   259→      <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-lg">
   260→        <DialogHeader>
   261→          <DialogTitle>{editing ? "Edit Expense" : "Add Expense"}</DialogTitle>
   262→          <DialogDescription>Track who took the money, why, and which business it belongs to.</DialogDescription>
   263→        </DialogHeader>
   264→        <div className="grid gap-3 sm:grid-cols-2">
   265→          <Field label="Date" required>
   266→            <Input type="date" value={form.date} onChange={(e) => set("date")(e.target.value)} className="h-10" />
   267→          </Field>
   268→          <Field label="Business" required>
   269→            <SelectInput
   270→              value={form.business}
   271→              onChange={(v) => setForm((f) => ({ ...f, business: v, categoryId: "", vehicleId: "" }))}
   272→              options={[{ label: "Manpower", value: "MANPOWER" }, { label: "Transport", value: "TRANSPORT" }]}
   273→            />
   274→          </Field>
   275→          <Field
   276→            label="Paid by / Owner"
   277→            required
   278→            hint={form.spentById === "COMMON" ? "Shared equally by all owners & the business" : `Attributed to ${selectedOwnerLabel}`}
   279→            className="sm:col-span-2"
   280→          >
   281→            <SelectInput
   282→              value={form.spentById}
   283→              onChange={set("spentById")}
   284→              placeholder={meId ? "Select who is taking the money…" : "Loading…"}
   285→              options={[
   286→                { label: "Common — shared by all owners & business", value: "COMMON" },
   287→                ...owners.map((o) => ({ label: o.name, value: o.id })),
   288→              ]}
   289→            />
   290→          </Field>
   291→          <Field label="Reason" hint="Why the money was taken / spent" className="sm:col-span-2">
   292→            <Input
   293→              value={form.reason}
   294→              onChange={(e) => set("reason")(e.target.value)}
   295→              className="h-10"
   296→              placeholder="e.g. Diesel for generator · Salary withdrawal · Office tea"
   297→            />
   298→          </Field>
   299→          <Field label="Category">
   300→            <SelectInput
   301→              value={form.categoryId || presetCategoryId}
   302→              onChange={set("categoryId")}
   303→              placeholder="Select category…"
   304→              options={categories.map((c) => ({ label: c.name, value: c.id }))}
   305→            />
   306→          </Field>
   307→          <div className="sm:col-span-1 sm:row-start-auto">
   308→            {!newCatOpen ? (
   309→              <Button
   310→                type="button" variant="outline" size="sm"
   311→                className="mt-1 h-9 w-full gap-1 text-xs sm:mt-6"
   312→                onClick={() => setNewCatOpen(true)}
   313→              >
   314→                <Plus className="h-3.5 w-3.5" aria-hidden />New category
   315→              </Button>
   316→            ) : (
   317→              <div className="mt-1 flex gap-1.5 sm:mt-6">
   318→                <Input
   319→                  value={newCatName}
   320→                  onChange={(e) => setNewCatName(e.target.value)}
   321→                  onKeyDown={(e) => { if (e.key === "Enter") { e.preventDefault(); void createCategory(); } }}
   322→                  className="h-9 text-xs"
   323→                  placeholder={`Category for ${form.business === "TRANSPORT" ? "Transport" : "Manpower"}`}
   324→                  aria-label="New category name"
   325→                  autoFocus
   326→                />
   327→                <Button type="button" size="sm" className="h-9 shrink-0 px-2.5" onClick={() => void createCategory()} disabled={saving} aria-label="Create category">
   328→                  <Check className="h-3.5 w-3.5" aria-hidden />
   329→                </Button>
   330→                <Button type="button" size="sm" variant="ghost" className="h-9 shrink-0 px-2.5" onClick={() => { setNewCatOpen(false); setNewCatName(""); }} aria-label="Cancel category creation">
   331→                  ✕
   332→                </Button>
   333→              </div>
   334→            )}
   335→          </div>
   336→          <Field label="Amount (₹)" required>
   337→            <MoneyInput value={form.amount} onChange={set("amount")} min={1} className="h-10" />
   338→          </Field>
   339→          <Field label="Method">
   340→            <SelectInput value={form.method} onChange={set("method")} options={METHOD_OPTIONS} />

   345→                value={form.vehicleId}
   346→                onChange={set("vehicleId")}

   352→          <Field label="Description" className="sm:col-span-2">
   353→            <Input value={form.description} onChange={(e) => set("description")(e.target.value)} className="h-10" placeholder="What was this expense for?" />
   354→          </Field>
   355→          <Field label="Notes" className="sm:col-span-2">
   356→            <Textarea value={form.notes} onChange={(e) => set("notes")(e.target.value)} rows={2} placeholder="Optional" />
   357→          </Field>
   358→        </div>
   359→        <DialogFooter className="gap-2">
   360→          <Button variant="outline" className="min-h-10 flex-1 sm:flex-none" onClick={() => onOpenChange(false)}>Cancel</Button>
   361→          <Button className="min-h-10 flex-1 sm:flex-none" onClick={submit} disabled={saving}>
   362→            {saving ? "Saving…" : editing ? "Save changes" : "Add expense"}
   363→          </Button>
   364→        </DialogFooter>
   365→      </DialogContent>
   366→    </Dialog>
   367→  );
   368→}
