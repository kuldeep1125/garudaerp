   664→  1263→  1263→            )}

   668→  1267→  1267→              </Button>
   669→  1268→  1268→            ) : (

   671→  1270→  1270→                <Users className="h-4 w-4" aria-hidden />
   672→  1271→  1271→                {creating ? "Deploying…" : `Deploy ${selected.length} employee${selected.length === 1 ? "" : "s"}`}
   673→  1272→  1272→              </Button>
   674→  1273→  1273→            )}
   675→  1274→  1274→          </div>
   676→  1275→  1275→        </div>
   677→  1276→  1276→      </DialogContent>
   678→  1277→  1277→    </Dialog>
   679→  1278→  1278→  );
   680→  1279→  1279→}

   737→  1336→  1336→    if (!form.propertyId) { toast.error("Select a property"); return; }

   741→  1340→  1340→    const p = parseAmount(form.payoutRate);

   744→  1343→  1343→      propertyId: form.propertyId,
   745→  1344→  1344→      name: form.name.trim(),
   746→  1345→  1345→      startDate: form.startDate,
   747→  1346→  1346→      endDate: form.endDate || undefined,
   748→  1347→  1347→      billingRate: b,
   749→  1348→  1348→      payoutRate: p,
   750→  1349→  1349→      shift: form.shift || undefined,
   751→  1350→  1350→      category: form.category || undefined,

   753→  1352→  1352→      paymentTerms: form.paymentTerms || undefined,
   754→  1353→  1353→      notes: form.notes || undefined,
   755→  1354→  1354→    };

   759→  1358→  1358→    if (res.ok) { onOpenChange(false); onDone(); }
   760→  1359→  1359→  };
