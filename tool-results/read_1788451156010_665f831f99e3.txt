    41→// Desktop sidebar (collapsible) — also the content of the mobile "More" sheet.
    42→function SideNav({ collapsed, onToggle, onNavigate }: { collapsed?: boolean; onToggle?: () => void; onNavigate?: () => void }) {
    43→  const { view, navigate } = useNav();
    44→  const { scope } = useBusiness();
    45→  const { lang } = useLang();
    46→

    48→    const order = ["MAIN", "MANPOWER", "TRANSPORT", "SYSTEM"] as const;

    50→      key: g,

    52→    })).filter((g) => g.items.length > 0);

    56→    <nav aria-label="Main navigation" className="flex h-full flex-col">

    58→        <button
    59→          className="flex items-center gap-2.5 min-w-0"
    60→          onClick={() => { navigate("dashboard"); onNavigate?.(); }}
    61→          aria-label="Go to dashboard"
    62→        >
    63→          <div className="flex h-9 w-9 shrink-0 items-center justify-center rounded-xl bg-emerald-600 text-white font-bold shadow-sm">
    64→            <span className="text-sm">B</span>
    65→          </div>
    66→          {!collapsed && (
    67→            <div className="min-w-0 text-left">
    68→              <p className="truncate text-sm font-bold leading-tight">BizHub</p>
    69→              <p className="truncate text-[10px] text-muted-foreground leading-tight">Business Control Center</p>
    70→            </div>
    71→          )}
    72→        </button>
    73→        {onToggle && !collapsed && (
    74→          <Button variant="ghost" size="icon" className="hidden lg:inline-flex h-8 w-8" onClick={onToggle} aria-label="Collapse sidebar">
    75→            <ChevronsLeft className="h-4 w-4" />

    89→              )}
    90→              <ul className="space-y-0.5">
    91→                {group.items
    92→                  .filter((v) => scope === "ALL" || v.group === "MAIN" || v.group === "SYSTEM" || v.group === scope)
    93→                  .map((v) => {
    94→                    const active = view === v.id || (v.id === "manpower" && view === "manpower") || (v.id === "transport" && view === "transport");
    95→                    return (
    96→                      <li key={v.id}>
    97→                        <button
    98→                          onClick={() => { navigate(v.id); onNavigate?.(); }}
    99→                          aria-current={active ? "page" : undefined}
   100→                          className={cn(
   101→                            "flex w-full items-center gap-2.5 rounded-lg px-2.5 py-2 text-[13px] transition-colors min-h-[40px]",
   102→                            collapsed && "justify-center px-0",
   103→                            active
   104→                              ? "bg-primary/12 text-primary font-semibold"
   105→                              : "text-muted-foreground font-medium hover:bg-muted hover:text-foreground"
   106→                          )}
   107→                          title={collapsed ? viewLabel(lang, v.id, v.label) : undefined}
   108→                        >
   109→                          <v.icon className={cn("h-4 w-4 shrink-0", active && GROUP_ACCENT[group.key], !active && "text-muted-foreground")} aria-hidden />

   115→              </ul>

   157→// Sticky top header
   158→function TopBar({ onOpenMore, onOpenPalette, onOpenShortcuts }: { onOpenMore: () => void; onOpenPalette: () => void; onOpenShortcuts: () => void }) {
   159→  const { owner, logout } = useAuth();
   160→  const { navigate } = useNav();
   161→  const { lang, setLang } = useLang();
   162→  const [notifCount, setNotifCount] = useState(0);
   163→  const [pulse, setPulse] = useState(false);
   164→  const [undoOpen, setUndoOpen] = useState(false);
   165→  const [notifRefresh, setNotifRefresh] = useState(0);
   166→  const { canInstall, install } = usePwaInstall();
   167→  const prevKeysRef = useRef<Set<string> | null>(null);
   168→  const firstLoadRef = useRef(true);
   169→  const pulseTimerRef = useRef<number | null>(null);
   170→
   171→  useEffect(() => {
   172→    let alive = true;
   173→    const load = async () => {
   174→      try {
   175→        const items = await import("@/lib/api-client").then(({ api }) =>
   176→          api.get<AppNotification[]>("/api/notifications")
   177→        );
   178→        if (!alive || !Array.isArray(items)) return;
   179→        // Respect muted categories (localStorage) — the bell only counts alerts
   180→        // the owner hasn't muted from the notifications view.
   181→        const muted = readMutedGroups();
   182→        const visible = muted.length > 0 ? items.filter((n) => !muted.includes(n.severity)) : items;
   183→        setNotifCount(visible.length);
   184→        const prevKeys = prevKeysRef.current;
   185→        prevKeysRef.current = new Set(items.map((n) => n.key));
   186→        // Announce only genuine arrivals (skip initial load), when the tab is
   187→        // visible and the alert isn't muted.
   188→        if (!firstLoadRef.current && prevKeys && document.visibilityState === "visible") {
   189→          const fresh = visible.filter((n) => !prevKeys.has(n.key));
   190→          if (fresh.length > 0) {
   191→            const rank = { CRITICAL: 0, WARNING: 1, INFO: 2 } as const;
   192→            const headline =
   193→              [...fresh].sort((a, b) => rank[a.severity] - rank[b.severity])[0] ?? fresh[0];
   194→            toast(headline.title, {
   195→              description:
   196→                fresh.length > 1
   197→                  ? `${headline.message} — and ${fresh.length - 1} more new alert${fresh.length === 2 ? "" : "s"}.`
   198→                  : headline.message,
   199→              icon: SEVERITY_ICON[headline.severity] ?? SEVERITY_ICON.INFO,
   200→              action: { label: "View", onClick: () => navigate("notifications") },
   201→            });
   202→            // Bell ripple — brief ping so the eye is drawn to the new alert.
   203→            setPulse(true);
   204→            if (pulseTimerRef.current) window.clearTimeout(pulseTimerRef.current);
   205→            pulseTimerRef.current = window.setTimeout(() => setPulse(false), 2400);
   206→          }
   207→        }
   208→        firstLoadRef.current = false;
   209→      } catch { /* ignore */ }
   210→    };
   211→    load();
   212→    const t = setInterval(load, 60_000);
   213→    // Re-check immediately when the user returns to the tab.
   214→    const onVis = () => { if (document.visibilityState === "visible") void load(); };
   215→    // Mute toggles + dismissals (this tab via custom event, other tabs via
   216→    // storage) re-count instantly.
   217→    const onMuteChange = () => { void load(); };
   218→    document.addEventListener("visibilitychange", onVis);
   219→    window.addEventListener(NOTIF_MUTED_EVENT, onMuteChange);
   220→    window.addEventListener(NOTIF_CHANGED_EVENT, onMuteChange);
   221→    window.addEventListener("storage", onMuteChange);
   222→    return () => {
   223→      alive = false; clearInterval(t);
   224→      document.removeEventListener("visibilitychange", onVis);
   225→      window.removeEventListener(NOTIF_MUTED_EVENT, onMuteChange);
   226→      window.removeEventListener(NOTIF_CHANGED_EVENT, onMuteChange);
   227→      window.removeEventListener("storage", onMuteChange);
   228→      if (pulseTimerRef.current) window.clearTimeout(pulseTimerRef.current);
   229→    };
   230→  }, [navigate, notifRefresh]);
   231→
   232→  const initials = (owner?.name ?? "?").slice(0, 2).toUpperCase();
   233→
   234→  return (
   235→    <>
   236→    <header className="sticky top-0 z-40 border-b bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/80">
   237→      <div className="mx-auto flex h-14 max-w-[1600px] items-center gap-2 px-3 sm:px-5">
   238→        <button className="flex items-center gap-2 md:hidden" onClick={() => navigate("dashboard")} aria-label="BizHub home">
   239→          <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-emerald-600 text-white font-bold">
   240→            <span className="text-xs">B</span>
   241→          </div>
   242→          <span className="text-sm font-bold">BizHub</span>
   243→        </button>
   244→
   245→        <Button variant="ghost" size="icon" className="hidden lg:inline-flex h-9 w-9 lg:hidden" aria-hidden tabIndex={-1} />
   246→
   247→        {/* Global search trigger — opens the command palette (Ctrl K) */}
   248→        <button
   249→          onClick={onOpenPalette}
   250→          className="ml-0 hidden sm:flex h-9 min-w-0 flex-1 max-w-md items-center gap-2 rounded-xl border bg-muted/40 px-3 text-sm text-muted-foreground hover:bg-muted hover:border-primary/40 transition-colors"
   251→          aria-label="Open command palette"
   252→        >
   253→          <Search className="h-4 w-4 shrink-0" />
   254→          <span className="min-w-0 truncate whitespace-nowrap hidden md:inline">{t(lang, "topbar.search")}</span>
   255→          <span className="md:hidden whitespace-nowrap">{t(lang, "topbar.searchShort")}</span>
   256→          <kbd className="ml-auto hidden md:inline-flex shrink-0 rounded border bg-background px-1.5 font-mono text-[10px]">Ctrl K</kbd>
   257→        </button>
   258→
   259→        <div className="ml-auto flex items-center gap-0.5 sm:gap-1">
   260→          <Button variant="ghost" size="icon" className="sm:hidden h-10 w-10" onClick={onOpenPalette} aria-label="Open command palette">
   261→            <Search className="h-5 w-5" />
   262→          </Button>
   263→          <Button
   264→            variant="ghost"
   265→            size="icon"
   266→            className="h-10 w-10"
   267→            onClick={() => setUndoOpen(true)}
   268→            aria-label="Undo center"
   269→            title="Undo center"
   270→          >
   271→            <History className="h-5 w-5" />
   272→          </Button>
   273→          <Button variant="ghost" size="icon" className="h-10 w-10 relative" onClick={() => navigate("notifications")} aria-label={`Notifications${notifCount ? `, ${notifCount} active` : ""}`}>
   274→            {pulse && <span className="absolute inset-0 rounded-full bg-red-500/30 animate-ping" aria-hidden />}
   275→            <Bell className={cn("h-5 w-5 transition-transform", pulse && "scale-110")} />
   276→            {notifCount > 0 && (
   277→              <span
   278→                className="absolute right-0.5 top-0.5 flex h-4 min-w-4 items-center justify-center rounded-full bg-red-500 px-1 text-[9px] font-bold tabular-nums text-white ring-2 ring-background"
   279→                aria-hidden
   280→              >
   281→                {notifCount > 9 ? "9+" : notifCount}
   282→              </span>
   283→            )}
   284→          </Button>
   285→          <ThemeToggle />
   286→          <DropdownMenu>
   287→            <DropdownMenuTrigger asChild>
   288→              <Button variant="ghost" size="icon" className="h-10 w-10" aria-label={t(lang, "topbar.language")} title={t(lang, "topbar.language")}>
   289→                <Languages className="h-5 w-5" />
   290→              </Button>
   291→            </DropdownMenuTrigger>
   292→            <DropdownMenuContent align="end" className="w-40">
   293→              <DropdownMenuLabel>{t(lang, "topbar.language")}</DropdownMenuLabel>
   294→              <DropdownMenuSeparator />
   295→              <DropdownMenuItem onClick={() => setLang("en")} aria-checked={lang === "en"} role="menuitemradio">
   296→                {lang === "en" ? <Check className="mr-2 h-4 w-4 text-primary" /> : <span className="mr-2 inline-block h-4 w-4" aria-hidden />}
   297→                {t(lang, "lang.en")}
   298→              </DropdownMenuItem>
   299→              <DropdownMenuItem onClick={() => setLang("hi")} aria-checked={lang === "hi"} role="menuitemradio">
   300→                {lang === "hi" ? <Check className="mr-2 h-4 w-4 text-primary" /> : <span className="mr-2 inline-block h-4 w-4" aria-hidden />}
   301→                {t(lang, "lang.hi")}
   302→              </DropdownMenuItem>
   303→            </DropdownMenuContent>
   304→          </DropdownMenu>
   305→          <DropdownMenu>
   306→            <DropdownMenuTrigger asChild>
   307→              <button className="ml-0.5 flex items-center gap-2 rounded-full outline-none focus-visible:ring-2 focus-visible:ring-ring" aria-label="Owner menu">
   308→                <Avatar className="h-9 w-9 border">
   309→                  <AvatarFallback className="bg-emerald-100 text-emerald-800 text-xs font-bold dark:bg-emerald-900 dark:text-emerald-200">{initials}</AvatarFallback>
   310→                </Avatar>
   311→              </button>
   312→            </DropdownMenuTrigger>
   313→            <DropdownMenuContent align="end" className="w-52">
   314→              <DropdownMenuLabel>
   315→                <p className="text-sm font-semibold">{owner?.name}</p>
   316→                <p className="text-xs text-muted-foreground">@{owner?.username}</p>
   317→              </DropdownMenuLabel>
   318→              <DropdownMenuSeparator />
   319→              <DropdownMenuItem onClick={() => navigate("owners")}>
   320→                <ShieldCheck className="mr-2 h-4 w-4" /> {t(lang, "menu.owners")}
   321→              </DropdownMenuItem>
   322→              <DropdownMenuItem onClick={() => navigate("settings")}>
   323→                <Building2 className="mr-2 h-4 w-4" /> {t(lang, "menu.settings")}
   324→              </DropdownMenuItem>
   325→              {canInstall && (
   326→                <DropdownMenuItem
   327→                  onClick={() =>
   328→                    install().then((r) => {
   329→                      if (r === "accepted") toast.success("BizHub installed — find it on your home screen");
   330→                    })
   331→                  }
   332→                >
   333→                  <Smartphone className="mr-2 h-4 w-4" /> {t(lang, "menu.install")}
   334→                </DropdownMenuItem>
   335→              )}
   336→              <DropdownMenuSeparator />
   337→              <DropdownMenuItem onClick={onOpenShortcuts}>
   338→                <Keyboard className="mr-2 h-4 w-4" /> Press ? for shortcuts
   339→              </DropdownMenuItem>

   344→            </DropdownMenuContent>

   347→      </div>

   355→    </>

   420→            <button
   421→              key={item.key}
   422→              onClick={item.action}
   423→              aria-current={active ? "page" : undefined}
   424→              className={cn(
   425→                "flex min-h-[56px] flex-col items-center justify-center gap-0.5 px-1 py-1.5 text-[10px] font-medium transition-colors active:scale-95",
   426→                active ? "text-primary" : "text-muted-foreground hover:text-foreground"
   427→              )}
   428→            >

   462→export function AppShell({ children }: { children: React.ReactNode }) {
   463→  const { navigate, view } = useNav();
   464→  const [collapsed, setCollapsed] = useState(false);
   465→  const [moreOpen, setMoreOpen] = useState(false);
   466→  const [paletteOpen, setPaletteOpen] = useState(false);
   467→  const [shortcutsOpen, setShortcutsOpen] = useState(false);
   468→
   469→  useEffect(() => {
   470→    // "G then key" chord: pendingG holds the timeout id while waiting for the
   471→    // second key (1s window). Cleared by typing anywhere else.
   472→    let pendingG: number | null = null;
   473→    const clearPendingG = () => {
   474→      if (pendingG !== null) window.clearTimeout(pendingG);
   475→      pendingG = null;
   476→    };
   477→    const GO_CHORDS: Record<string, string> = {
   478→      d: "dashboard", e: "employees", p: "properties", t: "trips",
   479→    };
   480→    const isTypingTarget = (el: EventTarget | null): boolean => {
   481→      const t = el as HTMLElement | null;
   482→      return Boolean(
   483→        t &&
   484→        (t.tagName === "INPUT" || t.tagName === "TEXTAREA" || t.tagName === "SELECT" || t.isContentEditable)
   485→      );
   486→    };
   487→    const onKey = (e: KeyboardEvent) => {
   488→      if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === "k") {
   489→        e.preventDefault();
   490→        clearPendingG();
   491→        setPaletteOpen((o) => !o);
   492→        return;
   493→      }
   494→      // Palette is open → its input owns the keyboard; ignore global chords.
   495→      if (paletteOpen) return;
   496→      if (e.altKey || e.ctrlKey || e.metaKey) return;
   497→      // Typing in a field always cancels any pending chord.
   498→      if (isTypingTarget(e.target)) {
   499→        clearPendingG();
   500→        return;
   501→      }
   502→      if (pendingG !== null) {
   503→        const target = GO_CHORDS[e.key.toLowerCase()];
   504→        clearPendingG();
   505→        if (target) {
   506→          e.preventDefault();
   507→          navigate(target);
   508→        }
   509→        return;
   510→      }
   511→      if (e.key === "g" || e.key === "G") {
   512→        pendingG = window.setTimeout(clearPendingG, 1000);
   513→        return;
   514→      }
   515→      if (e.key === "?") {
   516→        e.preventDefault();
   517→        setShortcutsOpen(true);
   518→      }
   519→    };
   520→    window.addEventListener("keydown", onKey);
   521→    return () => {
   522→      window.removeEventListener("keydown", onKey);
   523→      clearPendingG();
   524→    };
   525→  }, [navigate, paletteOpen]);
   526→
   527→  const currentView = getView(view);
   528→  const isWide = ["dashboard", "manpower", "transport"].includes(view);
   529→
   530→  return (
   531→    <div className="flex min-h-screen bg-muted/30">
   532→      {/* Desktop sidebar */}
   533→      <aside className={cn("sticky top-0 hidden h-screen shrink-0 border-r bg-background transition-all md:block", collapsed ? "w-[68px]" : "w-60")}>
   534→        <SideNav collapsed={collapsed} onToggle={() => setCollapsed((c) => !c)} />
   535→      </aside>
   536→
   537→      <div className="flex min-w-0 flex-1 flex-col">
   538→        <TopBar onOpenMore={() => setMoreOpen(true)} onOpenPalette={() => setPaletteOpen(true)} onOpenShortcuts={() => setShortcutsOpen(true)} />
   539→        <BusinessBanner />
   540→        <main
   541→          id="main-scroll"
   542→          className={cn("mx-auto w-full max-w-[1600px] flex-1 px-3 pb-24 pt-3 sm:px-5 sm:pt-4 md:pb-10", !isWide && "max-w-[1400px]")}
   543→          aria-label={currentView?.label ?? "Content"}
   544→        >
   545→          {/* Keyed by view → remounts on navigation and plays the enter animation */}
   546→          <div key={view} className="view-enter">
   547→            {children}
   548→          </div>
   549→        </main>
   550→        <footer className="mt-auto hidden border-t bg-background py-3 md:block">
   551→          <div className="mx-auto flex max-w-[1600px] items-center justify-between px-5 text-[11px] text-muted-foreground">
   552→            <p>BizHub — one control center for Manpower & Transport businesses.</p>
   553→            <p className="tabular-nums">{new Date().getFullYear()} · All financial actions are audited.</p>
   554→          </div>
   555→        </footer>
   556→      </div>
   557→
   558→      {/* Mobile bottom nav */}
   559→      <BottomNav onOpenMore={() => setMoreOpen(true)} />
   560→
   561→      {/* Global command palette (Ctrl/Cmd+K, search buttons) */}
   562→      <CommandPalette open={paletteOpen} onOpenChange={setPaletteOpen} />
   563→
   564→      {/* Keyboard shortcuts overlay (? key or owner menu) */}
   565→      <ShortcutsDialog open={shortcutsOpen} onOpenChange={setShortcutsOpen} />
   566→
   567→      {/* Mobile "More" sheet with full nav */}
   568→      <Sheet open={moreOpen} onOpenChange={setMoreOpen}>
   569→        <SheetTrigger className="hidden" aria-hidden />
   570→        <SheetContent side="left" className="w-[280px] p-0">
   571→          <SheetHeader className="sr-only">
   572→            <SheetTitle>All navigation</SheetTitle>
   573→          </SheetHeader>
   574→          <SideNav onNavigate={() => setMoreOpen(false)} />
   575→        </SheetContent>
   576→      </Sheet>
   577→    </div>
   578→  );
   579→}
