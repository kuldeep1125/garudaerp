   265→export function errMessage(e: unknown): string {
   266→  if (e instanceof ApiError) return e.message;
   267→  if (e instanceof Error) return e.message;
   268→  return "Something went wrong";

   271→/** Fetch-on-mount hook with loading/error/reload. Toasts load failures. */
   272→export function useAsync<T>(loader: () => Promise<T>, deps: React.DependencyList) {
   273→  const [data, setData] = useState<T | null>(null);
   274→  const [loading, setLoading] = useState(true);
   275→  const [error, setError] = useState<string | null>(null);
   276→  const load = useCallback(async () => {
   277→    setLoading(true);
   278→    try {
   279→      const d = await loader();
   280→      setData(d);
   281→      setError(null);
   282→    } catch (e) {
   283→      const msg = errMessage(e);
   284→      setError(msg);
   285→      toast.error(msg);
   286→    } finally {
   287→      setLoading(false);
   288→    }
   289→  }, deps);
   290→  useEffect(() => { void load(); }, [load]);
   291→  return { data, loading, error, reload: load, setData };
   292→}

   294→/** Mutation wrapper: consistent error toasts, optional success toast. */
   295→export function useMutation() {
   296→  const [saving, setSaving] = useState(false);
   297→  const mutate = useCallback(async (fn: () => Promise<unknown>, success?: string) => {
   298→    setSaving(true);
   299→    try {
   300→      const data = await fn();
   301→      if (success) toast.success(success);
   302→      return { ok: true as const, data };
   303→    } catch (e) {
   304→      toast.error(errMessage(e));
   305→      return { ok: false as const, error: e };
   306→    } finally {
   307→      setSaving(false);
   308→    }
   309→  }, []);
   310→  return { mutate, saving };
   311→}

   313→export function useDebounced<T>(value: T, delay = 300): T {
   314→  const [v, setV] = useState(value);
   315→  useEffect(() => {
   316→    const t = setTimeout(() => setV(value), delay);
   317→    return () => clearTimeout(t);
   318→  }, [value, delay]);
   319→  return v;
   320→}

   322→export function todayStr(offsetDays = 0): string {
   323→  const d = new Date();
   324→  d.setDate(d.getDate() + offsetDays);

   328→export function fmtDay(s?: string | null): string {
   329→  if (!s) return "—";
   330→  const d = new Date(s.length === 10 ? `${s}T00:00:00` : s);
   331→  if (Number.isNaN(d.getTime())) return s;
   332→  return d.toLocaleDateString("en-IN", { day: "2-digit", month: "short", year: "numeric" });
   333→}
   334→
   335→export function fmtDateTime(s?: string | null): string {
   336→  if (!s) return "—";
   337→  const d = new Date(s);
   338→  if (Number.isNaN(d.getTime())) return s;
   339→  return d.toLocaleString("en-IN", { day: "2-digit", month: "short", hour: "2-digit", minute: "2-digit" });
   340→}

   342→export function initials(name?: string | null): string {
   343→  if (!name) return "?";
   344→  const parts = name.trim().split(/\s+/).slice(0, 2);
   345→  return parts.map((p) => p[0]?.toUpperCase() ?? "").join("") || "?";
   346→}

   527→/** Div-based horizontal bar list (category breakdown etc). */
   528→export function MiniBars({ rows }: { rows: { label: string; value: number; color?: string }[] }) {
   529→  const max = Math.max(1, ...rows.map((r) => r.value));
   530→  if (rows.length === 0) return <p className="py-4 text-center text-xs text-muted-foreground">No data</p>;
   531→  return (
   532→    <div className="space-y-2.5">
   533→      {rows.map((r) => (
   534→        <div key={r.label}>
   535→          <div className="mb-1 flex items-center justify-between gap-2 text-xs">
   536→            <span className="min-w-0 truncate font-medium">{r.label}</span>
   537→            <span className="shrink-0 tabular-nums text-muted-foreground">{formatINR(r.value)}</span>
   538→          </div>
   539→          <div className="h-2 w-full overflow-hidden rounded-full bg-muted">
   540→            <div
   541→              className="h-full rounded-full transition-all"
   542→              style={{ width: `${Math.max(3, (r.value / max) * 100)}%`, background: r.color ?? CHART_COLORS.emerald }}
   543→            />
   544→          </div>
   545→        </div>
   546→      ))}
   547→    </div>
   548→  );
   549→}

  1118→                <Button

  1120→                  variant="outline"

  1122→                  onClick={() => submit(remainingAfterSkip)}

  1163→                resolved?.billingRate != null

  1189→                    variant="ghost"
  1190→                    size="sm"

  1194→                    {selected.length === employees.length ? "Deselect all" : "Select all"}

  1199→            )
  1200→          )}
  1201→
  1202→          {step === 3 && (
  1203→            <>
  1204→              <div className="rounded-xl border bg-muted/40 px-3.5 py-2.5 text-xs">

  1206→                <p className="mt-0.5 text-muted-foreground">
  1207→                  {fmtDay(date)} · {shift} shift{workCategory ? ` · ${workCategory}` : ""} · {selected.length} employee{selected.length === 1 ? "" : "s"}
  1208→                </p>

  1210→              <div className="overflow-hidden rounded-xl border">
  1211→                <div className="grid grid-cols-[1fr_auto_auto] gap-2 border-b bg-muted/50 px-3 py-2 text-[10px] font-semibold uppercase tracking-wide text-muted-foreground">

  1235→                        aria-label={`Payout rate for ${e.fullName}`}

  1246→        <div className="border-t bg-card px-5 py-3">

  1274→          </div>

  1285→export function ContractFormDialog({ open, onOpenChange, contract, defaultPropertyId, properties: propsProp, onDone }: {
  1286→  open: boolean;
  1287→  onOpenChange: (v: boolean) => void;
  1288→  contract?: ContractRec | null;
  1289→  defaultPropertyId?: string;
  1290→  properties?: { id: string; name: string }[];
  1291→  onDone: () => void;
  1292→}) {
  1293→  const editing = Boolean(contract);
  1294→  const [fetchedProps, setFetchedProps] = useState<{ id: string; name: string }[]>([]);
  1295→  const properties = propsProp ?? fetchedProps;
  1296→  const [form, setForm] = useState({
  1297→    propertyId: defaultPropertyId ?? "", name: "", startDate: todayStr(), endDate: "",
  1298→    billingRate: "", payoutRate: "", shift: "DAY", category: "", maxEmployees: "",
  1299→    paymentTerms: "", notes: "",
  1300→  });
  1301→  const { mutate, saving } = useMutation();
  1302→
  1303→  useEffect(() => {
  1304→    if (propsProp || properties.length > 0) return;
  1305→    let cancelled = false;
  1306→    api.get<ListResp<PropertyRec>>("/api/properties" + qs({ pageSize: 200 }))
  1307→      .then((d) => { if (!cancelled) setFetchedProps(d.items.map((p) => ({ id: p.id, name: p.name }))); })
  1308→      .catch(() => {});
  1309→    return () => { cancelled = true; };
  1310→  }, [propsProp]);
  1311→
  1312→  // Reset form each time the dialog opens (render-time state adjustment).
  1313→  const [prevOpen, setPrevOpen] = useState(open);
  1314→  if (open !== prevOpen) {
  1315→    setPrevOpen(open);
  1316→    if (open) {
  1317→      setForm({
  1318→        propertyId: contract?.propertyId ?? defaultPropertyId ?? "",
  1319→        name: contract?.name ?? "",
  1320→        startDate: contract?.startDate?.slice(0, 10) ?? todayStr(),
  1321→        endDate: contract?.endDate?.slice(0, 10) ?? "",
  1322→        billingRate: contract ? String(contract.billingRate) : "",
  1323→        payoutRate: contract ? String(contract.payoutRate) : "",
  1324→        shift: contract?.shift ?? "DAY",
  1325→        category: contract?.category ?? "",
  1326→        maxEmployees: contract?.maxEmployees ? String(contract.maxEmployees) : "",
  1327→        paymentTerms: contract?.paymentTerms ?? "",
  1328→        notes: contract?.notes ?? "",
  1329→      });
  1330→    }
  1331→  }
  1332→
  1333→  const set = (k: keyof typeof form) => (v: string) => setForm((f) => ({ ...f, [k]: v }));
  1334→
  1335→  const submit = async () => {
  1336→    if (!form.propertyId) { toast.error("Select a property"); return; }
  1337→    if (!form.name.trim()) { toast.error("Contract name is required"); return; }
  1338→    if (!form.startDate) { toast.error("Start date is required"); return; }
  1339→    const b = parseAmount(form.billingRate);
  1340→    const p = parseAmount(form.payoutRate);
  1341→    if (b <= 0 || p <= 0) { toast.error("Billing and payout rates are required"); return; }
  1342→    const body = {
  1343→      propertyId: form.propertyId,
  1344→      name: form.name.trim(),
  1345→      startDate: form.startDate,
  1346→      endDate: form.endDate || undefined,
  1347→      billingRate: b,
  1348→      payoutRate: p,
  1349→      shift: form.shift || undefined,
  1350→      category: form.category || undefined,
  1351→      maxEmployees: form.maxEmployees ? Number(form.maxEmployees) : undefined,
  1352→      paymentTerms: form.paymentTerms || undefined,
  1353→      notes: form.notes || undefined,
  1354→    };
  1355→    const res = editing
  1356→      ? await mutate(() => api.put(`/api/contracts/${contract!.id}`, body), "Contract updated — affects future deployments only")
  1357→      : await mutate(() => api.post("/api/contracts", body), "Contract created");
  1358→    if (res.ok) { onOpenChange(false); onDone(); }
  1359→  };
  1360→
  1361→  return (
  1362→    <Dialog open={open} onOpenChange={onOpenChange}>
  1363→      <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-lg">
  1364→        <DialogHeader>
  1365→          <DialogTitle>{editing ? "Edit Contract" : "New Contract"}</DialogTitle>
  1366→          <DialogDescription>
  1367→            {editing
  1368→              ? "Rate changes affect future deployments only — historical records keep their snapshot."
  1369→              : "Creating an ACTIVE contract ends the property's previous active contract."}
  1370→          </DialogDescription>
  1371→        </DialogHeader>
  1372→        <div className="grid gap-3 sm:grid-cols-2">
  1373→          <Field label="Property" required className="sm:col-span-2">
  1374→            <SelectInput
  1375→              value={form.propertyId}
  1376→              onChange={set("propertyId")}
  1377→              placeholder="Select property…"
  1378→              disabled={Boolean(defaultPropertyId) && !editing}
  1379→              options={properties.map((p) => ({ label: p.name, value: p.id }))}
  1380→            />
  1381→          </Field>
  1382→          <Field label="Contract name" required className="sm:col-span-2">
  1383→            <Input value={form.name} onChange={(e) => set("name")(e.target.value)} className="h-10" placeholder="e.g. Day-shift 2025" />
  1384→          </Field>
  1385→          <Field label="Start date" required><Input type="date" value={form.startDate} onChange={(e) => set("startDate")(e.target.value)} className="h-10" /></Field>
  1386→          <Field label="End date"><Input type="date" value={form.endDate} onChange={(e) => set("endDate")(e.target.value)} className="h-10" /></Field>
  1387→          <Field label="Billing rate (₹/shift)" required><Input type="number" inputMode="numeric" value={form.billingRate} onChange={(e) => set("billingRate")(e.target.value)} className="h-10" /></Field>
  1388→          <Field label="Payout rate (₹/shift)" required><Input type="number" inputMode="numeric" value={form.payoutRate} onChange={(e) => set("payoutRate")(e.target.value)} className="h-10" /></Field>
  1389→          <Field label="Shift">
  1390→            <SelectInput value={form.shift} onChange={set("shift")} options={[{ label: "Day", value: "DAY" }, { label: "Night", value: "NIGHT" }, { label: "All", value: "ALL" }]} />
  1391→          </Field>
  1392→          <Field label="Work category"><Input value={form.category} onChange={(e) => set("category")(e.target.value)} className="h-10" placeholder="Service, Kitchen…" /></Field>
  1393→          <Field label="Max employees"><Input type="number" inputMode="numeric" value={form.maxEmployees} onChange={(e) => set("maxEmployees")(e.target.value)} className="h-10" /></Field>
  1394→          <Field label="Payment terms"><Input value={form.paymentTerms} onChange={(e) => set("paymentTerms")(e.target.value)} className="h-10" placeholder="Weekly, every Monday" /></Field>
  1395→          <Field label="Notes" className="sm:col-span-2"><Textarea value={form.notes} onChange={(e) => set("notes")(e.target.value)} rows={2} /></Field>
  1396→        </div>
  1397→        <DialogFooter className="gap-2">
  1398→          <Button variant="outline" className="min-h-10 flex-1 sm:flex-none" onClick={() => onOpenChange(false)}>Cancel</Button>
  1399→          <Button className="min-h-10 flex-1 sm:flex-none" onClick={submit} disabled={saving}>{saving ? "Saving…" : editing ? "Save changes" : "Create contract"}</Button>
  1400→        </DialogFooter>
  1401→      </DialogContent>
  1402→    </Dialog>
  1403→  );
  1404→}
