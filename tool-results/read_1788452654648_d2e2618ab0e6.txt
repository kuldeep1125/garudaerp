    88→function ymd(d: Date): string {
    89→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;
    90→}
    91→
    92→function computeRange(range: RangeKey, customFrom: string, customTo: string): { from: string; to: string } {
    93→  const now = new Date();
    94→  const t = todayStr();
    95→  switch (range) {
    96→    case "today":
    97→      return { from: t, to: t };
    98→    case "yesterday":
    99→      return { from: todayStr(-1), to: todayStr(-1) };
   100→    case "week": {
   101→      const dow = now.getDay();
   102→      const monday = new Date(now);
   103→      monday.setDate(now.getDate() - (dow === 0 ? 6 : dow - 1));
   104→      return { from: ymd(monday), to: t };
   105→    }
   106→    case "lastweek": {
   107→      const dow = now.getDay();
   108→      const monday = new Date(now);
   109→      monday.setDate(now.getDate() - (dow === 0 ? 6 : dow - 1) - 7);
   110→      const sunday = new Date(monday);
   111→      sunday.setDate(monday.getDate() + 6);
   112→      return { from: ymd(monday), to: ymd(sunday) };
   113→    }
   114→    case "month":
   115→      return { from: `${t.slice(0, 7)}-01`, to: t };
   116→    case "lastmonth": {
   117→      const first = new Date(now.getFullYear(), now.getMonth() - 1, 1);
   118→      const last = new Date(now.getFullYear(), now.getMonth(), 0);
   119→      return { from: ymd(first), to: ymd(last) };
   120→    }
   121→    case "custom": {
   122→      let from = customFrom || t;
   123→      let to = customTo || customFrom || t;
   124→      if (from > to) [from, to] = [to, from];
   125→      return { from, to };
   126→    }
   127→  }
   128→}

   130→function fmtCell(v: unknown, type?: string): React.ReactNode {
   131→  if (v === null || v === undefined || v === "") return <span className="text-muted-foreground">—</span>;
   132→  if (type === "money" || type === "currency") return formatINR(Number(v));
   133→  if (type === "date") return fmtDayText(String(v));
   134→  return String(v);
   135→}
   136→
   137→function fmtDayText(v: string): string {
   138→  const d = new Date(v.length === 10 ? `${v}T00:00:00` : v);
   139→  return Number.isNaN(d.getTime()) ? v : d.toLocaleDateString("en-IN", { day: "numeric", month: "short" });
   140→}
   141→
   142→function fmtCellText(v: unknown, type?: string): string {
   143→  if (v === null || v === undefined || v === "") return "";
   144→  if (type === "money" || type === "currency") return formatINR(Number(v));
   145→  if (type === "date") return fmtDayText(String(v));
   146→  return String(v);
   147→}
   148→
   149→function labelFor(key: string, cols: ReportColumn[]): string {
   150→  const found = cols.find((c) => c.key === key);
   151→  if (found) return found.label;
   152→  return key.replace(/([A-Z])/g, " $1").replace(/^./, (s) => s.toUpperCase());
   153→}
   154→
   155→function fmtTotal(v: unknown, cols: ReportColumn[], key: string): string {
   156→  const t = cols.find((c) => c.key === key)?.type;
   157→  if (t === "money" || t === "currency") return formatINR(Number(v));
   158→  if (typeof v === "number") return v.toLocaleString("en-IN");
   159→  return String(v ?? "—");
   160→}
   161→
   162→/** Small shift badge — sun for DAY, moon for NIGHT, dual for FULL. */
   163→function ShiftPill({ shift }: { shift: string }) {
   164→  const s = shift.toUpperCase();
   165→  const isNight = s === "NIGHT";
   166→  return (
   167→    <span
   168→      className={cn(
   169→        "inline-flex items-center gap-1 rounded-full px-2 py-0.5 text-[10px] font-semibold",
   170→        s === "FULL" ? "bg-primary text-primary-foreground" : isNight ? "bg-secondary text-secondary-foreground" : "bg-accent text-accent-foreground"
   171→      )}
   172→    >
   173→      {s === "DAY" ? "DAY" : s === "NIGHT" ? "NIGHT" : "FULL (D+N)"}
   174→    </span>
   175→  );
   176→}
   177→
   178→function metaText(meta: unknown): string | null {
   179→  if (meta === null || meta === undefined) return null;
   180→  if (typeof meta === "string") return meta;
   181→  if (typeof meta === "object") {
   182→    try {
   183→      const parts = Object.entries(meta as Record<string, unknown>).map(([k, v]) => `${k}: ${String(v)}`);
   184→      return parts.length > 0 ? parts.join(" · ") : null;
   185→    } catch {
   186→      return null;
   187→    }
   188→  }
   189→  return String(meta);
   190→}
