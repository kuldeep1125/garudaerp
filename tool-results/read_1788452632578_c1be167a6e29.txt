    54→
    55→function BackupCard() {
    56→  const { lang } = useLang();
    57→  const [exporting, setExporting] = useState(false);
    58→
    59→  const exportBackup = async () => {
    60→    setExporting(true);
    61→    try {
    62→      const backup = await api.get<BackupResp>("/api/settings/backup");
    63→      const total = Object.values(backup.counts).reduce((s, n) => s + n, 0);
    64→      downloadJSON(`bizhub-backup-${todayStr()}.json`, backup);
    65→      toast.success(
    66→        t(lang, "settings.backupDone")
    67→          .replace("{total}", total.toLocaleString("en-IN"))
    68→          .replace("{collections}", String(Object.keys(backup.counts).length))
    69→      );
    70→    } catch (e) {
    71→      toast.error(errMessage(e));
    72→    } finally {
    73→      setExporting(false);
    74→    }
    75→  };
    76→
    77→  return (
    78→    <Card>
    79→      <CardHeader>
    80→        <CardTitle className="flex items-center gap-2 text-base">
    81→          <Database className="h-4 w-4 text-primary" aria-hidden />
    82→          {t(lang, "settings.backupTitle")}
    83→        </CardTitle>
    84→        <CardDescription>{t(lang, "settings.backupDesc")}</CardDescription>
    85→      </CardHeader>
    86→      <CardContent className="space-y-3">
    87→        <Button variant="outline" className="min-h-10 gap-2" onClick={() => void exportBackup()} disabled={exporting}>
    88→          {exporting ? (
    89→            <Loader2 className="h-4 w-4 animate-spin" aria-hidden />
    90→          ) : (
    91→            <Download className="h-4 w-4" aria-hidden />
    92→          )}
    93→          {exporting ? t(lang, "settings.backupWorking") : t(lang, "settings.backupAction")}
    94→        </Button>
    95→        <p className="text-xs leading-relaxed text-muted-foreground">{t(lang, "settings.backupNote")}</p>
    96→      </CardContent>
    97→    </Card>
    98→  );
    99→}

   101→// ---------------------------------------------------------------------------
   102→// Data integrity — zero-mismatch health checks (GET/POST /api/settings/integrity)
   103→// ---------------------------------------------------------------------------
   104→
   105→interface IntegrityResp {
   106→  ok: boolean;
   107→  checks: { id: string; label: string; detail: string; count: number; status: "ok" | "warn" }[];
   108→  totals: { trips: number; settlements: number };
   109→}
   110→
   111→function IntegrityCard() {
   112→  const { data, loading, error, reload } = useAsync<IntegrityResp>(() => api.get("/api/settings/integrity"), []);
   113→  const [repairing, setRepairing] = useState(false);
   114→
   115→  const repair = async () => {
   116→    setRepairing(true);
   117→    try {
   118→      const r = await api.post<{ repaired: { properties: number; cancelledCleaned: number } }>("/api/settings/integrity");
   119→      toast.success(`Repaired — allocations re-checked across ${r.repaired.properties} property(ies)`);
   120→      void reload();
   121→    } catch (e) {
   122→      toast.error(errMessage(e));
   123→    } finally {
   124→      setRepairing(false);
   125→    }
   126→  };
   127→
   128→  const issues = data?.checks.reduce((s, c) => s + c.count, 0) ?? 0;
   129→
   130→  return (
   131→    <Card>
   132→      <CardHeader>
   133→        <CardTitle className="flex items-center gap-2 text-base">
   134→          <ShieldCheck className="h-4 w-4 text-primary" aria-hidden />
   135→          Data integrity
   136→        </CardTitle>
   137→        <CardDescription>
   138→          Continuous zero-mismatch audit — derived numbers are re-checked against their source records.
   139→        </CardDescription>
   140→      </CardHeader>
   141→      <CardContent className="space-y-3">
   142→        {loading ? (
   143→          <div className="space-y-2" aria-busy="true">
   144→            <Skeleton className="h-9 w-full" />
   145→            <Skeleton className="h-9 w-full" />
   146→            <Skeleton className="h-9 w-full" />
   147→          </div>
   148→        ) : error ? (
   149→          <div className="flex items-center justify-between gap-2">
   150→            <p className="text-xs text-red-600 dark:text-red-400">{error}</p>
   151→            <Button variant="outline" size="sm" className="h-8" onClick={() => void reload()}>Retry</Button>
   152→          </div>
   153→        ) : data ? (
   154→          <>
   155→            <div
   156→              className={cn(
   157→                "flex items-center gap-2 rounded-lg border px-3 py-2 text-xs font-medium",
   158→                data.ok
   159→                  ? "border-emerald-200 bg-emerald-50/70 text-emerald-800 dark:border-emerald-900 dark:bg-emerald-950/30 dark:text-emerald-300"
   160→                  : "border-amber-300 bg-amber-50/70 text-amber-800 dark:border-amber-900 dark:bg-amber-950/30 dark:text-amber-300"
   161→              )}
   162→              role="status"
   163→            >
   164→              {data.ok ? <Check className="h-4 w-4 shrink-0" aria-hidden /> : <AlertTriangle className="h-4 w-4 shrink-0" aria-hidden />}
   165→              {data.ok
   166→                ? "All checks passed — books are consistent"
   167→                : `${issues} issue${issues === 1 ? "" : "s"} found — auto-repair can fix allocation drift`}
   168→            </div>
   169→            <ul className="space-y-1.5">
   170→              {data.checks.map((c) => (
   171→                <li key={c.id} className="flex items-center justify-between gap-2 rounded-lg border px-3 py-2">
   172→                  <div className="min-w-0">
   173→                    <p className="text-xs font-semibold">{c.label}</p>
   174→                    <p className="truncate text-[10px] text-muted-foreground">{c.detail}</p>
   175→                  </div>
   176→                  <span
   177→                    className={cn(
   178→                      "shrink-0 rounded-full px-2 py-0.5 text-[10px] font-bold",
   179→                      c.count === 0
   180→                        ? "bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-300"
   181→                        : "bg-red-100 text-red-700 dark:bg-red-950 dark:text-red-300"
   182→                    )}
   183→                  >
   184→                    {c.count === 0 ? "OK" : c.count}
   185→                  </span>
   186→                </li>
   187→              ))}
   188→            </ul>
   189→            <div className="flex gap-2">
   190→              <Button variant="outline" size="sm" className="h-9 gap-1.5" onClick={() => void reload()} disabled={loading}>
   191→                <RefreshCw className="h-3.5 w-3.5" aria-hidden />
   192→                Re-run checks
   193→              </Button>
   194→              {issues > 0 && (
   195→                <Button size="sm" className="h-9 gap-1.5" onClick={() => void repair()} disabled={repairing}>
   196→                  {repairing ? <Loader2 className="h-3.5 w-3.5 animate-spin" aria-hidden /> : <ShieldCheck className="h-3.5 w-3.5" aria-hidden />}
   197→                  Auto-repair
   198→                </Button>
   199→              )}
   200→            </div>
   201→          </>
   202→        ) : null}
   203→      </CardContent>
   204→    </Card>
   205→  );
   206→}

   243→        <AlertDialog>

   260→              value={confirmText}
   261→              onChange={(e) => setConfirmText(e.target.value)}

   276→                {resetting ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <AlertTriangle className="h-4 w-4" aria-hidden />}

   278→              </AlertDialogAction>
   279→            </AlertDialogFooter>
   280→          </AlertDialogContent>
   281→        </AlertDialog>
   282→      </CardContent>
   283→    </Card>

   352→export default function SettingsView({ navigate }: ViewProps) {

   358→
   359→  const [form, setForm] = useState<SettingsForm | null>(null);
   360→  const { mutate, saving } = useMutation();
   361→
   362→  // Derived defaults from server data; user edits take over via `form`.
   363→  const biz = data?.business;
   364→  const derived: SettingsForm | null = biz
   365→    ? {
   366→        name: biz.name ?? "",
   367→        address: biz.address ?? "",

   370→        logoText: biz.logoText ?? "",

   375→  const dirty = Boolean(form && derived && JSON.stringify(form) !== JSON.stringify(derived));

   388→        address: current.address || undefined,
   389→        contact: current.contact || undefined,
   390→        gstin: current.gstin || undefined,
   391→        logoText: current.logoText || undefined,
   392→      }),
   393→      "Settings saved"
