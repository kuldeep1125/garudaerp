    66→function rangeDates(r: RangeKey): { from: string; to: string } {
    67→  const now = new Date();
    68→  const f = (d: Date) => `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;
    69→  if (r === "today") { const t = f(now); return { from: t, to: t }; }
    70→  if (r === "yesterday") return { from: f(new Date(now.getFullYear(), now.getMonth(), now.getDate() - 1)), to: f(new Date(now.getFullYear(), now.getMonth(), now.getDate() - 1)) };
    71→  if (r === "week") return { from: f(new Date(now.getFullYear(), now.getMonth(), now.getDate() - 6)), to: f(now) };
    72→  if (r === "lastweek") return { from: f(new Date(now.getFullYear(), now.getMonth(), now.getDate() - 13)), to: f(new Date(now.getFullYear(), now.getMonth(), now.getDate() - 7)) };
    73→  if (r === "lastmonth") return { from: f(new Date(now.getFullYear(), now.getMonth() - 1, 1)), to: f(new Date(now.getFullYear(), now.getMonth(), 0)) };
    74→  return { from: f(new Date(now.getFullYear(), now.getMonth(), 1)), to: f(new Date(now.getFullYear(), now.getMonth() + 1, 0)) };
    75→}
    76→
    77→function nowLocalValue(): string {
    78→  return new Date(Date.now() - new Date().getTimezoneOffset() * 60000).toISOString().slice(0, 16);
    79→}
    80→
    81→function tripTotal(t: TripRec): number {
    82→  return t.finalAmount ?? (t.agreedAmount + (t.extraCharges ?? 0));
    83→}
    84→
    85→function tripOutstanding(t: TripRec): number {
    86→  return Math.max(0, tripTotal(t) - (t.paidAmount ?? 0));
    87→}
    88→
    89→function tripPaidPct(t: TripRec): number {
    90→  const target = tripTotal(t);
    91→  if (target <= 0) return 0;
    92→  return Math.min(100, Math.round(((t.paidAmount ?? 0) / target) * 100));
    93→}

    99→function NewTripDialog({ open, onOpenChange, vehicles, clients, onDone }: {
   100→  open: boolean;
   101→  onOpenChange: (v: boolean) => void;
   102→  vehicles: VehicleRec[];
   103→  clients: ClientRec[];
   104→  onDone: () => void;
   105→}) {
   106→  const [form, setForm] = useState({
   107→    vehicleId: "", clientId: "", startAt: nowLocalValue(), endAt: "", tripType: "RENTAL",
   108→    rentalType: "DAILY", pickup: "", destination: "", driver: "", fuelResponsibility: "CLIENT",
   109→    agreedAmount: "", advanceReceived: "", notes: "",
   110→  });
   111→  const { mutate, saving } = useMutation();
   112→
   113→  const [prevOpen, setPrevOpen] = useState(open);
   114→  if (open !== prevOpen) {
   115→    setPrevOpen(open);
   116→    if (open) {
   117→      setForm({
   118→        vehicleId: "", clientId: "", startAt: nowLocalValue(), endAt: "", tripType: "RENTAL",
   119→        rentalType: "DAILY", pickup: "", destination: "", driver: "", fuelResponsibility: "CLIENT",
   120→        agreedAmount: "", advanceReceived: "", notes: "",
   121→      });
   122→    }
   123→  }
   124→
   125→  const set = (k: keyof typeof form) => (v: string) => setForm((f) => ({ ...f, [k]: v }));
   126→
   127→  const submit = async () => {
   128→    if (!form.vehicleId) { toast.error("Select a vehicle"); return; }
   129→    if (!form.clientId) { toast.error("Select a client"); return; }
   130→    if (!form.startAt) { toast.error("Start date & time is required"); return; }
   131→    const amt = parseAmount(form.agreedAmount);
   132→    if (amt <= 0) { toast.error("Enter the agreed amount"); return; }
   133→    const res = await mutate(
   134→      () => api.post("/api/trips", {
   135→        vehicleId: form.vehicleId,
   136→        clientId: form.clientId,
   137→        startAt: new Date(form.startAt).toISOString(),
   138→        endAt: form.endAt ? new Date(form.endAt).toISOString() : undefined,
   139→        tripType: form.tripType,
   140→        rentalType: form.rentalType || undefined,
   141→        pickup: form.pickup || undefined,
   142→        destination: form.destination || undefined,
   143→        driver: form.driver || undefined,
   144→        fuelResponsibility: form.fuelResponsibility || undefined,
   145→        agreedAmount: amt,
   146→        advanceReceived: form.advanceReceived ? parseAmount(form.advanceReceived) : undefined,
   147→        notes: form.notes || undefined,
   148→      }),
   149→      "Rental created — vehicle marked as engaged",
   150→      (data) => ({ module: "TRIP", recordId: (data as { id: string }).id, onUndo: onDone })
   151→    );
   152→    if (res.ok) { onOpenChange(false); onDone(); }
   153→  };
   154→
   155→  return (
   156→    <Dialog open={open} onOpenChange={onOpenChange}>
   157→      <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-lg">
   158→        <DialogHeader>
   159→          <DialogTitle>New Rental / Trip</DialogTitle>
   160→          <DialogDescription>Overlapping bookings for the same vehicle are blocked automatically.</DialogDescription>
   161→        </DialogHeader>
   162→        <div className="grid gap-3 sm:grid-cols-2">
   163→          <Field label="Vehicle" required className="sm:col-span-2">
   164→            <SelectInput
   165→              value={form.vehicleId}
   166→              onChange={set("vehicleId")}
   167→              placeholder="Select vehicle…"
   168→              options={vehicles.map((v) => ({ label: `${v.name} · ${v.status}`, value: v.id }))}
   169→            />
   170→          </Field>
   171→          <Field label="Client" required className="sm:col-span-2">
   172→            <SelectInput
   173→              value={form.clientId}
   174→              onChange={set("clientId")}
   175→              placeholder="Select client…"
   176→              options={clients.map((c) => ({ label: c.company ? `${c.name} (${c.company})` : c.name, value: c.id }))}
   177→            />
   178→          </Field>
   179→          <Field label="Start at" required>
   180→            <Input type="datetime-local" value={form.startAt} onChange={(e) => set("startAt")(e.target.value)} className="h-10" />
   181→          </Field>
   182→          <Field label="End at" hint="Leave empty for an open-ended rental">
   183→            <Input type="datetime-local" value={form.endAt} onChange={(e) => set("endAt")(e.target.value)} className="h-10" />
   184→          </Field>
   185→          <Field label="Type" required>
   186→            <SelectInput value={form.tripType} onChange={set("tripType")} options={TRIP_TYPE_OPTIONS} />
   187→          </Field>
   188→          <Field label="Rental type">
   189→            <SelectInput value={form.rentalType} onChange={set("rentalType")} options={RENTAL_TYPE_OPTIONS} />
   190→          </Field>
   191→          <Field label="Pickup">
   192→            <Input value={form.pickup} onChange={(e) => set("pickup")(e.target.value)} className="h-10" placeholder="Pickup point" />
   193→          </Field>
   194→          <Field label="Destination">
   195→            <Input value={form.destination} onChange={(e) => set("destination")(e.target.value)} className="h-10" placeholder="Drop point" />
   196→          </Field>
   197→          <Field label="Driver">
   198→            <Input value={form.driver} onChange={(e) => set("driver")(e.target.value)} className="h-10" placeholder="Driver name" />
   199→          </Field>
   200→          <Field label="Fuel responsibility">
   201→            <SelectInput value={form.fuelResponsibility} onChange={set("fuelResponsibility")} options={FUEL_OPTIONS} />
   202→          </Field>
   203→          <Field label="Agreed amount (₹)" required>
   204→            <MoneyInput value={form.agreedAmount} onChange={set("agreedAmount")} min={1} className="h-10" />
   205→          </Field>
   206→          <Field label="Advance received (₹)">
   207→            <MoneyInput value={form.advanceReceived} onChange={set("advanceReceived")} min={0} className="h-10" />
   208→          </Field>
   209→          <Field label="Notes" className="sm:col-span-2">
   210→            <Textarea value={form.notes} onChange={(e) => set("notes")(e.target.value)} rows={2} placeholder="Optional instructions" />
   211→          </Field>
   212→        </div>
   213→        <DialogFooter className="gap-2">
   214→          <Button variant="outline" className="min-h-10 flex-1 sm:flex-none" onClick={() => onOpenChange(false)}>Cancel</Button>
   215→          <Button className="min-h-10 flex-1 sm:flex-none" onClick={submit} disabled={saving}>
   216→            {saving ? "Creating…" : "Create rental"}
   217→          </Button>
   218→        </DialogFooter>
   219→      </DialogContent>
   220→    </Dialog>
   221→  );
   222→}

   448→              loading={trips.loading}
   449→              emptyIcon={Route}
   450→              emptyTitle="No trips match"
   451→              emptyDescription="Adjust filters or create a new rental."
   452→            />
   453→          )}
   454→        </CardContent>
   455→      </Card>
   456→
   457→      {/* Detail dialog */}
   458→      <Dialog open={Boolean(detail)} onOpenChange={(v) => !v && setDetailId(null)}>
   459→        <DialogContent className="max-h-[88vh] overflow-y-auto sm:max-w-lg">
   460→          {detailTrip && (
   461→            <>
   462→              <DialogHeader>
   463→                <DialogTitle className="flex flex-wrap items-center gap-2">
   464→                  {detailTrip.vehicleName ?? "Vehicle"} <span className="text-muted-foreground">→</span> {detailTrip.clientName ?? "Client"}
   465→                </DialogTitle>
   466→                <DialogDescription className="flex flex-wrap items-center gap-1.5">
   467→                  <StatusBadge status={detailTrip.status} />
   468→                  <StatusBadge status={detailTrip.paymentStatus} />
   469→                  <StatusBadge status={detailTrip.tripType} />
   470→                  {detailTrip.rentalType && <StatusBadge status={detailTrip.rentalType} />}
   471→                </DialogDescription>
   472→              </DialogHeader>
   473→
   474→              <div className="rounded-xl border bg-muted/30 px-3.5 py-2">
   475→                <KV label="Vehicle" value={detailTrip.vehicleReg ? `${detailTrip.vehicleName ?? ""} · ${detailTrip.vehicleReg}` : detailTrip.vehicleName ?? "—"} />
   476→                <KV label="Period" value={`${fmtDay(detailTrip.startAt)} → ${detailTrip.endAt ? fmtDay(detailTrip.endAt) : "ongoing"}`} />
   477→                <KV label="Pickup" value={detailTrip.pickup || "—"} />
   478→                <KV label="Destination" value={detailTrip.destination || "—"} />
   479→                <KV label="Driver" value={detailTrip.driver || "—"} />
   480→                <KV label="Fuel responsibility" value={detailTrip.fuelResponsibility === "OWNER" ? "Owner" : detailTrip.fuelResponsibility === "CLIENT" ? "Client" : "—"} />
   481→                <KV label="Agreed amount" value={formatINR(detailTrip.agreedAmount)} />
   482→                <KV label="Advance received" value={formatINR(detailTrip.advanceReceived ?? 0)} className="text-emerald-600 dark:text-emerald-400" />
   483→                <KV label="Extra charges" value={formatINR(detailTrip.extraCharges ?? 0)} />
   484→                <KV label="Final amount" value={formatINR(tripTotal(detailTrip))} />
   485→                <KV label="Paid" value={formatINR(detailTrip.paidAmount ?? 0)} className="text-emerald-600 dark:text-emerald-400" />
   486→                <KV
   487→                  label="Outstanding"
   488→                  value={formatINR(tripOutstanding(detailTrip))}
   489→                  className={tripOutstanding(detailTrip) > 0 ? "text-red-600 dark:text-red-400" : "text-emerald-600 dark:text-emerald-400"}
   490→                />
   491→                {detailTrip.notes && <KV label="Notes" value={<span className="text-xs font-normal">{detailTrip.notes}</span>} />}
   492→                <KV label="Created by" value={detailTrip.createdByName ?? "—"} />
   493→              </div>
   494→
   495→              <Separator />
   496→
   497→              <DialogFooter className="flex-col gap-2 sm:flex-row sm:flex-wrap">
   498→                {(detailTrip.status === "ACTIVE" || detailTrip.status === "CONFIRMED") && (
   499→                  <>
   500→                    <Button variant="outline" className="min-h-10 gap-1.5" onClick={() => { setDetailId(null); setEndFor(detailTrip.id); }}>
   501→                      <BadgeCheck className="h-4 w-4" aria-hidden />End Trip
   502→                    </Button>
   503→                    <Button
   504→                      variant="outline" className="min-h-10 text-red-600 hover:bg-red-50 dark:text-red-400 dark:hover:bg-red-950"
   505→                      onClick={() => { setDetailId(null); setCancelFor(detailTrip.id); }}
   506→                    >
   507→                      Cancel Trip
   508→                    </Button>
   509→                  </>
   510→                )}
   511→                {(detailTrip.status === "ACTIVE" || detailTrip.status === "CONFIRMED" || detailTrip.status === "COMPLETED") && (
   512→                  <Button className="min-h-10" onClick={() => { setDetailId(null); setPayFor(detailTrip.id); }}>
   513→                    Record Payment
   514→                  </Button>
   515→                )}
   516→              </DialogFooter>
   517→            </>
   518→          )}
   519→        </DialogContent>
   520→      </Dialog>
   521→
   522→      {/* End trip dialog */}
   523→      <Dialog open={Boolean(endFor)} onOpenChange={(v) => !v && setEndFor(null)}>
   524→        <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-sm">
   525→          <DialogHeader>
   526→            <DialogTitle>End trip</DialogTitle>
   527→            <DialogDescription>Vehicle becomes available after completion (unless another trip is active).</DialogDescription>
   528→          </DialogHeader>
   529→          <div className="space-y-3">
   530→            <Field label="Ended at" required>
   531→              <Input type="datetime-local" value={endForm.endAt} onChange={(e) => setEndForm((f) => ({ ...f, endAt: e.target.value }))} className="h-10" />
   532→            </Field>
   533→            <div className="grid gap-3 sm:grid-cols-2">
   534→              <Field label="Final amount (₹)" required>
   535→                <MoneyInput value={endForm.finalAmount} onChange={(v) => setEndForm((f) => ({ ...f, finalAmount: v }))} className="h-10" />
   536→              </Field>
   537→              <Field label="Extra charges (₹)">
   538→                <MoneyInput value={endForm.extraCharges} onChange={(v) => setEndForm((f) => ({ ...f, extraCharges: v }))} min={0} className="h-10" />
   539→              </Field>
   540→            </div>
   541→          </div>
   542→          <DialogFooter className="gap-2">
   543→            <Button variant="outline" className="min-h-10 flex-1 sm:flex-none" onClick={() => setEndFor(null)}>Back</Button>
   544→            <Button className="min-h-10 flex-1 sm:flex-none" onClick={() => void submitEnd()} disabled={saving}>{saving ? "Saving…" : "Complete trip"}</Button>
   545→          </DialogFooter>
   546→        </DialogContent>
   547→      </Dialog>
   548→
   549→      {/* Record payment dialog */}
   550→      <Dialog open={Boolean(payFor)} onOpenChange={(v) => !v && setPayFor(null)}>
   551→        <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-sm">
   552→          <DialogHeader>
   553→            <DialogTitle>Record trip payment</DialogTitle>
   554→            <DialogDescription>Amount is added to the trip&apos;s paid total; payment status updates automatically.</DialogDescription>
   555→          </DialogHeader>
   556→          <div className="grid gap-3 sm:grid-cols-2">
   557→            <Field label="Amount (₹)" required>
   558→              <MoneyInput value={payForm.amount} onChange={(v) => setPayForm((f) => ({ ...f, amount: v }))} min={1} className="h-10" />
   559→            </Field>
   560→            <Field label="Method">
   561→              <SelectInput value={payForm.method} onChange={(v) => setPayForm((f) => ({ ...f, method: v }))} options={METHOD_OPTIONS} />
   562→            </Field>
   563→          </div>
   564→          <ConfirmAmount amount={parseAmount(payForm.amount)} subject={paySubject} onSubmit={() => void submitPayment()}>
   565→            {({ confirm, armed }) => (
   566→              <DialogFooter className="gap-2">
   567→                <Button variant="outline" className="min-h-10 flex-1 sm:flex-none" onClick={() => setPayFor(null)}>Cancel</Button>
   568→                <Button className="min-h-10 flex-1 sm:flex-none" onClick={confirm} disabled={saving || armed}>
   569→                  {saving ? "Saving…" : armed ? "Confirm below" : "Record payment"}
   570→                </Button>
   571→              </DialogFooter>
   572→            )}
   573→          </ConfirmAmount>
   574→        </DialogContent>
   575→      </Dialog>
   576→
   577→      {/* Cancel trip confirm */}
   578→      <AlertDialog open={Boolean(cancelFor)} onOpenChange={(v) => !v && setCancelFor(null)}>
   579→        <AlertDialogContent>
   580→          <AlertDialogHeader>
   581→            <AlertDialogTitle>Cancel this trip?</AlertDialogTitle>
   582→            <AlertDialogDescription>The vehicle will be released and the trip marked CANCELLED. This cannot be undone.</AlertDialogDescription>
   583→          </AlertDialogHeader>
   584→          <AlertDialogFooter>
   585→            <AlertDialogCancel className="min-h-10">Keep trip</AlertDialogCancel>
   586→            <AlertDialogAction
   587→              className={cn("min-h-10 bg-red-600 text-white hover:bg-red-700")}
   588→              onClick={(e) => { e.preventDefault(); void cancelTrip(); }}
   589→            >
   590→              {saving ? "Cancelling…" : "Cancel trip"}
   591→            </AlertDialogAction>
   592→          </AlertDialogFooter>
   593→        </AlertDialogContent>
   594→      </AlertDialog>
   595→
   596→      <NewTripDialog
   597→        open={newOpen}
   598→        onOpenChange={setNewOpen}
   599→        vehicles={vehicles.data?.items ?? []}
   600→        clients={clients.data?.items ?? []}
   601→        onDone={() => void trips.reload()}
   602→      />
   603→    </div>
   604→  );
   605→}
