    44→    44→function daysUntil(dateStr?: string | null): number | null {
    45→    45→  if (!dateStr) return null;
    46→    46→  const d = new Date(dateStr.length === 10 ? `${dateStr}T00:00:00` : dateStr);
    47→    47→  if (Number.isNaN(d.getTime())) return null;
    48→    48→  const today = new Date();
    49→    49→  today.setHours(0, 0, 0, 0);
    50→    50→  return Math.round((d.getTime() - today.getTime()) / 86400000);
    51→    51→}

    53→    53→function tripTotal(t: TripRec): number {
    54→    54→  return t.finalAmount ?? (t.agreedAmount + (t.extraCharges ?? 0));

    57→    57→function expiryCls(dateStr?: string | null): string {
    58→    58→  const d = daysUntil(dateStr);
    59→    59→  if (d === null) return "";
    60→    60→  if (d < 0) return "text-red-600 dark:text-red-400 font-semibold";
    61→    61→  if (d <= 30) return "text-amber-600 dark:text-amber-400 font-semibold";
    62→    62→  return "";
    63→    63→}
    64→    64→
    65→    65→function expiryHint(dateStr?: string | null): string {
    66→    66→  const d = daysUntil(dateStr);
    67→    67→  if (d === null) return "";
    68→    68→  if (d < 0) return ` (expired ${Math.abs(d)}d ago)`;
    69→    69→  if (d <= 30) return ` (in ${d}d)`;
    70→    70→  return "";
    71→    71→}
