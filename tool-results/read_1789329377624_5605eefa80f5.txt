   272→/** Fetch-on-mount hook with loading/error/reload. Toasts load failures. */
   273→export function useAsync<T>(loader: () => Promise<T>, deps: React.DependencyList) {
   274→  const [data, setData] = useState<T | null>(null);
   275→  const [loading, setLoading] = useState(true);
   276→  const [error, setError] = useState<string | null>(null);
   277→  const load = useCallback(async () => {
   278→    setLoading(true);
   279→    try {
   280→      const d = await loader();
   281→      setData(d);
   282→      setError(null);
   283→    } catch (e) {
   284→      const msg = errMessage(e);
   285→      setError(msg);
   286→      toast.error(msg);
   287→    } finally {
   288→      setLoading(false);
   289→    }
   290→  }, deps);
   291→  useEffect(() => { void load(); }, [load]);
   292→  return { data, loading, error, reload: load, setData };
   293→}

   295→/** Mutation wrapper: consistent error toasts, optional success toast.
   296→ *  Pass an `undo` resolver to attach an "Undo" action to the success toast —
   297→ *  it reverses the entry server-side (zero mismatch) and calls onUndo. */
   298→export interface UndoSpec {
   299→  module: string; // PAYMENT / ADVANCE / EXPENSE / DEPLOYMENT / TRIP / EMI / MAINTENANCE / ADJUSTMENT
   300→  recordId?: string;
   301→  onUndo?: () => void;
   302→}
   303→
   304→export async function undoRequest(spec: {
   305→  auditLogId?: string;
   306→  module?: string;
   307→  recordId?: string;
   308→  onUndo?: () => void;
   309→}): Promise<boolean> {
   310→  try {
   311→    await api.post("/api/undo", {
   312→      ...(spec.auditLogId ? { auditLogId: spec.auditLogId } : { module: spec.module, recordId: spec.recordId }),
   313→    });
   314→    toast.success("Entry reversed — everything is back to how it was", { icon: "↩️" });
   315→    spec.onUndo?.();
   316→    return true;
   317→  } catch (e) {
   318→    toast.error(errMessage(e));
   319→    return false;
   320→  }
   321→}

   326→export function useMutation() {
   327→  const [saving, setSaving] = useState(false);
   328→  const mutate = useCallback(async (
   329→    fn: () => Promise<unknown>,
   330→    success?: string,
   331→    undo?: (data: unknown) => UndoSpec | null | undefined
   332→  ) => {
   333→    setSaving(true);
   334→    try {
   335→      const data = await fn();
   336→      if (success) {
   337→        const spec = undo?.(data);
   338→        if (spec?.module && spec.recordId) {
   339→          toast.success(success, {
   340→            duration: 8000,
   341→            action: { label: "Undo", onClick: () => void undoRequest(spec) },
   342→          });
   343→        } else {
   344→          toast.success(success);
   345→        }
   346→      }
   347→      return { ok: true as const, data };
   348→    } catch (e) {
   349→      toast.error(errMessage(e));
   350→      return { ok: false as const, error: e };
   351→    } finally {
   352→      setSaving(false);
   353→    }
   354→  }, []);
   355→  return { mutate, saving };
   356→}

   358→export function useDebounced<T>(value: T, delay = 300): T {
   359→  const [v, setV] = useState(value);
   360→  useEffect(() => {
   361→    const t = setTimeout(() => setV(value), delay);
   362→    return () => clearTimeout(t);
   363→  }, [value, delay]);
   364→  return v;
   365→}

   373→export function fmtDay(s?: string | null): string {
   374→  if (!s) return "—";
   375→  const d = new Date(s.length === 10 ? `${s}T00:00:00` : s);
   376→  if (Number.isNaN(d.getTime())) return s;
   377→  return d.toLocaleDateString("en-IN", { day: "2-digit", month: "short", year: "numeric" });
   378→}

   384→  return d.toLocaleString("en-IN", { day: "2-digit", month: "short", hour: "2-digit", minute: "2-digit" });

   468→/** Payments at or above this amount ask for an explicit confirmation strip before saving. */
   469→export const LARGE_AMOUNT_THRESHOLD = 50_000;
   470→
   471→/**
   472→ * Guard rail for large money entries: render-props the dialog footer. When the
   473→ * parsed amount is ≥ LARGE_AMOUNT_THRESHOLD, the first press of the primary
   474→ * button shows an amber confirmation strip instead of submitting; the payment
   475→ * proceeds only via the strip's Confirm button. Under the threshold the
   476→ * confirm handler fires immediately (existing behaviour, unchanged).
   477→ */
   478→export function ConfirmAmount({ amount, subject, onSubmit, children }: {
   479→  amount: number;
   480→  subject: string;
   481→  /** the real save routine — runs only after explicit confirmation for large amounts */
   482→  onSubmit: () => void;
   483→  /** Render-prop footer: `confirm` wires into the primary save button, `edit`
   484→   *  collapses the echo strip, `armed` is true while the strip is showing. */
   485→  children: (h: { confirm: () => void; edit: () => void; armed: boolean }) => React.ReactNode;
   486→}) {
   487→  const [armed, setArmed] = useState(false);
   488→  const show = amount >= LARGE_AMOUNT_THRESHOLD && armed;
   489→
   490→  const confirm = () => {
   491→    if (amount >= LARGE_AMOUNT_THRESHOLD && !armed) {
   492→      setArmed(true);
   493→      return;
   494→    }
   495→    setArmed(false);
   496→    onSubmit();
   497→  };
   498→
   499→  return (
   500→    <>
   501→      {show && (
   502→        <div
   503→          role="alert"
   504→          className="flex flex-wrap items-center gap-2 rounded-xl border border-amber-300 bg-amber-50 px-3 py-2.5 dark:border-amber-900 dark:bg-amber-950/40"
   505→        >
   506→          <Banknote className="h-4 w-4 shrink-0 text-amber-600 dark:text-amber-400" aria-hidden />
   507→          <p className="min-w-0 flex-1 text-xs font-medium leading-snug text-amber-800 dark:text-amber-300">
   508→            You are about to record <span className="font-bold tabular-nums">{formatINR(amount)}</span> for{" "}
   509→            <span className="font-semibold">{subject}</span>.
   510→          </p>
   511→          <span className="flex shrink-0 items-center gap-1.5">
   512→            <Button
   513→              size="sm"
   514→              className="h-8 bg-amber-600 text-white hover:bg-amber-700 focus-visible:ring-amber-600/40"
   515→              onClick={confirm}
   516→            >
   517→              Confirm
   518→            </Button>
   519→            <Button
   520→              size="sm"
   521→              variant="outline"
   522→              className="h-8 border-amber-300 text-amber-800 hover:bg-amber-100 dark:border-amber-900 dark:text-amber-300 dark:hover:bg-amber-950"
   523→              onClick={() => setArmed(false)}
   524→            >
   525→              Edit amount
   526→            </Button>
   527→          </span>
   528→        </div>
   529→      )}
   530→      {children({ confirm, edit: () => setArmed(false), armed: show })}
   531→    </>
   532→  );
   533→}

   538→export function SelectInput({ value, onChange, options, placeholder = "Select…", className, disabled }: {
   539→  value: string;
   540→  onChange: (v: string) => void;
   541→  options: Option[];
   542→  placeholder?: string;
   543→  className?: string;
   544→  disabled?: boolean;
   545→}) {
   546→  return (
   547→    <Select
   548→      // Always controlled: `value || sentinel` keeps Radix from flipping
   549→      // between uncontrolled (undefined) and controlled when value is "".
   550→      // A non-matching sentinel still renders the placeholder visually.
   551→      value={value || EMPTY_SENTINEL}
   552→      onValueChange={(v) => onChange(v === EMPTY_SENTINEL ? "" : v)}
   553→      disabled={disabled}
   554→    >
   555→      <SelectTrigger className={cn("h-10 w-full", className)} aria-label={placeholder}>
   556→        <SelectValue placeholder={placeholder} />
   557→      </SelectTrigger>
   558→      <SelectContent position="popper" className="max-h-72">
   559→        {/* Radix forbids empty-string item values — map them to a sentinel. */}
   560→        {options.map((o) => (
   561→          <SelectItem key={o.value || EMPTY_SENTINEL} value={o.value || EMPTY_SENTINEL} className="text-sm">{o.label}</SelectItem>
   562→        ))}
   563→      </SelectContent>
   564→    </Select>
   565→  );
   566→}

   775→export function EmployeePicker({ employees, value, onChange, multi = true, height = "max-h-60", unavailable }: {
   776→  employees: PickerEmployee[];
   777→  value: string[];
   778→  onChange: (ids: string[]) => void;
   779→  multi?: boolean;
   780→  height?: string;
   781→  /** Return a short reason string to mark an employee as unselectable (e.g. already booked all day), or null when selectable. */
   782→  unavailable?: (e: PickerEmployee) => string | null;
   783→}) {
   784→  const [search, setSearch] = useState("");
   785→  const filtered = useMemo(() => {
   786→    const q = search.trim().toLowerCase();
   787→    if (!q) return employees;
   788→    return employees.filter((e) => `${e.fullName} ${e.code ?? ""}`.toLowerCase().includes(q));
   789→  }, [employees, search]);
   790→
   791→  const toggle = (id: string) => {
   792→    if (unavailable) {
   793→      const emp = employees.find((e) => e.id === id);
   794→      if (emp && unavailable(emp)) return; // blocked — ignore toggle
   795→    }
   796→    if (!multi) { onChange([id]); return; }
   797→    onChange(value.includes(id) ? value.filter((v) => v !== id) : [...value, id]);
   798→  };
   799→
   800→  return (
   801→    <div className="space-y-2">
   802→      <div className="relative">
   803→        <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" aria-hidden />
   804→        <Input
   805→          value={search}
   806→          onChange={(e) => setSearch(e.target.value)}
   807→          placeholder="Search employees…"
   808→          className="h-9 rounded-xl bg-card pl-9"
   809→          aria-label="Search employees"
   810→        />
   811→      </div>
   812→      {multi && value.length > 0 && (
   813→        <div className="flex items-center justify-between text-xs text-muted-foreground">
   814→          <span>{value.length} selected</span>
   815→          <button type="button" className="font-medium text-primary hover:underline" onClick={() => onChange([])}>
   816→            Clear all
   817→          </button>
   818→        </div>
   819→      )}
   820→      <div className={cn("overflow-y-auto rounded-xl border", height)} role="listbox" aria-multiselectable={multi}>
   821→        {filtered.length === 0 && (
   822→          <p className="p-6 text-center text-xs text-muted-foreground">No employees found</p>
   823→        )}
   824→        {filtered.map((e) => {
   825→          const on = value.includes(e.id);
   826→          const blockedReason = unavailable?.(e) ?? null;
   827→          const blocked = Boolean(blockedReason);
   828→          return (
   829→            <div
   830→              key={e.id}
   831→              role="option"
   832→              aria-selected={on}
   833→              aria-disabled={blocked || undefined}
   834→              tabIndex={blocked ? -1 : 0}
   835→              onClick={() => { if (!blocked) toggle(e.id); }}
   836→              onKeyDown={(ev) => { if (!blocked && (ev.key === "Enter" || ev.key === " ")) { ev.preventDefault(); toggle(e.id); } }}
   837→              className={cn(
   838→                "flex items-center gap-2.5 border-b px-3 py-2.5 transition-colors last:border-b-0",
   839→                blocked ? "cursor-not-allowed opacity-60" : "cursor-pointer hover:bg-muted/60",
   840→                on && "bg-primary/5"
   841→              )}
   842→            >
   843→              {multi ? (
   844→                <span className={cn(
   845→                  "flex h-4 w-4 shrink-0 items-center justify-center rounded border transition-colors",
   846→                  on ? "border-primary bg-primary text-primary-foreground" : "border-input bg-background"
   847→                )} aria-hidden>
   848→                  {on && <Check className="h-3 w-3" />}
   849→                </span>
   850→              ) : (
   851→                <span className={cn(
   852→                  "flex h-4 w-4 shrink-0 items-center justify-center rounded-full border transition-colors",
   853→                  on ? "border-primary" : "border-input"
   854→                )} aria-hidden>
   855→                  {on && <span className="h-2 w-2 rounded-full bg-primary" />}
   856→                </span>
   857→              )}
   858→              <div className="min-w-0 flex-1">
   859→                <p className="truncate text-sm font-medium">{e.fullName}</p>
   860→                <p className="truncate text-[11px] text-muted-foreground">
   861→                  {e.code ?? ""}
   862→                  {e.employmentType === "SALARIED"
   863→                    ? ` · Salaried ${formatINR(e.monthlySalary ?? 0)}/mo`
   864→                    : e.standardRate ? ` · ${formatINR(e.standardRate)}/shift` : ""}
   865→                  {e.hasContractor && e.contractorName ? ` · via ${e.contractorName}` : ""}
   866→                </p>
   867→              </div>
   868→              {blocked && blockedReason ? (
   869→                <span className="shrink-0 rounded-full bg-amber-100 px-2 py-0.5 text-[10px] font-semibold tabular-nums text-amber-800 dark:bg-amber-950 dark:text-amber-300">
   870→                  {blockedReason}
   871→                </span>
   872→              ) : (e.advanceBalance ?? 0) > 0 && (
   873→                <span className="shrink-0 rounded-full bg-red-50 px-2 py-0.5 text-[10px] font-medium tabular-nums text-red-600 dark:bg-red-950 dark:text-red-400">
   874→                  Adv {formatINR(e.advanceBalance ?? 0, { compact: true })}
   875→                </span>
   876→              )}
   877→            </div>
   878→          );
   879→        })}
   880→      </div>
   881→    </div>
   882→  );
   883→}

  1246→export function DeployWizard({ open, onOpenChange, defaultDate, defaultPropertyId, onDone }: {

  1249→  defaultDate?: string;

  1251→  onDone: () => void;

  1263→  // Same-day bookings for the picked date — across ALL properties. Drives the
  1264→  // "already booked" badges in the picker and the overlap guard in step 3.
  1265→  const [booked, setBooked] = useState<Record<string, { shift: string; propertyName: string }[]>>({});
  1266→  const [bookedLoadedFor, setBookedLoadedFor] = useState("");
  1267→
  1268→  // Load properties + reset on open (render-time state adjustment).
  1269→  const [prevOpen, setPrevOpen] = useState(open);
  1270→  if (open !== prevOpen) {
  1271→    setPrevOpen(open);
  1272→    if (open) {
  1273→      setStep(1);
  1274→      setPropertyId(defaultPropertyId ?? "");
  1275→      setDate(defaultDate ?? todayStr());
  1276→      setNotes("");
  1277→      setSelected([]);
  1278→      setRowState({});
  1279→      setDuplicates(null);
  1280→      setBooked({});

  1286→    if (!open) return;
  1287→    let cancelled = false;
  1288→    api.get<ListResp<PropertyRec>>("/api/properties" + qs({ status: "ACTIVE", pageSize: 200 }))
  1289→      .then((d) => { if (!cancelled) setProperties(d.items); })
  1290→      .catch(() => {});

  1292→  }, [open]);

  1294→  // Load the day's deployments (any property) whenever the wizard reaches the
  1295→  // employee step — the picker greys out fully-booked people, and step 3 uses
  1296→  // the same data to block overlapping shift choices.
  1297→  useEffect(() => {
  1298→    if (!open || step < 2) return;
  1299→    const key = `${date}|${selected.join(",")}`;
  1300→    if (bookedLoadedFor === key) return;
  1301→    let cancelled = false;
  1302→    api.get<ListResp<DeploymentRec>>("/api/deployments" + qs({ from: date, to: date, pageSize: 500 }))
  1303→      .then((d) => {
  1304→        if (cancelled) return;
  1305→        const map: Record<string, { shift: string; propertyName: string }[]> = {};
  1306→        for (const row of d.items) {
  1307→          const list = map[row.employeeId] ?? [];
  1308→          list.push({ shift: row.shift, propertyName: row.propertyName });
  1309→          map[row.employeeId] = list;
  1310→        }
  1311→        setBooked(map);
  1312→        setBookedLoadedFor(key);
  1313→      })
  1314→      .catch(() => { if (!cancelled) setBookedLoadedFor(key); });
  1315→    return () => { cancelled = true; };

  1317→
  1318→  const property = useMemo(() => properties.find((p) => p.id === propertyId) ?? null, [properties, propertyId]);
  1319→
  1320→  const selectedEmployees = useMemo(
  1321→    () => employees.filter((e) => selected.includes(e.id)),

  1329→  const fullyBooked = (id: string) => {
  1330→    const list = bookedList(id);
  1331→    return list.some((w) => w.shift === "FULL") ||
  1332→      (list.some((w) => w.shift === "DAY") && list.some((w) => w.shift === "NIGHT"));
  1333→  };

  1335→  const clashesFor = (id: string, shift: string) =>

  1347→  // booking (any property). Blocks the Deploy button until resolved.
  1348→  const conflictingSelected = useMemo(
  1349→    () => selectedEmployees.filter((e) => clashesFor(e.id, rowShift(e.id)).length > 0),
  1350→    [selectedEmployees, rowState, booked]
  1351→  );

  1353→    const o = rowState[e.id];
  1354→    if (o?.billingRate !== undefined && o?.billingRate !== "") return parseAmount(o.billingRate);
  1355→    return property?.billingRate ?? 0;
  1356→  };
  1357→  /** Salaried employees never earn per-shift payouts — their cost is the monthly salary. */
  1358→  const isSalariedRow = (e: PickerEmployee) => e.employmentType === "SALARIED";
  1359→  const rowPayoutRate = (e: PickerEmployee) => {
  1360→    if (isSalariedRow(e)) return 0;
  1361→    const o = rowState[e.id];
  1362→    if (o?.payoutRate !== undefined && o?.payoutRate !== "") return parseAmount(o.payoutRate);
  1363→    return e.standardRate ?? 0;
  1364→  };
  1365→  /** Contractor commission for a row — snapshotted per deployment from the employee master. */
  1366→  const rowContractorCut = (e: PickerEmployee) =>
  1367→    e.hasContractor && e.contractorName && (e.contractorRateCut ?? 0) > 0
  1368→      ? { name: e.contractorName, perShift: e.contractorRateCut ?? 0 }

  1370→

  1491→                  options={properties.map((p) => ({
