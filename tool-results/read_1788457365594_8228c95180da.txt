    41→    41→// Media-query rail detection for md–lg widths (tablets / narrow laptops):
    42→    42→// the sidebar becomes an icons-only rail there, while <md hides it entirely
    43→    43→// and lg+ shows the full collapsible sidebar. useSyncExternalStore keeps this
    44→    44→// SSR-safe (server snapshot = false, so hydration never mismatches).
    45→    45→const RAIL_QUERY = "(min-width: 768px) and (max-width: 1023.98px)";
    46→    46→function useIsRailWidth(): boolean {
    47→    47→  return useSyncExternalStore(
    48→    48→    (onStoreChange) => {
    49→    49→      const mq = window.matchMedia(RAIL_QUERY);
    50→    50→      mq.addEventListener("change", onStoreChange);
    51→    51→      return () => mq.removeEventListener("change", onStoreChange);
    52→    52→    },
    53→    53→    () => window.matchMedia(RAIL_QUERY).matches,
    54→    54→    () => false
    55→    55→  );
    56→    56→}

   176→   176→function TopBar({ onOpenMore, onOpenPalette, onOpenShortcuts }: { onOpenMore: () => void; onOpenPalette: () => void; onOpenShortcuts: () => void }) {
   177→   177→  const { owner, logout } = useAuth();
   178→   178→  const { navigate } = useNav();
   179→   179→  const { lang, setLang } = useLang();
   180→   180→  const [notifCount, setNotifCount] = useState(0);
   181→   181→  const [pulse, setPulse] = useState(false);
   182→   182→  const [undoOpen, setUndoOpen] = useState(false);
   183→   183→  const [notifRefresh, setNotifRefresh] = useState(0);
   184→   184→  const { canInstall, install } = usePwaInstall();
   185→   185→  const prevKeysRef = useRef<Set<string> | null>(null);
   186→   186→  const firstLoadRef = useRef(true);
   187→   187→  const pulseTimerRef = useRef<number | null>(null);
   188→   188→
   189→   189→  useEffect(() => {
   190→   190→    let alive = true;
   191→   191→    const load = async () => {
   192→   192→      try {
   193→   193→        const items = await import("@/lib/api-client").then(({ api }) =>
   194→   194→          api.get<AppNotification[]>("/api/notifications")
   195→   195→        );
   196→   196→        if (!alive || !Array.isArray(items)) return;
   197→   197→        // Respect muted categories (localStorage) — the bell only counts alerts
   198→   198→        // the owner hasn't muted from the notifications view.
   199→   199→        const muted = readMutedGroups();
   200→   200→        const visible = muted.length > 0 ? items.filter((n) => !muted.includes(n.severity)) : items;
   201→   201→        setNotifCount(visible.length);
   202→   202→        const prevKeys = prevKeysRef.current;
   203→   203→        prevKeysRef.current = new Set(items.map((n) => n.key));
   204→   204→        // Announce only genuine arrivals (skip initial load), when the tab is
   205→   205→        // visible and the alert isn't muted.
   206→   206→        if (!firstLoadRef.current && prevKeys && document.visibilityState === "visible") {
   207→   207→          const fresh = visible.filter((n) => !prevKeys.has(n.key));
   208→   208→          if (fresh.length > 0) {
   209→   209→            const rank = { CRITICAL: 0, WARNING: 1, INFO: 2 } as const;
   210→   210→            const headline =
   211→   211→              [...fresh].sort((a, b) => rank[a.severity] - rank[b.severity])[0] ?? fresh[0];
   212→   212→            toast(headline.title, {
   213→   213→              description:
   214→   214→                fresh.length > 1
   215→   215→                  ? `${headline.message} — and ${fresh.length - 1} more new alert${fresh.length === 2 ? "" : "s"}.`
   216→   216→                  : headline.message,
   217→   217→              icon: SEVERITY_ICON[headline.severity] ?? SEVERITY_ICON.INFO,
   218→   218→              action: { label: "View", onClick: () => navigate("notifications") },
   219→   219→            });
   220→   220→            // Bell ripple — brief ping so the eye is drawn to the new alert.
   221→   221→            setPulse(true);
   222→   222→            if (pulseTimerRef.current) window.clearTimeout(pulseTimerRef.current);
   223→   223→            pulseTimerRef.current = window.setTimeout(() => setPulse(false), 2400);
   224→   224→          }
   225→   225→        }
   226→   226→        firstLoadRef.current = false;
   227→   227→      } catch { /* ignore */ }
   228→   228→    };
   229→   229→    load();
   230→   230→    const t = setInterval(load, 60_000);
   231→   231→    // Re-check immediately when the user returns to the tab.
   232→   232→    const onVis = () => { if (document.visibilityState === "visible") void load(); };
   233→   233→    // Mute toggles + dismissals (this tab via custom event, other tabs via
   234→   234→    // storage) re-count instantly.
   235→   235→    const onMuteChange = () => { void load(); };
   236→   236→    document.addEventListener("visibilitychange", onVis);
   237→   237→    window.addEventListener(NOTIF_MUTED_EVENT, onMuteChange);
   238→   238→    window.addEventListener(NOTIF_CHANGED_EVENT, onMuteChange);
   239→   239→    window.addEventListener("storage", onMuteChange);
   240→   240→    return () => {
   241→   241→      alive = false; clearInterval(t);
   242→   242→      document.removeEventListener("visibilitychange", onVis);
   243→   243→      window.removeEventListener(NOTIF_MUTED_EVENT, onMuteChange);
   244→   244→      window.removeEventListener(NOTIF_CHANGED_EVENT, onMuteChange);
   245→   245→      window.removeEventListener("storage", onMuteChange);
   246→   246→      if (pulseTimerRef.current) window.clearTimeout(pulseTimerRef.current);
   247→   247→    };
   248→   248→  }, [navigate, notifRefresh]);
   249→   249→
   250→   250→  const initials = (owner?.name ?? "?").slice(0, 2).toUpperCase();
   251→   251→
   252→   252→  return (
   253→   253→    <>
   254→   254→    <header className="sticky top-0 z-40 border-b bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/80">
   255→   255→      <div className="mx-auto flex h-14 max-w-[1600px] items-center gap-2 px-3 sm:px-5">
   256→   256→        <button className="flex items-center gap-2 md:hidden" onClick={() => navigate("dashboard")} aria-label="BizHub home">
   257→   257→          <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-emerald-600 text-white font-bold">
   258→   258→            <span className="text-xs">B</span>
   259→   259→          </div>
   260→   260→          <span className="text-sm font-bold">BizHub</span>
   261→   261→        </button>
   262→   262→
   263→   263→        <Button variant="ghost" size="icon" className="hidden lg:inline-flex h-9 w-9 lg:hidden" aria-hidden tabIndex={-1} />
   264→   264→
   265→   265→        {/* Global search trigger — opens the command palette (Ctrl K) */}
   266→   266→        <button
   267→   267→          onClick={onOpenPalette}
   268→   268→          className="ml-0 hidden sm:flex h-9 min-w-0 flex-1 max-w-md items-center gap-2 rounded-xl border bg-muted/40 px-3 text-sm text-muted-foreground hover:bg-muted hover:border-primary/40 transition-colors"
   269→   269→          aria-label="Open command palette"
   270→   270→        >
   271→   271→          <Search className="h-4 w-4 shrink-0" />
   272→   272→          <span className="min-w-0 truncate whitespace-nowrap hidden md:inline">{t(lang, "topbar.search")}</span>

   480→   480→export function AppShell({ children }: { children: React.ReactNode }) {
   481→   481→  const { navigate, view } = useNav();
   482→   482→  const [collapsed, setCollapsed] = useState(false);
   483→   483→  // md–lg widths force the icons-only rail (no toggle there); lg+ keeps the
   484→   484→  // full sidebar with the manual collapse button.
   485→   485→  const railMode = useIsRailWidth();
   486→   486→  const railActive = railMode || collapsed;
   487→   487→  const [moreOpen, setMoreOpen] = useState(false);
   488→   488→  const [paletteOpen, setPaletteOpen] = useState(false);
   489→   489→  const [shortcutsOpen, setShortcutsOpen] = useState(false);
   490→   490→
   491→   491→  useEffect(() => {
   492→   492→    // "G then key" chord: pendingG holds the timeout id while waiting for the
   493→   493→    // second key (1s window). Cleared by typing anywhere else.
   494→   494→    let pendingG: number | null = null;
   495→   495→    const clearPendingG = () => {
   496→   496→      if (pendingG !== null) window.clearTimeout(pendingG);
   497→   497→      pendingG = null;
   498→   498→    };
   499→   499→    const GO_CHORDS: Record<string, string> = {
   500→   500→      d: "dashboard", e: "employees", p: "properties", t: "trips",
   501→   501→    };
   502→   502→    const isTypingTarget = (el: EventTarget | null): boolean => {
   503→   503→      const t = el as HTMLElement | null;
   504→   504→      return Boolean(
   505→   505→        t &&
   506→   506→        (t.tagName === "INPUT" || t.tagName === "TEXTAREA" || t.tagName === "SELECT" || t.isContentEditable)
   507→   507→      );
   508→   508→    };
   509→   509→    const onKey = (e: KeyboardEvent) => {
   510→   510→      if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === "k") {
   511→   511→        e.preventDefault();
   512→   512→        clearPendingG();
   513→   513→        setPaletteOpen((o) => !o);
   514→   514→        return;
   515→   515→      }
   516→   516→      // Palette is open → its input owns the keyboard; ignore global chords.
   517→   517→      if (paletteOpen) return;
   518→   518→      if (e.altKey || e.ctrlKey || e.metaKey) return;
   519→   519→      // Typing in a field always cancels any pending chord.
   520→   520→      if (isTypingTarget(e.target)) {
   521→   521→        clearPendingG();
   522→   522→        return;
   523→   523→      }
   524→   524→      if (pendingG !== null) {
   525→   525→        const target = GO_CHORDS[e.key.toLowerCase()];
   526→   526→        clearPendingG();
   527→   527→        if (target) {
   528→   528→          e.preventDefault();
   529→   529→          navigate(target);
   530→   530→        }
   531→   531→        return;
   532→   532→      }
   533→   533→      if (e.key === "g" || e.key === "G") {
   534→   534→        pendingG = window.setTimeout(clearPendingG, 1000);
   535→   535→        return;
   536→   536→      }
   537→   537→      if (e.key === "?") {
   538→   538→        e.preventDefault();
   539→   539→        setShortcutsOpen(true);
   540→   540→      }
   541→   541→    };
   542→   542→    window.addEventListener("keydown", onKey);
   543→   543→    return () => {
   544→   544→      window.removeEventListener("keydown", onKey);
   545→   545→      clearPendingG();
   546→   546→    };
   547→   547→  }, [navigate, paletteOpen]);
   548→   548→
   549→   549→  const currentView = getView(view);
   550→   550→  const isWide = ["dashboard", "manpower", "transport"].includes(view);
   551→   551→
   552→   552→  return (
   553→   553→    <div className="flex min-h-screen bg-muted/30">
   554→   554→      {/* Desktop sidebar — icons-only rail at md–lg, full (collapsible) at lg+ */}
   555→   555→      <aside className={cn("sticky top-0 hidden h-screen shrink-0 border-r bg-background transition-all md:block", railActive ? "w-[68px]" : "w-60")}>
   556→   556→        <SideNav collapsed={railActive} onToggle={railMode ? undefined : () => setCollapsed((c) => !c)} />
   557→   557→      </aside>
   558→   558→
   559→   559→      <div className="flex min-w-0 flex-1 flex-col">
   560→   560→        <TopBar onOpenMore={() => setMoreOpen(true)} onOpenPalette={() => setPaletteOpen(true)} onOpenShortcuts={() => setShortcutsOpen(true)} />
   561→   561→        <BusinessBanner />
   562→   562→        <main
   563→   563→          id="main-scroll"
   564→   564→          className={cn("mx-auto w-full max-w-[1600px] flex-1 px-3 pb-24 pt-3 sm:px-5 sm:pt-4 md:pb-10", !isWide && "max-w-[1400px]")}
   565→   565→          aria-label={currentView?.label ?? "Content"}
   566→   566→        >
   567→   567→          {/* Keyed by view → remounts on navigation and plays the enter animation */}
   568→   568→          <div key={view} className="view-enter">
   569→   569→            {children}
   570→   570→          </div>
   571→   571→        </main>
   572→   572→        <footer className="mt-auto hidden border-t bg-background py-3 md:block">
   573→   573→          <div className="mx-auto flex max-w-[1600px] items-center justify-between px-5 text-[11px] text-muted-foreground">
   574→   574→            <p>BizHub — one control center for Manpower & Transport businesses.</p>
   575→   575→            <p className="tabular-nums">{new Date().getFullYear()} · All financial actions are audited.</p>
   576→   576→          </div>
   577→   577→        </footer>
   578→   578→      </div>
   579→   579→
   580→   580→      {/* Mobile bottom nav */}
   581→   581→      <BottomNav onOpenMore={() => setMoreOpen(true)} />
   582→   582→
   583→   583→      {/* Global command palette (Ctrl/Cmd+K, search buttons) */}
   584→   584→      <CommandPalette open={paletteOpen} onOpenChange={setPaletteOpen} />
   585→   585→
   586→   586→      {/* Keyboard shortcuts overlay (? key or owner menu) */}
   587→   587→      <ShortcutsDialog open={shortcutsOpen} onOpenChange={setShortcutsOpen} />
   588→   588→
   589→   589→      {/* Mobile "More" sheet with full nav */}
   590→   590→      <Sheet open={moreOpen} onOpenChange={setMoreOpen}>
   591→   591→        <SheetTrigger className="hidden" aria-hidden />
   592→   592→        <SheetContent side="left" className="w-[280px] p-0">
   593→   593→          <SheetHeader className="sr-only">
   594→   594→            <SheetTitle>All navigation</SheetTitle>
   595→   595→          </SheetHeader>
   596→   596→          <SideNav onNavigate={() => setMoreOpen(false)} />
   597→   597→        </SheetContent>
   598→   598→      </Sheet>
   599→   599→    </div>
   600→   600→  );
   601→   601→}
