   265→export function errMessage(e: unknown): string {
   266→  if (e instanceof ApiError) return e.message;
   267→  if (e instanceof Error) return e.message;
   268→  return "Something went wrong";

   271→/** Fetch-on-mount hook with loading/error/reload. Toasts load failures. */
   272→export function useAsync<T>(loader: () => Promise<T>, deps: React.DependencyList) {
   273→  const [data, setData] = useState<T | null>(null);
   274→  const [loading, setLoading] = useState(true);
   275→  const [error, setError] = useState<string | null>(null);
   276→  const load = useCallback(async () => {
   277→    setLoading(true);
   278→    try {
   279→      const d = await loader();
   280→      setData(d);
   281→      setError(null);
   282→    } catch (e) {
   283→      const msg = errMessage(e);
   284→      setError(msg);
   285→      toast.error(msg);
   286→    } finally {
   287→      setLoading(false);
   288→    }
   289→  }, deps);
   290→  useEffect(() => { void load(); }, [load]);
   291→  return { data, loading, error, reload: load, setData };
   292→}

   294→/** Mutation wrapper: consistent error toasts, optional success toast.
   295→ *  Pass an `undo` resolver to attach an "Undo" action to the success toast —
   296→ *  it reverses the entry server-side (zero mismatch) and calls onUndo. */
   297→export interface UndoSpec {
   298→  module: string; // PAYMENT / ADVANCE / EXPENSE / DEPLOYMENT / TRIP / EMI / MAINTENANCE / ADJUSTMENT
   299→  recordId?: string;
   300→  onUndo?: () => void;
   301→}
   302→
   303→export async function undoRequest(spec: {
   304→  auditLogId?: string;
   305→  module?: string;
   306→  recordId?: string;
   307→  onUndo?: () => void;
   308→}): Promise<boolean> {
   309→  try {
   310→    await api.post("/api/undo", {
   311→      ...(spec.auditLogId ? { auditLogId: spec.auditLogId } : { module: spec.module, recordId: spec.recordId }),
   312→    });
   313→    toast.success("Entry reversed — everything is back to how it was", { icon: "↩️" });
   314→    spec.onUndo?.();
   315→    return true;
   316→  } catch (e) {
   317→    toast.error(errMessage(e));
   318→    return false;
   319→  }
   320→}

   322→export function useMutation() {
   323→  const [saving, setSaving] = useState(false);
   324→  const mutate = useCallback(async (
   325→    fn: () => Promise<unknown>,
   326→    success?: string,
   327→    undo?: (data: unknown) => UndoSpec | null | undefined
   328→  ) => {
   329→    setSaving(true);
   330→    try {
   331→      const data = await fn();
   332→      if (success) {
   333→        const spec = undo?.(data);
   334→        if (spec?.module && spec.recordId) {
   335→          toast.success(success, {
   336→            duration: 8000,
   337→            action: { label: "Undo", onClick: () => void undoRequest(spec) },
   338→          });
   339→        } else {
   340→          toast.success(success);
   341→        }
   342→      }
   343→      return { ok: true as const, data };
   344→    } catch (e) {
   345→      toast.error(errMessage(e));
   346→      return { ok: false as const, error: e };
   347→    } finally {
   348→      setSaving(false);
   349→    }
   350→  }, []);
   351→  return { mutate, saving };
   352→}

   354→export function useDebounced<T>(value: T, delay = 300): T {
   355→  const [v, setV] = useState(value);
   356→  useEffect(() => {
   357→    const t = setTimeout(() => setV(value), delay);
   358→    return () => clearTimeout(t);
   359→  }, [value, delay]);
   360→  return v;
   361→}

   364→  const d = new Date();

   367→}

   369→export function fmtDay(s?: string | null): string {
   370→  if (!s) return "—";
   371→  const d = new Date(s.length === 10 ? `${s}T00:00:00` : s);
   372→  if (Number.isNaN(d.getTime())) return s;
   373→  return d.toLocaleDateString("en-IN", { day: "2-digit", month: "short", year: "numeric" });
   374→}

   376→export function fmtDateTime(s?: string | null): string {
   377→  if (!s) return "—";
   378→  const d = new Date(s);
   379→  if (Number.isNaN(d.getTime())) return s;
   380→  return d.toLocaleString("en-IN", { day: "2-digit", month: "short", hour: "2-digit", minute: "2-digit" });
   381→}

   414→// Sentinel for "no selection" options (Radix Select forbids empty-string values).
   415→export const EMPTY_SENTINEL = "__all__";
   416→
   417→export function SelectInput({ value, onChange, options, placeholder = "Select…", className, disabled }: {
   418→  value: string;
   419→  onChange: (v: string) => void;
   420→  options: Option[];
   421→  placeholder?: string;
   422→  className?: string;
   423→  disabled?: boolean;
   424→}) {
   425→  return (
   426→    <Select
   427→      // Always controlled: `value || sentinel` keeps Radix from flipping
   428→      // between uncontrolled (undefined) and controlled when value is "".
   429→      // A non-matching sentinel still renders the placeholder visually.
   430→      value={value || EMPTY_SENTINEL}
   431→      onValueChange={(v) => onChange(v === EMPTY_SENTINEL ? "" : v)}
   432→      disabled={disabled}
   433→    >
   434→      <SelectTrigger className={cn("h-10 w-full", className)} aria-label={placeholder}>
   435→        <SelectValue placeholder={placeholder} />
   436→      </SelectTrigger>
   437→      <SelectContent position="popper" className="max-h-72">
   438→        {/* Radix forbids empty-string item values — map them to a sentinel. */}
   439→        {options.map((o) => (
   440→          <SelectItem key={o.value || EMPTY_SENTINEL} value={o.value || EMPTY_SENTINEL} className="text-sm">{o.label}</SelectItem>
   441→        ))}
   442→      </SelectContent>
   443→    </Select>
   444→  );
   445→}

  1335→              </Button>
  1336→            )}
  1337→          </div>
  1338→        </div>
  1339→      </DialogContent>
  1340→    </Dialog>
