    58→function attClass(n: number): string {
    59→  if (n <= 0) return ATT_HEAT[0];
    60→  if (n === 1) return ATT_HEAT[1];
    61→  if (n === 2) return ATT_HEAT[2];
    62→  return ATT_HEAT[3];

    65→const WD_LETTERS = ["S", "M", "T", "W", "T", "F", "S"];
    66→
    67→function monthLabel(month: string): string {
    68→  const [y, m] = month.split("-").map(Number);
    69→  return new Date(y, (m ?? 1) - 1, 1).toLocaleDateString("en-IN", { month: "long", year: "numeric" });
    70→}
    71→
    72→function shiftMonth(month: string, delta: number): string {
    73→  const [y, m] = month.split("-").map(Number);
    74→  const d = new Date(y, (m ?? 1) - 1 + delta, 1);
    75→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`;
    76→}
    77→
    78→function isWeekend(dateStr: string): boolean {
    79→  const [y, m, d] = dateStr.split("-").map(Number);
    80→  const wd = new Date(y, (m ?? 1) - 1, d ?? 1).getDay();
    81→  return wd === 0 || wd === 6;
    82→}
    83→
    84→function monthBounds(offset = 0): { from: string; to: string } {
    85→  const now = new Date();
    86→  const first = new Date(now.getFullYear(), now.getMonth() + offset, 1);
    87→  const last = new Date(now.getFullYear(), now.getMonth() + offset + 1, 0);
    88→  const f = (d: Date) => `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;
    89→  return { from: f(first), to: f(last) };
    90→}

   457→          <>
   458→            <Card className="border-border/70 shadow-sm"><CardContent className="p-3 sm:p-4">
   459→              <p className="text-[11px] sm:text-xs text-muted-foreground">Deployments</p>
   460→              <p className="mt-1 text-lg sm:text-xl font-bold tabular-nums">{totals?.count ?? rows.length}</p>
   461→            </CardContent></Card>
   462→            <Card className="border-border/70 shadow-sm"><CardContent className="p-3 sm:p-4">
   463→              <p className="text-[11px] sm:text-xs text-muted-foreground">Billing</p>
   464→              <p className="mt-1 text-lg sm:text-xl font-bold tabular-nums">{formatINR(totals?.billing ?? 0, { compact: true })}</p>
   465→            </CardContent></Card>
   466→            <Card className="border-border/70 shadow-sm"><CardContent className="p-3 sm:p-4">
   467→              <p className="text-[11px] sm:text-xs text-muted-foreground">Payout</p>
   468→              <p className="mt-1 text-lg sm:text-xl font-bold tabular-nums">{formatINR(totals?.payout ?? 0, { compact: true })}</p>
   469→            </CardContent></Card>
   470→            <Card className="border-border/70 shadow-sm"><CardContent className="p-3 sm:p-4">
   471→              <p className="text-[11px] sm:text-xs text-muted-foreground">Margin</p>
   472→              <p className={cn("mt-1 text-lg sm:text-xl font-bold tabular-nums", (totals?.margin ?? 0) >= 0 ? "text-emerald-600 dark:text-emerald-400" : "text-red-600 dark:text-red-400")}>
   473→                {formatINR(totals?.margin ?? 0, { compact: true })}
   474→              </p>
   475→            </CardContent></Card>
   476→          </>
   477→        )}
   478→      </div>
   479→
   480→      {error && (
   481→        <div className="rounded-xl border border-red-200 bg-red-50/60 p-4 text-center dark:border-red-900 dark:bg-red-950/30">
   482→          <p className="text-xs text-red-700 dark:text-red-300">{error}</p>
   483→          <Button variant="outline" size="sm" className="mt-2 h-8" onClick={() => void load()}>Retry</Button>
   484→        </div>
   485→      )}
   486→
   487→      {!error && (
   488→        <Card>
   489→          <CardContent className="p-3 sm:p-4">
   490→            <DataTable
   491→              columns={columns}
   492→              rows={rows}
   493→              rowKey={(r) => r.id}
   494→              onRowClick={openDetail}
   495→              exportName="deployments"
   496→              loading={loading}
   497→              emptyIcon={CalendarCheck}
   498→              emptyTitle="No deployments for this filter"
   499→              emptyDescription="Use the Deploy Employees button to create today's roster."
   500→            />
   501→          </CardContent>
   502→        </Card>
   503→      )}
   504→
   505→      <DeployWizard open={wizardOpen} onOpenChange={setWizardOpen} defaultDate={date || undefined} defaultPropertyId={propertyId || undefined} onDone={load} />
   506→
   507→      {/* Detail dialog */}
   508→      <Dialog open={Boolean(detail)} onOpenChange={(v) => !v && setDetail(null)}>
   509→        <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-lg">
   510→          {detail && (
   511→            <>
   512→              <DialogHeader>
   513→                <DialogTitle className="flex flex-wrap items-center gap-2">
   514→                  {detail.employeeName}
   515→                  <StatusBadge status={detail.paidStatus ?? "UNPAID"} />
   516→                </DialogTitle>
   517→                <DialogDescription>
   518→                  {fmtDay(detail.date)} · {detail.propertyName} · {detail.shift === "FULL" ? "Full (day + night)" : detail.shift} shift
   519→                </DialogDescription>
   520→              </DialogHeader>
   521→
   522→              <div className="grid grid-cols-3 gap-2 text-center">
   523→                <div className="rounded-xl bg-muted/60 p-2.5">
   524→                  <p className="text-[10px] uppercase text-muted-foreground">Billing</p>
   525→                  <p className="text-sm font-bold tabular-nums">{formatINR(detail.billingAmount ?? 0)}</p>
   526→                  <p className="text-[10px] text-muted-foreground tabular-nums">{formatINR(detail.billingRate ?? 0)} × {detailUnits}</p>
   527→                </div>
   528→                <div className="rounded-xl bg-muted/60 p-2.5">
   529→                  <p className="text-[10px] uppercase text-muted-foreground">Payout</p>
   530→                  <p className="text-sm font-bold tabular-nums">{formatINR(detail.payoutAmount ?? 0)}</p>
   531→                  <p className="text-[10px] text-muted-foreground tabular-nums">{formatINR(detail.payoutRate ?? 0)} × {detailUnits}</p>
   532→                </div>
   533→                <div className="rounded-xl bg-muted/60 p-2.5">
   534→                  <p className="text-[10px] uppercase text-muted-foreground">Margin</p>
   535→                  <p className={cn("text-sm font-bold tabular-nums", (detail.billingAmount ?? 0) - (detail.payoutAmount ?? 0) >= 0 ? "text-emerald-600 dark:text-emerald-400" : "text-red-600 dark:text-red-400")}>
   536→                    {formatINR((detail.billingAmount ?? 0) - (detail.payoutAmount ?? 0))}
   537→                  </p>
   538→                  <p className="text-[10px] text-muted-foreground">this record</p>
   539→                </div>
   540→              </div>
   541→
   542→              {(detail.adjustmentAmount ?? 0) !== 0 && (
   543→                <p className="rounded-lg bg-amber-50 px-3 py-2 text-xs text-amber-800 dark:bg-amber-950/40 dark:text-amber-300">
   544→                  Adjustment {formatINR(detail.adjustmentAmount ?? 0)}{detail.adjustmentNote ? ` — ${detail.adjustmentNote}` : ""}
   545→                </p>
   546→              )}
   547→              {detail.notes && <p className="text-xs text-muted-foreground">Notes: {detail.notes}</p>}
   548→
   549→              <Separator />
   550→
   551→              {editRates ? (
   552→                <div className="space-y-3">
   553→                  <div className="space-y-1.5">
   554→                    <Label className="text-xs">Shift</Label>
   555→                    <div className="flex rounded-lg border bg-muted/50 p-1" role="group" aria-label="Shift">
   556→                      {SHIFT_OPTIONS.map((s) => (
   557→                        <button
   558→                          key={s.value}
   559→                          type="button"
   560→                          className={cn(
   561→                            "min-h-9 flex-1 rounded-md text-xs font-medium transition-colors",
   562→                            rateForm.shift === s.value ? "bg-primary text-primary-foreground shadow-sm" : "text-muted-foreground hover:text-foreground"
   563→                          )}
   564→                          aria-pressed={rateForm.shift === s.value}
   565→                          onClick={() => setRateForm((f) => ({ ...f, shift: s.value }))}
   566→                        >
   567→                          {s.label}
   568→                        </button>
   569→                      ))}
   570→                    </div>
   571→                  </div>
   572→                  <div className="grid gap-3 sm:grid-cols-2">
   573→                    <div className="space-y-1.5">
   574→                      <Label className="text-xs">Billing rate (₹/shift)</Label>
   575→                      <Input type="number" inputMode="numeric" className="h-9" value={rateForm.billingRate} onChange={(e) => setRateForm((f) => ({ ...f, billingRate: e.target.value }))} />
   576→                    </div>
   577→                    <div className="space-y-1.5">
   578→                      <Label className="text-xs">Payout rate (₹/shift)</Label>
   579→                      <Input type="number" inputMode="numeric" className="h-9" value={rateForm.payoutRate} onChange={(e) => setRateForm((f) => ({ ...f, payoutRate: e.target.value }))} />
   580→                    </div>
   581→                    <div className="space-y-1.5">
   582→                      <Label className="text-xs">Adjustment amount (± ₹)</Label>
   583→                      <Input type="number" inputMode="numeric" className="h-9" value={rateForm.adjustmentAmount} onChange={(e) => setRateForm((f) => ({ ...f, adjustmentAmount: e.target.value }))} placeholder="e.g. -100 or 250" />
   584→                    </div>
   585→                    <div className="space-y-1.5">
   586→                      <Label className="text-xs">Adjustment note</Label>
   587→                      <Input className="h-9" value={rateForm.adjustmentNote} onChange={(e) => setRateForm((f) => ({ ...f, adjustmentNote: e.target.value }))} />
   588→                    </div>
   589→                    <div className="space-y-1.5 sm:col-span-2">
   590→                      <Label className="text-xs">Notes</Label>
   591→                      <Input className="h-9" value={rateForm.notes} onChange={(e) => setRateForm((f) => ({ ...f, notes: e.target.value }))} />
   592→                    </div>
   593→                  </div>
   594→                  <p className="text-[11px] text-muted-foreground">Amounts recompute as rate × shift units (Full = 2) + adjustment.</p>
   595→                  <div className="flex gap-2">
   596→                    <Button variant="outline" className="min-h-10 flex-1" onClick={() => setEditRates(false)}>Cancel</Button>
   597→                    <Button className="min-h-10 flex-1" onClick={saveRates} disabled={saving}>{saving ? "Saving…" : "Save"}</Button>
   598→                  </div>
   599→                </div>
   600→              ) : (
   601→                <div className="space-y-2.5">
   602→                  <div className="flex items-center justify-between gap-2">
   603→                    <p className="text-[11px] text-muted-foreground">
   604→                      By {detail.createdByName ?? "—"} · {fmtDateTime(detail.createdAt)}
   605→                    </p>
   606→                    <div className="flex gap-1.5">
   607→                      <Button variant="outline" size="sm" className="h-8 gap-1.5" onClick={() => setEditRates(true)}>
   608→                        <Pencil className="h-3 w-3" aria-hidden />Edit
   609→                      </Button>
   610→                      <Button variant="outline" size="sm" className="h-8 gap-1.5 border-red-200 text-red-600 hover:bg-red-50 dark:border-red-900 dark:text-red-400 dark:hover:bg-red-950" onClick={() => setConfirmDelete(true)}>
   611→                        <Trash2 className="h-3 w-3" aria-hidden />Remove
   612→                      </Button>
   613→                    </div>
   614→                  </div>
   615→                  <p className="rounded-lg bg-muted/60 px-3 py-2 text-[11px] text-muted-foreground">
   616→                    Removed by mistake? The toast Undo (or Audit Log) restores it exactly — amounts, allocation and totals included.
   617→                  </p>
   618→                </div>
   619→              )}
   620→            </>
   621→          )}
   622→        </DialogContent>
   623→      </Dialog>
   624→
   625→      {/* Delete confirmation */}
   626→      <AlertDialog open={confirmDelete} onOpenChange={setConfirmDelete}>
   627→        <AlertDialogContent>
   628→          <AlertDialogHeader>
   629→            <AlertDialogTitle>Remove this deployment?</AlertDialogTitle>
   630→            <AlertDialogDescription>
   631→              {detail && `${detail.employeeName} — ${fmtDay(detail.date)} · ${detail.propertyName} · ${detail.shift} shift (${formatINR(detail.billingAmount ?? 0)} billing).`} The entry is snapshotted, so Undo restores it with zero mismatch.
   632→            </AlertDialogDescription>
   633→          </AlertDialogHeader>
   634→          <AlertDialogFooter>
   635→            <AlertDialogCancel className="min-h-10">Cancel</AlertDialogCancel>
   636→            <AlertDialogAction className="min-h-10 bg-red-600 text-white hover:bg-red-700" onClick={() => void deleteDeployment()}>
   637→              {saving ? "Removing…" : "Remove entry"}
   638→            </AlertDialogAction>
   639→          </AlertDialogFooter>
   640→        </AlertDialogContent>
   641→      </AlertDialog>
   642→    </div>
   643→  );
   644→}
