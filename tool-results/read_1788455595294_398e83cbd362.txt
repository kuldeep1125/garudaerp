   103→function greetingForHour(): string {
   104→  const h = new Date().getHours();
   105→  if (h < 5) return "Burning the midnight oil";
   106→  if (h < 12) return "Good morning";
   107→  if (h < 17) return "Good afternoon";
   108→  return "Good evening";
   109→}

   111→function fmtRel(iso: string): string {
   112→  const diff = Date.now() - new Date(iso).getTime();
   113→  const mins = Math.floor(diff / 60_000);
   114→  if (mins < 1) return "just now";
   115→  if (mins < 60) return `${mins}m ago`;
   116→  const hrs = Math.floor(mins / 60);
   117→  if (hrs < 24) return `${hrs}h ago`;
   118→  const days = Math.floor(hrs / 24);
   119→  if (days < 7) return `${days}d ago`;
   120→  return new Date(iso).toLocaleDateString("en-IN", { day: "numeric", month: "short" });
   121→}

   127→// Month-over-month delta chip: ▲/▼ with % — colored by whether the move is good.
   128→function Delta({ pct, goodUp = true }: { pct: number | null; goodUp?: boolean }) {
   129→  if (pct === null) {
   130→    return <span className="rounded-full bg-muted px-1.5 py-0.5 text-[10px] font-semibold text-muted-foreground">new</span>;
   131→  }
   132→  if (pct === 0) {
   133→    return (
   134→      <span className="inline-flex items-center gap-0.5 rounded-full bg-muted px-1.5 py-0.5 text-[10px] font-semibold text-muted-foreground">
   135→        <Minus className="h-2.5 w-2.5" aria-hidden />0%
   136→      </span>
   137→    );
   138→  }
   139→  const up = pct > 0;
   140→  const good = up === goodUp;
   141→  const Icon = up ? TrendingUp : TrendingDown;
   142→  return (
   143→    <span
   144→      className={cn(
   145→        "inline-flex items-center gap-0.5 rounded-full px-1.5 py-0.5 text-[10px] font-bold tabular-nums",
   146→        good
   147→          ? "bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-300"
   148→          : "bg-red-100 text-red-700 dark:bg-red-950 dark:text-red-300"
   149→      )}
   150→    >
   151→      <Icon className="h-2.5 w-2.5" aria-hidden />
   152→      {Math.abs(pct)}%
   153→    </span>
   154→  );
   155→}
   156→
   157→function pctOf(cur: number, prev: number): number | null {
   158→  if (prev === 0) return cur === 0 ? 0 : null;
   159→  return Math.round(((cur - prev) / Math.abs(prev)) * 100);
   160→}

   162→function monthShort(month: string): string {
   163→  const [y, m] = month.split("-").map(Number);
   164→  return new Date(y, m - 1, 1).toLocaleDateString("en-IN", { month: "short" });
   165→}
