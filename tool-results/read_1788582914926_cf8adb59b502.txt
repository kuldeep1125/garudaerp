    66→function rangeDates(r: RangeKey): { from: string; to: string } {
    67→  const now = new Date();
    68→  const f = (d: Date) => `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;
    69→  if (r === "today") { const t = f(now); return { from: t, to: t }; }
    70→  if (r === "yesterday") return { from: f(new Date(now.getFullYear(), now.getMonth(), now.getDate() - 1)), to: f(new Date(now.getFullYear(), now.getMonth(), now.getDate() - 1)) };
    71→  if (r === "week") return { from: f(new Date(now.getFullYear(), now.getMonth(), now.getDate() - 6)), to: f(now) };
    72→  if (r === "lastweek") return { from: f(new Date(now.getFullYear(), now.getMonth(), now.getDate() - 13)), to: f(new Date(now.getFullYear(), now.getMonth(), now.getDate() - 7)) };
    73→  if (r === "lastmonth") return { from: f(new Date(now.getFullYear(), now.getMonth() - 1, 1)), to: f(new Date(now.getFullYear(), now.getMonth(), 0)) };
    74→  return { from: f(new Date(now.getFullYear(), now.getMonth(), 1)), to: f(new Date(now.getFullYear(), now.getMonth() + 1, 0)) };
    75→}

   118→        date: expense?.date?.slice(0, 10) ?? todayStr(),
   119→        business: expense?.business ?? "MANPOWER",
   120→        categoryId: expense?.categoryId ?? "",
   121→        amount: expense ? String(expense.amount) : "",

   180→            <SelectInput
   181→              value={form.categoryId}
   182→              onChange={set("categoryId")}

   184→              options={categories.map((c) => ({ label: c.name, value: c.id }))}

   186→          </Field>
   187→          <Field label="Amount (₹)" required>
   188→            <MoneyInput value={form.amount} onChange={set("amount")} min={1} className="h-10" />

   190→          <Field label="Method">
   191→            <SelectInput value={form.method} onChange={set("method")} options={METHOD_OPTIONS} />

   196→                value={form.vehicleId}

   204→            <Input value={form.description} onChange={(e) => set("description")(e.target.value)} className="h-10" placeholder="What was this expense for?" />

   211→          <Button variant="outline" className="min-h-10 flex-1 sm:flex-none" onClick={() => onOpenChange(false)}>Cancel</Button>
   212→          <Button className="min-h-10 flex-1 sm:flex-none" onClick={submit} disabled={saving}>
   213→            {saving ? "Saving…" : editing ? "Save changes" : "Add expense"}
   214→          </Button>
   215→        </DialogFooter>
   216→      </DialogContent>
   217→    </Dialog>
   218→  );
   219→}
