   252→   252→/** Fetch-on-mount hook with loading/error/reload. Toasts load failures. */
   253→   253→export function useAsync<T>(loader: () => Promise<T>, deps: React.DependencyList) {
   254→   254→  const [data, setData] = useState<T | null>(null);
   255→   255→  const [loading, setLoading] = useState(true);
   256→   256→  const [error, setError] = useState<string | null>(null);
   257→   257→  const load = useCallback(async () => {
   258→   258→    setLoading(true);
   259→   259→    try {
   260→   260→      const d = await loader();
   261→   261→      setData(d);
   262→   262→      setError(null);
   263→   263→    } catch (e) {
   264→   264→      const msg = errMessage(e);
   265→   265→      setError(msg);
   266→   266→      toast.error(msg);
   267→   267→    } finally {
   268→   268→      setLoading(false);
   269→   269→    }
   270→   270→  }, deps);
   271→   271→  useEffect(() => { void load(); }, [load]);
   272→   272→  return { data, loading, error, reload: load, setData };
   273→   273→}

   280→   280→  recordId?: string;

   286→   286→  module?: string;
   287→   287→  recordId?: string;

   293→   293→    });

   295→   295→    spec.onUndo?.();

   303→   303→export function useMutation() {
   304→   304→  const [saving, setSaving] = useState(false);
   305→   305→  const mutate = useCallback(async (
   306→   306→    fn: () => Promise<unknown>,
   307→   307→    success?: string,
   308→   308→    undo?: (data: unknown) => UndoSpec | null | undefined
   309→   309→  ) => {
   310→   310→    setSaving(true);
   311→   311→    try {
   312→   312→      const data = await fn();
   313→   313→      if (success) {
   314→   314→        const spec = undo?.(data);
   315→   315→        if (spec?.module && spec.recordId) {
   316→   316→          toast.success(success, {
   317→   317→            duration: 8000,
   318→   318→            action: { label: "Undo", onClick: () => void undoRequest(spec) },
   319→   319→          });
   320→   320→        } else {
   321→   321→          toast.success(success);
   322→   322→        }
   323→   323→      }
   324→   324→      return { ok: true as const, data };
   325→   325→    } catch (e) {
   326→   326→      toast.error(errMessage(e));
   327→   327→      return { ok: false as const, error: e };
   328→   328→    } finally {
   329→   329→      setSaving(false);
   330→   330→    }
   331→   331→  }, []);
   332→   332→  return { mutate, saving };
   333→   333→}
