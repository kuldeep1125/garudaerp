   246→export function errMessage(e: unknown): string {
   247→  if (e instanceof ApiError) return e.message;
   248→  if (e instanceof Error) return e.message;
   249→  return "Something went wrong";
   250→}

   252→/** Fetch-on-mount hook with loading/error/reload. Toasts load failures. */
   253→export function useAsync<T>(loader: () => Promise<T>, deps: React.DependencyList) {
   254→  const [data, setData] = useState<T | null>(null);
   255→  const [loading, setLoading] = useState(true);
   256→  const [error, setError] = useState<string | null>(null);
   257→  const load = useCallback(async () => {
   258→    setLoading(true);
   259→    try {
   260→      const d = await loader();
   261→      setData(d);
   262→      setError(null);
   263→    } catch (e) {
   264→      const msg = errMessage(e);
   265→      setError(msg);
   266→      toast.error(msg);
   267→    } finally {
   268→      setLoading(false);
   269→    }
   270→  }, deps);
   271→  useEffect(() => { void load(); }, [load]);
   272→  return { data, loading, error, reload: load, setData };
   273→}

   275→/** Mutation wrapper: consistent error toasts, optional success toast.
   276→ *  Pass an `undo` resolver to attach an "Undo" action to the success toast —
   277→ *  it reverses the entry server-side (zero mismatch) and calls onUndo. */
   278→export interface UndoSpec {
   279→  module: string; // PAYMENT / ADVANCE / EXPENSE / DEPLOYMENT / TRIP / EMI / MAINTENANCE / ADJUSTMENT
   280→  recordId?: string;
   281→  onUndo?: () => void;
   282→}
   283→
   284→export async function undoRequest(spec: {
   285→  auditLogId?: string;
   286→  module?: string;
   287→  recordId?: string;
   288→  onUndo?: () => void;
   289→}): Promise<boolean> {
   290→  try {
   291→    await api.post("/api/undo", {
   292→      ...(spec.auditLogId ? { auditLogId: spec.auditLogId } : { module: spec.module, recordId: spec.recordId }),
   293→    });
   294→    toast.success("Entry reversed — everything is back to how it was", { icon: "↩️" });
   295→    spec.onUndo?.();
   296→    return true;
   297→  } catch (e) {
   298→    toast.error(errMessage(e));
   299→    return false;
   300→  }
   301→}

   303→export function useMutation() {
   304→  const [saving, setSaving] = useState(false);
   305→  const mutate = useCallback(async (
   306→    fn: () => Promise<unknown>,
   307→    success?: string,
   308→    undo?: (data: unknown) => UndoSpec | null | undefined
   309→  ) => {
   310→    setSaving(true);
   311→    try {
   312→      const data = await fn();
   313→      if (success) {
   314→        const spec = undo?.(data);
   315→        if (spec?.module && spec.recordId) {
   316→          toast.success(success, {
   317→            duration: 8000,
   318→            action: { label: "Undo", onClick: () => void undoRequest(spec) },
   319→          });
   320→        } else {
   321→          toast.success(success);
   322→        }
   323→      }
   324→      return { ok: true as const, data };
   325→    } catch (e) {
   326→      toast.error(errMessage(e));
   327→      return { ok: false as const, error: e };
   328→    } finally {
   329→      setSaving(false);
   330→    }
   331→  }, []);
   332→  return { mutate, saving };
   333→}

   335→export function useDebounced<T>(value: T, delay = 300): T {
   336→  const [v, setV] = useState(value);
   337→  useEffect(() => {
   338→    const t = setTimeout(() => setV(value), delay);
   339→    return () => clearTimeout(t);
   340→  }, [value, delay]);
   341→  return v;
   342→}

   344→export function todayStr(offsetDays = 0): string {
   345→  const d = new Date();
   346→  d.setDate(d.getDate() + offsetDays);
   347→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;
   348→}

   350→export function fmtDay(s?: string | null): string {
   351→  if (!s) return "—";
   352→  const d = new Date(s.length === 10 ? `${s}T00:00:00` : s);
   353→  if (Number.isNaN(d.getTime())) return s;
   354→  return d.toLocaleDateString("en-IN", { day: "2-digit", month: "short", year: "numeric" });
   355→}
   356→
   357→export function fmtDateTime(s?: string | null): string {
   358→  if (!s) return "—";
   359→  const d = new Date(s);
   360→  if (Number.isNaN(d.getTime())) return s;
   361→  return d.toLocaleString("en-IN", { day: "2-digit", month: "short", hour: "2-digit", minute: "2-digit" });
   362→}

   367→  return parts.map((p) => p[0]?.toUpperCase() ?? "").join("") || "?";
   368→}

   435→// Sentinel for "no selection" options (Radix Select forbids empty-string values).
   436→export const EMPTY_SENTINEL = "__all__";
   437→
   438→export function SelectInput({ value, onChange, options, placeholder = "Select…", className, disabled }: {
   439→  value: string;
   440→  onChange: (v: string) => void;
   441→  options: Option[];
   442→  placeholder?: string;
   443→  className?: string;
   444→  disabled?: boolean;
   445→}) {
   446→  return (
   447→    <Select
   448→      // Always controlled: `value || sentinel` keeps Radix from flipping
   449→      // between uncontrolled (undefined) and controlled when value is "".
   450→      // A non-matching sentinel still renders the placeholder visually.
   451→      value={value || EMPTY_SENTINEL}
   452→      onValueChange={(v) => onChange(v === EMPTY_SENTINEL ? "" : v)}
   453→      disabled={disabled}
   454→    >
   455→      <SelectTrigger className={cn("h-10 w-full", className)} aria-label={placeholder}>
   456→        <SelectValue placeholder={placeholder} />
   457→      </SelectTrigger>
   458→      <SelectContent position="popper" className="max-h-72">
   459→        {/* Radix forbids empty-string item values — map them to a sentinel. */}
   460→        {options.map((o) => (
   461→          <SelectItem key={o.value || EMPTY_SENTINEL} value={o.value || EMPTY_SENTINEL} className="text-sm">{o.label}</SelectItem>
   462→        ))}
   463→      </SelectContent>
   464→    </Select>
   465→  );
   466→}

   637→export interface PickerEmployee {
   638→  id: string;
   639→  fullName: string;
   640→  code?: string | null;
   641→  standardRate?: number | null;
   642→  advanceBalance?: number | null;

   645→export function EmployeePicker({ employees, value, onChange, multi = true, height = "max-h-60", unavailable }: {
   646→  employees: PickerEmployee[];
   647→  value: string[];
   648→  onChange: (ids: string[]) => void;
   649→  multi?: boolean;
   650→  height?: string;
   651→  /** Return a short reason string to mark an employee as unselectable (e.g. already booked all day), or null when selectable. */
   652→  unavailable?: (e: PickerEmployee) => string | null;
   653→}) {
   654→  const [search, setSearch] = useState("");
   655→  const filtered = useMemo(() => {
   656→    const q = search.trim().toLowerCase();
   657→    if (!q) return employees;
   658→    return employees.filter((e) => `${e.fullName} ${e.code ?? ""}`.toLowerCase().includes(q));
   659→  }, [employees, search]);
   660→
   661→  const toggle = (id: string) => {
   662→    if (unavailable) {
   663→      const emp = employees.find((e) => e.id === id);
   664→      if (emp && unavailable(emp)) return; // blocked — ignore toggle
   665→    }
   666→    if (!multi) { onChange([id]); return; }
   667→    onChange(value.includes(id) ? value.filter((v) => v !== id) : [...value, id]);
   668→  };
   669→
   670→  return (
   671→    <div className="space-y-2">
   672→      <div className="relative">
   673→        <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" aria-hidden />
   674→        <Input
   675→          value={search}
   676→          onChange={(e) => setSearch(e.target.value)}
   677→          placeholder="Search employees…"
   678→          className="h-9 rounded-xl bg-card pl-9"
   679→          aria-label="Search employees"
   680→        />
   681→      </div>
   682→      {multi && value.length > 0 && (
   683→        <div className="flex items-center justify-between text-xs text-muted-foreground">
   684→          <span>{value.length} selected</span>
   685→          <button type="button" className="font-medium text-primary hover:underline" onClick={() => onChange([])}>
   686→            Clear all
   687→          </button>
   688→        </div>
   689→      )}
   690→      <div className={cn("overflow-y-auto rounded-xl border", height)} role="listbox" aria-multiselectable={multi}>
   691→        {filtered.length === 0 && (
   692→          <p className="p-6 text-center text-xs text-muted-foreground">No employees found</p>
   693→        )}
   694→        {filtered.map((e) => {
   695→          const on = value.includes(e.id);
   696→          const blockedReason = unavailable?.(e) ?? null;
   697→          const blocked = Boolean(blockedReason);
   698→          return (
   699→            <div
   700→              key={e.id}
   701→              role="option"
   702→              aria-selected={on}
   703→              aria-disabled={blocked || undefined}
   704→              tabIndex={blocked ? -1 : 0}
   705→              onClick={() => { if (!blocked) toggle(e.id); }}
   706→              onKeyDown={(ev) => { if (!blocked && (ev.key === "Enter" || ev.key === " ")) { ev.preventDefault(); toggle(e.id); } }}
   707→              className={cn(
   708→                "flex items-center gap-2.5 border-b px-3 py-2.5 transition-colors last:border-b-0",
   709→                blocked ? "cursor-not-allowed opacity-60" : "cursor-pointer hover:bg-muted/60",
   710→                on && "bg-primary/5"
   711→              )}
   712→            >
   713→              {multi ? (
   714→                <span className={cn(
   715→                  "flex h-4 w-4 shrink-0 items-center justify-center rounded border transition-colors",
   716→                  on ? "border-primary bg-primary text-primary-foreground" : "border-input bg-background"
   717→                )} aria-hidden>
   718→                  {on && <Check className="h-3 w-3" />}
   719→                </span>
   720→              ) : (
   721→                <span className={cn(
   722→                  "flex h-4 w-4 shrink-0 items-center justify-center rounded-full border transition-colors",
   723→                  on ? "border-primary" : "border-input"
   724→                )} aria-hidden>
   725→                  {on && <span className="h-2 w-2 rounded-full bg-primary" />}
   726→                </span>
   727→              )}
   728→              <div className="min-w-0 flex-1">
   729→                <p className="truncate text-sm font-medium">{e.fullName}</p>
   730→                <p className="text-[11px] text-muted-foreground">
   731→                  {e.code ?? ""}{e.standardRate ? ` · ${formatINR(e.standardRate)}/shift` : ""}
   732→                </p>
   733→              </div>
   734→              {blocked && blockedReason ? (
   735→                <span className="shrink-0 rounded-full bg-amber-100 px-2 py-0.5 text-[10px] font-semibold tabular-nums text-amber-800 dark:bg-amber-950 dark:text-amber-300">
   736→                  {blockedReason}
   737→                </span>
   738→              ) : (e.advanceBalance ?? 0) > 0 && (
   739→                <span className="shrink-0 rounded-full bg-red-50 px-2 py-0.5 text-[10px] font-medium tabular-nums text-red-600 dark:bg-red-950 dark:text-red-400">
   740→                  Adv {formatINR(e.advanceBalance ?? 0, { compact: true })}
   741→                </span>
   742→              )}
   743→            </div>
   744→          );
   745→        })}
   746→      </div>
   747→    </div>
   748→  );
   749→}

  1474→                <ChevronLeft className="h-4 w-4" aria-hidden />Back
  1475→              </Button>
  1476→            )}
  1477→            {step < 3 ? (
  1478→              <Button
  1479→                className="min-h-10 gap-1.5"
  1480→                disabled={!canNext}
  1481→                onClick={() => {
  1482→                  if (step === 2) {
  1483→                    // Pre-pick the still-free half for people already booked today.
  1484→                    setRowState((r) => {
  1485→                      const next = { ...r };
  1486→                      for (const id of selected) {
  1487→                        const current = next[id]?.shift ?? "DAY";
  1488→                        if (clashesFor(id, current).length > 0 && !fullyBooked(id)) {
  1489→                          next[id] = { ...next[id], shift: smartShift(id) };
  1490→                        }
  1491→                      }
  1492→                      return next;
  1493→                    });
  1494→                  }
  1495→                  setStep(step + 1);
  1496→                }}
  1497→              >
  1498→                Next<ArrowRight className="h-4 w-4" aria-hidden />
  1499→              </Button>
  1500→            ) : (
  1501→              <Button
  1502→                className="min-h-10 gap-1.5"
  1503→                onClick={() => submit()}
  1504→                disabled={creating || selected.length === 0 || conflictingSelected.length > 0}
  1505→                title={conflictingSelected.length > 0 ? "Resolve shift conflicts first" : undefined}
  1506→              >
  1507→                <Users className="h-4 w-4" aria-hidden />
  1508→                {creating ? "Deploying…" : `Deploy ${selected.length} employee${selected.length === 1 ? "" : "s"}`}
  1509→              </Button>
  1510→            )}
  1511→          </div>
  1512→        </div>
  1513→      </DialogContent>
  1514→    </Dialog>
  1515→  );
  1516→}
