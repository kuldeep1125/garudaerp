   246→export function errMessage(e: unknown): string {
   247→  if (e instanceof ApiError) return e.message;
   248→  if (e instanceof Error) return e.message;
   249→  return "Something went wrong";
   250→}

   252→/** Fetch-on-mount hook with loading/error/reload. Toasts load failures. */
   253→export function useAsync<T>(loader: () => Promise<T>, deps: React.DependencyList) {
   254→  const [data, setData] = useState<T | null>(null);
   255→  const [loading, setLoading] = useState(true);
   256→  const [error, setError] = useState<string | null>(null);
   257→  const load = useCallback(async () => {
   258→    setLoading(true);
   259→    try {
   260→      const d = await loader();
   261→      setData(d);
   262→      setError(null);
   263→    } catch (e) {
   264→      const msg = errMessage(e);
   265→      setError(msg);
   266→      toast.error(msg);
   267→    } finally {
   268→      setLoading(false);
   269→    }
   270→  }, deps);
   271→  useEffect(() => { void load(); }, [load]);
   272→  return { data, loading, error, reload: load, setData };
   273→}

   275→/** Mutation wrapper: consistent error toasts, optional success toast.
   276→ *  Pass an `undo` resolver to attach an "Undo" action to the success toast —
   277→ *  it reverses the entry server-side (zero mismatch) and calls onUndo. */
   278→export interface UndoSpec {
   279→  module: string; // PAYMENT / ADVANCE / EXPENSE / DEPLOYMENT / TRIP / EMI / MAINTENANCE / ADJUSTMENT
   280→  recordId?: string;
   281→  onUndo?: () => void;
   282→}
   283→
   284→export async function undoRequest(spec: {
   285→  auditLogId?: string;
   286→  module?: string;
   287→  recordId?: string;
   288→  onUndo?: () => void;
   289→}): Promise<boolean> {
   290→  try {
   291→    await api.post("/api/undo", {
   292→      ...(spec.auditLogId ? { auditLogId: spec.auditLogId } : { module: spec.module, recordId: spec.recordId }),
   293→    });
   294→    toast.success("Entry reversed — everything is back to how it was", { icon: "↩️" });
   295→    spec.onUndo?.();
   296→    return true;
   297→  } catch (e) {
   298→    toast.error(errMessage(e));
   299→    return false;
   300→  }
   301→}

   303→export function useMutation() {
   304→  const [saving, setSaving] = useState(false);
   305→  const mutate = useCallback(async (
   306→    fn: () => Promise<unknown>,
   307→    success?: string,
   308→    undo?: (data: unknown) => UndoSpec | null | undefined
   309→  ) => {
   310→    setSaving(true);
   311→    try {
   312→      const data = await fn();
   313→      if (success) {
   314→        const spec = undo?.(data);
   315→        if (spec?.module && spec.recordId) {
   316→          toast.success(success, {
   317→            duration: 8000,
   318→            action: { label: "Undo", onClick: () => void undoRequest(spec) },
   319→          });
   320→        } else {
   321→          toast.success(success);
   322→        }
   323→      }
   324→      return { ok: true as const, data };
   325→    } catch (e) {
   326→      toast.error(errMessage(e));
   327→      return { ok: false as const, error: e };
   328→    } finally {
   329→      setSaving(false);
   330→    }
   331→  }, []);
   332→  return { mutate, saving };
   333→}

   335→export function useDebounced<T>(value: T, delay = 300): T {
   336→  const [v, setV] = useState(value);
   337→  useEffect(() => {
   338→    const t = setTimeout(() => setV(value), delay);
   339→    return () => clearTimeout(t);
   340→  }, [value, delay]);
   341→  return v;
   342→}

   350→export function fmtDay(s?: string | null): string {
   351→  if (!s) return "—";
   352→  const d = new Date(s.length === 10 ? `${s}T00:00:00` : s);
   353→  if (Number.isNaN(d.getTime())) return s;
   354→  return d.toLocaleDateString("en-IN", { day: "2-digit", month: "short", year: "numeric" });
   355→}

   357→export function fmtDateTime(s?: string | null): string {
   358→  if (!s) return "—";
   359→  const d = new Date(s);
   360→  if (Number.isNaN(d.getTime())) return s;
   361→  return d.toLocaleString("en-IN", { day: "2-digit", month: "short", hour: "2-digit", minute: "2-digit" });
   362→}

   439→/** Payments at or above this amount ask for an explicit confirmation strip before saving. */
   440→export const LARGE_AMOUNT_THRESHOLD = 50_000;
   441→
   442→/**
   443→ * Guard rail for large money entries: render-props the dialog footer. When the
   444→ * parsed amount is ≥ LARGE_AMOUNT_THRESHOLD, the first press of the primary
   445→ * button shows an amber confirmation strip instead of submitting; the payment
   446→ * proceeds only via the strip's Confirm button. Under the threshold the
   447→ * confirm handler fires immediately (existing behaviour, unchanged).
   448→ */
   449→export function ConfirmAmount({ amount, subject, onSubmit, children }: {
   450→  amount: number;
   451→  subject: string;
   452→  /** the real save routine — runs only after explicit confirmation for large amounts */
   453→  onSubmit: () => void;
   454→  /** Render-prop footer: `confirm` wires into the primary save button, `edit`
   455→   *  collapses the echo strip, `armed` is true while the strip is showing. */
   456→  children: (h: { confirm: () => void; edit: () => void; armed: boolean }) => React.ReactNode;
   457→}) {
   458→  const [armed, setArmed] = useState(false);
   459→  const show = amount >= LARGE_AMOUNT_THRESHOLD && armed;
   460→
   461→  const confirm = () => {
   462→    if (amount >= LARGE_AMOUNT_THRESHOLD && !armed) {
   463→      setArmed(true);
   464→      return;
   465→    }
   466→    setArmed(false);
   467→    onSubmit();
   468→  };
   469→
   470→  return (
   471→    <>
   472→      {show && (
   473→        <div
   474→          role="alert"
   475→          className="flex flex-wrap items-center gap-2 rounded-xl border border-amber-300 bg-amber-50 px-3 py-2.5 dark:border-amber-900 dark:bg-amber-950/40"
   476→        >
   477→          <Banknote className="h-4 w-4 shrink-0 text-amber-600 dark:text-amber-400" aria-hidden />
   478→          <p className="min-w-0 flex-1 text-xs font-medium leading-snug text-amber-800 dark:text-amber-300">
   479→            You are about to record <span className="font-bold tabular-nums">{formatINR(amount)}</span> for{" "}
   480→            <span className="font-semibold">{subject}</span>.
   481→          </p>
   482→          <span className="flex shrink-0 items-center gap-1.5">
   483→            <Button
   484→              size="sm"
   485→              className="h-8 bg-amber-600 text-white hover:bg-amber-700 focus-visible:ring-amber-600/40"
   486→              onClick={confirm}
   487→            >
   488→              Confirm
   489→            </Button>
   490→            <Button
   491→              size="sm"
   492→              variant="outline"
   493→              className="h-8 border-amber-300 text-amber-800 hover:bg-amber-100 dark:border-amber-900 dark:text-amber-300 dark:hover:bg-amber-950"
   494→              onClick={() => setArmed(false)}
   495→            >
   496→              Edit amount
   497→            </Button>
   498→          </span>
   499→        </div>
   500→      )}
   501→      {children({ confirm, edit: () => setArmed(false), armed: show })}
   502→    </>
   503→  );
   504→}

   507→export const EMPTY_SENTINEL = "__all__";

   509→export function SelectInput({ value, onChange, options, placeholder = "Select…", className, disabled }: {
   510→  value: string;
   511→  onChange: (v: string) => void;
   512→  options: Option[];
   513→  placeholder?: string;
   514→  className?: string;
   515→  disabled?: boolean;
   516→}) {
   517→  return (
   518→    <Select
   519→      // Always controlled: `value || sentinel` keeps Radix from flipping
   520→      // between uncontrolled (undefined) and controlled when value is "".
   521→      // A non-matching sentinel still renders the placeholder visually.
   522→      value={value || EMPTY_SENTINEL}
   523→      onValueChange={(v) => onChange(v === EMPTY_SENTINEL ? "" : v)}
   524→      disabled={disabled}
   525→    >
   526→      <SelectTrigger className={cn("h-10 w-full", className)} aria-label={placeholder}>
   527→        <SelectValue placeholder={placeholder} />
   528→      </SelectTrigger>
   529→      <SelectContent position="popper" className="max-h-72">
   530→        {/* Radix forbids empty-string item values — map them to a sentinel. */}
   531→        {options.map((o) => (
   532→          <SelectItem key={o.value || EMPTY_SENTINEL} value={o.value || EMPTY_SENTINEL} className="text-sm">{o.label}</SelectItem>
   533→        ))}
   534→      </SelectContent>
   535→    </Select>
   536→  );
   537→}

   708→export interface PickerEmployee {
   709→  id: string;
   710→  fullName: string;

   712→  standardRate?: number | null;
   713→  advanceBalance?: number | null;

   716→export function EmployeePicker({ employees, value, onChange, multi = true, height = "max-h-60", unavailable }: {
   717→  employees: PickerEmployee[];
   718→  value: string[];
   719→  onChange: (ids: string[]) => void;
   720→  multi?: boolean;
   721→  height?: string;
   722→  /** Return a short reason string to mark an employee as unselectable (e.g. already booked all day), or null when selectable. */
   723→  unavailable?: (e: PickerEmployee) => string | null;
   724→}) {
   725→  const [search, setSearch] = useState("");
   726→  const filtered = useMemo(() => {
   727→    const q = search.trim().toLowerCase();
   728→    if (!q) return employees;
   729→    return employees.filter((e) => `${e.fullName} ${e.code ?? ""}`.toLowerCase().includes(q));
   730→  }, [employees, search]);
   731→
   732→  const toggle = (id: string) => {
   733→    if (unavailable) {
   734→      const emp = employees.find((e) => e.id === id);
   735→      if (emp && unavailable(emp)) return; // blocked — ignore toggle
   736→    }
   737→    if (!multi) { onChange([id]); return; }
   738→    onChange(value.includes(id) ? value.filter((v) => v !== id) : [...value, id]);
   739→  };
   740→
   741→  return (
   742→    <div className="space-y-2">
   743→      <div className="relative">
   744→        <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" aria-hidden />
   745→        <Input
   746→          value={search}
   747→          onChange={(e) => setSearch(e.target.value)}
   748→          placeholder="Search employees…"
   749→          className="h-9 rounded-xl bg-card pl-9"
   750→          aria-label="Search employees"
   751→        />
   752→      </div>
   753→      {multi && value.length > 0 && (
   754→        <div className="flex items-center justify-between text-xs text-muted-foreground">
   755→          <span>{value.length} selected</span>
   756→          <button type="button" className="font-medium text-primary hover:underline" onClick={() => onChange([])}>
   757→            Clear all
   758→          </button>
   759→        </div>
   760→      )}
   761→      <div className={cn("overflow-y-auto rounded-xl border", height)} role="listbox" aria-multiselectable={multi}>
   762→        {filtered.length === 0 && (
   763→          <p className="p-6 text-center text-xs text-muted-foreground">No employees found</p>
   764→        )}
   765→        {filtered.map((e) => {
   766→          const on = value.includes(e.id);
   767→          const blockedReason = unavailable?.(e) ?? null;
   768→          const blocked = Boolean(blockedReason);
   769→          return (
   770→            <div
   771→              key={e.id}
   772→              role="option"
   773→              aria-selected={on}
   774→              aria-disabled={blocked || undefined}
   775→              tabIndex={blocked ? -1 : 0}
   776→              onClick={() => { if (!blocked) toggle(e.id); }}
   777→              onKeyDown={(ev) => { if (!blocked && (ev.key === "Enter" || ev.key === " ")) { ev.preventDefault(); toggle(e.id); } }}
   778→              className={cn(
   779→                "flex items-center gap-2.5 border-b px-3 py-2.5 transition-colors last:border-b-0",
   780→                blocked ? "cursor-not-allowed opacity-60" : "cursor-pointer hover:bg-muted/60",
   781→                on && "bg-primary/5"
   782→              )}
   783→            >
   784→              {multi ? (
   785→                <span className={cn(
   786→                  "flex h-4 w-4 shrink-0 items-center justify-center rounded border transition-colors",
   787→                  on ? "border-primary bg-primary text-primary-foreground" : "border-input bg-background"
   788→                )} aria-hidden>
   789→                  {on && <Check className="h-3 w-3" />}
   790→                </span>
   791→              ) : (
   792→                <span className={cn(
   793→                  "flex h-4 w-4 shrink-0 items-center justify-center rounded-full border transition-colors",
   794→                  on ? "border-primary" : "border-input"
   795→                )} aria-hidden>
   796→                  {on && <span className="h-2 w-2 rounded-full bg-primary" />}
   797→                </span>
   798→              )}
   799→              <div className="min-w-0 flex-1">
   800→                <p className="truncate text-sm font-medium">{e.fullName}</p>
   801→                <p className="text-[11px] text-muted-foreground">
   802→                  {e.code ?? ""}{e.standardRate ? ` · ${formatINR(e.standardRate)}/shift` : ""}
   803→                </p>
   804→              </div>
   805→              {blocked && blockedReason ? (
   806→                <span className="shrink-0 rounded-full bg-amber-100 px-2 py-0.5 text-[10px] font-semibold tabular-nums text-amber-800 dark:bg-amber-950 dark:text-amber-300">
   807→                  {blockedReason}
   808→                </span>
   809→              ) : (e.advanceBalance ?? 0) > 0 && (
   810→                <span className="shrink-0 rounded-full bg-red-50 px-2 py-0.5 text-[10px] font-medium tabular-nums text-red-600 dark:bg-red-950 dark:text-red-400">
   811→                  Adv {formatINR(e.advanceBalance ?? 0, { compact: true })}
   812→                </span>
   813→              )}
   814→            </div>
   815→          );
   816→        })}
   817→      </div>
   818→    </div>
   819→  );
   820→}

   823→export function useActiveEmployees(enabled: boolean) {
   824→  const [employees, setEmployees] = useState<PickerEmployee[]>([]);
   825→  const [loaded, setLoaded] = useState(false);
   826→  useEffect(() => {

   832→    return () => { cancelled = true; };
   833→  }, [enabled, loaded]);

  1600→          </div>
  1601→        </div>

  1603→    </Dialog>
