    41→// Media-query rail detection for md–lg widths (tablets / narrow laptops):
    42→// the sidebar becomes an icons-only rail there, while <md hides it entirely
    43→// and lg+ shows the full collapsible sidebar. useSyncExternalStore keeps this
    44→// SSR-safe (server snapshot = false, so hydration never mismatches).
    45→const RAIL_QUERY = "(min-width: 768px) and (max-width: 1023.98px)";
    46→function useIsRailWidth(): boolean {
    47→  return useSyncExternalStore(
    48→    (onStoreChange) => {
    49→      const mq = window.matchMedia(RAIL_QUERY);
    50→      mq.addEventListener("change", onStoreChange);
    51→      return () => mq.removeEventListener("change", onStoreChange);
    52→    },
    53→    () => window.matchMedia(RAIL_QUERY).matches,
    54→    () => false
    55→  );
    56→}

   175→// Sticky top header
   176→function TopBar({ onOpenMore, onOpenPalette, onOpenShortcuts }: { onOpenMore: () => void; onOpenPalette: () => void; onOpenShortcuts: () => void }) {
   177→  const { owner, logout } = useAuth();
   178→  const { navigate } = useNav();
   179→  const { lang, setLang } = useLang();
   180→  const [notifCount, setNotifCount] = useState(0);
   181→  const [pulse, setPulse] = useState(false);
   182→  const [undoOpen, setUndoOpen] = useState(false);
   183→  const [notifRefresh, setNotifRefresh] = useState(0);
   184→  const { canInstall, install } = usePwaInstall();
   185→  const prevKeysRef = useRef<Set<string> | null>(null);
   186→  const firstLoadRef = useRef(true);
   187→  const pulseTimerRef = useRef<number | null>(null);
   188→
   189→  useEffect(() => {
   190→    let alive = true;
   191→    const load = async () => {
   192→      try {
   193→        const items = await import("@/lib/api-client").then(({ api }) =>
   194→          api.get<AppNotification[]>("/api/notifications")
   195→        );
   196→        if (!alive || !Array.isArray(items)) return;
   197→        // Respect muted categories (localStorage) — the bell only counts alerts
   198→        // the owner hasn't muted from the notifications view.
   199→        const muted = readMutedGroups();
   200→        const visible = muted.length > 0 ? items.filter((n) => !muted.includes(n.severity)) : items;
   201→        setNotifCount(visible.length);
   202→        const prevKeys = prevKeysRef.current;
   203→        prevKeysRef.current = new Set(items.map((n) => n.key));
   204→        // Announce only genuine arrivals (skip initial load), when the tab is
   205→        // visible and the alert isn't muted.
   206→        if (!firstLoadRef.current && prevKeys && document.visibilityState === "visible") {
   207→          const fresh = visible.filter((n) => !prevKeys.has(n.key));
   208→          if (fresh.length > 0) {
   209→            const rank = { CRITICAL: 0, WARNING: 1, INFO: 2 } as const;
   210→            const headline =
   211→              [...fresh].sort((a, b) => rank[a.severity] - rank[b.severity])[0] ?? fresh[0];
   212→            toast(headline.title, {
   213→              description:
   214→                fresh.length > 1
   215→                  ? `${headline.message} — and ${fresh.length - 1} more new alert${fresh.length === 2 ? "" : "s"}.`
   216→                  : headline.message,
   217→              icon: SEVERITY_ICON[headline.severity] ?? SEVERITY_ICON.INFO,
   218→              action: { label: "View", onClick: () => navigate("notifications") },
   219→            });
   220→            // Bell ripple — brief ping so the eye is drawn to the new alert.
   221→            setPulse(true);
   222→            if (pulseTimerRef.current) window.clearTimeout(pulseTimerRef.current);
   223→            pulseTimerRef.current = window.setTimeout(() => setPulse(false), 2400);
   224→          }
   225→        }
   226→        firstLoadRef.current = false;
   227→      } catch { /* ignore */ }
   228→    };
   229→    load();
   230→    const t = setInterval(load, 60_000);
   231→    // Re-check immediately when the user returns to the tab.
   232→    const onVis = () => { if (document.visibilityState === "visible") void load(); };
   233→    // Mute toggles + dismissals (this tab via custom event, other tabs via
   234→    // storage) re-count instantly.
   235→    const onMuteChange = () => { void load(); };
   236→    document.addEventListener("visibilitychange", onVis);
   237→    window.addEventListener(NOTIF_MUTED_EVENT, onMuteChange);
   238→    window.addEventListener(NOTIF_CHANGED_EVENT, onMuteChange);
   239→    window.addEventListener("storage", onMuteChange);
   240→    return () => {
   241→      alive = false; clearInterval(t);
   242→      document.removeEventListener("visibilitychange", onVis);
   243→      window.removeEventListener(NOTIF_MUTED_EVENT, onMuteChange);
   244→      window.removeEventListener(NOTIF_CHANGED_EVENT, onMuteChange);
   245→      window.removeEventListener("storage", onMuteChange);
   246→      if (pulseTimerRef.current) window.clearTimeout(pulseTimerRef.current);
   247→    };
   248→  }, [navigate, notifRefresh]);
   249→
   250→  const initials = (owner?.name ?? "?").slice(0, 2).toUpperCase();
   251→
   252→  return (
   253→    <>
   254→    <header className="sticky top-0 z-40 border-b bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/80">
   255→      <div className="mx-auto flex h-14 max-w-[1600px] items-center gap-2 px-3 sm:px-5">
   256→        <button className="flex items-center gap-2 md:hidden" onClick={() => navigate("dashboard")} aria-label="BizHub home">
   257→          <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-emerald-600 text-white font-bold">
   258→            <span className="text-xs">B</span>
   259→          </div>
   260→          <span className="text-sm font-bold">BizHub</span>
   261→        </button>
   262→
   263→        <Button variant="ghost" size="icon" className="hidden lg:inline-flex h-9 w-9 lg:hidden" aria-hidden tabIndex={-1} />
   264→
   265→        {/* Global search trigger — opens the command palette (Ctrl K) */}
   266→        <button
   267→          onClick={onOpenPalette}
   268→          className="ml-0 hidden sm:flex h-9 min-w-0 flex-1 max-w-md items-center gap-2 rounded-xl border bg-muted/40 px-3 text-sm text-muted-foreground hover:bg-muted hover:border-primary/40 transition-colors"
   269→          aria-label="Open command palette"
   270→        >
   271→          <Search className="h-4 w-4 shrink-0" />
   272→          <span className="min-w-0 truncate whitespace-nowrap hidden md:inline">{t(lang, "topbar.search")}</span>
   273→          <span className="md:hidden whitespace-nowrap">{t(lang, "topbar.searchShort")}</span>
   274→          <kbd className="ml-auto hidden md:inline-flex shrink-0 rounded border bg-background px-1.5 font-mono text-[10px]">Ctrl K</kbd>
   275→        </button>
   276→
   277→        <div className="ml-auto flex items-center gap-0.5 sm:gap-1">
   278→          <Button variant="ghost" size="icon" className="sm:hidden h-10 w-10" onClick={onOpenPalette} aria-label="Open command palette">
   279→            <Search className="h-5 w-5" />
   280→          </Button>
   281→          <Button
   282→            variant="ghost"
   283→            size="icon"
   284→            className="h-10 w-10"
   285→            onClick={() => setUndoOpen(true)}
   286→            aria-label="Undo center"
   287→            title="Undo center"
   288→          >
   289→            <History className="h-5 w-5" />
   290→          </Button>
   291→          <Button variant="ghost" size="icon" className="h-10 w-10 relative" onClick={() => navigate("notifications")} aria-label={`Notifications${notifCount ? `, ${notifCount} active` : ""}`}>
   292→            {pulse && <span className="absolute inset-0 rounded-full bg-red-500/30 animate-ping" aria-hidden />}
   293→            <Bell className={cn("h-5 w-5 transition-transform", pulse && "scale-110")} />
   294→            {notifCount > 0 && (
   295→              <span
   296→                className="absolute right-0.5 top-0.5 flex h-4 min-w-4 items-center justify-center rounded-full bg-red-500 px-1 text-[9px] font-bold tabular-nums text-white ring-2 ring-background"
   297→                aria-hidden
   298→              >
   299→                {notifCount > 9 ? "9+" : notifCount}
   300→              </span>
   301→            )}
   302→          </Button>
   303→          <ThemeToggle />
   304→          <DropdownMenu>
   305→            <DropdownMenuTrigger asChild>
   306→              <Button variant="ghost" size="icon" className="h-10 w-10" aria-label={t(lang, "topbar.language")} title={t(lang, "topbar.language")}>
   307→                <Languages className="h-5 w-5" />
   308→              </Button>
   309→            </DropdownMenuTrigger>
   310→            <DropdownMenuContent align="end" className="w-40">
   311→              <DropdownMenuLabel>{t(lang, "topbar.language")}</DropdownMenuLabel>
   312→              <DropdownMenuSeparator />
   313→              <DropdownMenuItem onClick={() => setLang("en")} aria-checked={lang === "en"} role="menuitemradio">
   314→                {lang === "en" ? <Check className="mr-2 h-4 w-4 text-primary" /> : <span className="mr-2 inline-block h-4 w-4" aria-hidden />}
   315→                {t(lang, "lang.en")}
   316→              </DropdownMenuItem>
   317→              <DropdownMenuItem onClick={() => setLang("hi")} aria-checked={lang === "hi"} role="menuitemradio">
   318→                {lang === "hi" ? <Check className="mr-2 h-4 w-4 text-primary" /> : <span className="mr-2 inline-block h-4 w-4" aria-hidden />}
   319→                {t(lang, "lang.hi")}
   320→              </DropdownMenuItem>
   321→            </DropdownMenuContent>
   322→          </DropdownMenu>
   323→          <DropdownMenu>
   324→            <DropdownMenuTrigger asChild>
   325→              <button className="ml-0.5 flex items-center gap-2 rounded-full outline-none focus-visible:ring-2 focus-visible:ring-ring" aria-label="Owner menu">
   326→                <Avatar className="h-9 w-9 border">
   327→                  <AvatarFallback className="bg-emerald-100 text-emerald-800 text-xs font-bold dark:bg-emerald-900 dark:text-emerald-200">{initials}</AvatarFallback>
   328→                </Avatar>
   329→              </button>
   330→            </DropdownMenuTrigger>
   331→            <DropdownMenuContent align="end" className="w-52">
   332→              <DropdownMenuLabel>
   333→                <p className="text-sm font-semibold">{owner?.name}</p>

   480→export function AppShell({ children }: { children: React.ReactNode }) {
   481→  const { navigate, view } = useNav();
   482→  const [collapsed, setCollapsed] = useState(false);
   483→  // md–lg widths force the icons-only rail (no toggle there); lg+ keeps the
   484→  // full sidebar with the manual collapse button.
   485→  const railMode = useIsRailWidth();
   486→  const railActive = railMode || collapsed;
   487→  const [moreOpen, setMoreOpen] = useState(false);
   488→  const [paletteOpen, setPaletteOpen] = useState(false);
   489→  const [shortcutsOpen, setShortcutsOpen] = useState(false);
   490→
   491→  useEffect(() => {
   492→    // "G then key" chord: pendingG holds the timeout id while waiting for the
   493→    // second key (1s window). Cleared by typing anywhere else.
   494→    let pendingG: number | null = null;
   495→    const clearPendingG = () => {
   496→      if (pendingG !== null) window.clearTimeout(pendingG);
   497→      pendingG = null;
   498→    };
   499→    const GO_CHORDS: Record<string, string> = {
   500→      d: "dashboard", e: "employees", p: "properties", t: "trips",
   501→    };
   502→    const isTypingTarget = (el: EventTarget | null): boolean => {
   503→      const t = el as HTMLElement | null;
   504→      return Boolean(
   505→        t &&
   506→        (t.tagName === "INPUT" || t.tagName === "TEXTAREA" || t.tagName === "SELECT" || t.isContentEditable)
   507→      );
   508→    };
   509→    const onKey = (e: KeyboardEvent) => {
   510→      if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === "k") {
   511→        e.preventDefault();
   512→        clearPendingG();
   513→        setPaletteOpen((o) => !o);
   514→        return;
   515→      }
   516→      // Palette is open → its input owns the keyboard; ignore global chords.
   517→      if (paletteOpen) return;
   518→      if (e.altKey || e.ctrlKey || e.metaKey) return;
   519→      // Typing in a field always cancels any pending chord.
   520→      if (isTypingTarget(e.target)) {
   521→        clearPendingG();
   522→        return;
   523→      }
   524→      if (pendingG !== null) {
   525→        const target = GO_CHORDS[e.key.toLowerCase()];
   526→        clearPendingG();
   527→        if (target) {
   528→          e.preventDefault();
   529→          navigate(target);
   530→        }
   531→        return;
   532→      }
   533→      if (e.key === "g" || e.key === "G") {
   534→        pendingG = window.setTimeout(clearPendingG, 1000);
   535→        return;
   536→      }
   537→      if (e.key === "?") {
   538→        e.preventDefault();
   539→        setShortcutsOpen(true);
   540→      }
   541→    };
   542→    window.addEventListener("keydown", onKey);
   543→    return () => {
   544→      window.removeEventListener("keydown", onKey);
   545→      clearPendingG();
   546→    };
   547→  }, [navigate, paletteOpen]);
   548→
   549→  const currentView = getView(view);
   550→  const isWide = ["dashboard", "manpower", "transport"].includes(view);
   551→
   552→  return (
   553→    <div className="flex min-h-screen bg-muted/30">
   554→      {/* Desktop sidebar — icons-only rail at md–lg, full (collapsible) at lg+ */}
   555→      <aside className={cn("sticky top-0 hidden h-screen shrink-0 border-r bg-background transition-all md:block", railActive ? "w-[68px]" : "w-60")}>
   556→        <SideNav collapsed={railActive} onToggle={railMode ? undefined : () => setCollapsed((c) => !c)} />
   557→      </aside>
   558→
   559→      <div className="flex min-w-0 flex-1 flex-col">
   560→        <TopBar onOpenMore={() => setMoreOpen(true)} onOpenPalette={() => setPaletteOpen(true)} onOpenShortcuts={() => setShortcutsOpen(true)} />
   561→        <BusinessBanner />
   562→        <main
   563→          id="main-scroll"
   564→          className={cn("mx-auto w-full max-w-[1600px] flex-1 px-3 pb-24 pt-3 sm:px-5 sm:pt-4 md:pb-10", !isWide && "max-w-[1400px]")}
   565→          aria-label={currentView?.label ?? "Content"}
   566→        >
   567→          {/* Keyed by view → remounts on navigation and plays the enter animation */}
   568→          <div key={view} className="view-enter">
   569→            {children}
   570→          </div>
   571→        </main>
   572→        <footer className="mt-auto hidden border-t bg-background py-3 md:block">
   573→          <div className="mx-auto flex max-w-[1600px] items-center justify-between px-5 text-[11px] text-muted-foreground">
   574→            <p>BizHub — one control center for Manpower & Transport businesses.</p>
   575→            <p className="tabular-nums">{new Date().getFullYear()} · All financial actions are audited.</p>
   576→          </div>
   577→        </footer>
   578→      </div>
   579→
   580→      {/* Mobile bottom nav */}
   581→      <BottomNav onOpenMore={() => setMoreOpen(true)} />
   582→
   583→      {/* Global command palette (Ctrl/Cmd+K, search buttons) */}
   584→      <CommandPalette open={paletteOpen} onOpenChange={setPaletteOpen} />
   585→
   586→      {/* Keyboard shortcuts overlay (? key or owner menu) */}
   587→      <ShortcutsDialog open={shortcutsOpen} onOpenChange={setShortcutsOpen} />
   588→
   589→      {/* Mobile "More" sheet with full nav */}
   590→      <Sheet open={moreOpen} onOpenChange={setMoreOpen}>
   591→        <SheetTrigger className="hidden" aria-hidden />
   592→        <SheetContent side="left" className="w-[280px] p-0">
   593→          <SheetHeader className="sr-only">
   594→            <SheetTitle>All navigation</SheetTitle>
   595→          </SheetHeader>
   596→          <SideNav onNavigate={() => setMoreOpen(false)} />
   597→        </SheetContent>
   598→      </Sheet>
   599→    </div>
   600→  );
   601→}
