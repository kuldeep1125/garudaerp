    67→    67→function monthLabel(month: string): string {
    68→    68→  const [y, m] = month.split("-").map(Number);
    69→    69→  return new Date(y, (m ?? 1) - 1, 1).toLocaleDateString("en-IN", { month: "long", year: "numeric" });
    70→    70→}

    72→    72→function shiftMonth(month: string, delta: number): string {
    73→    73→  const [y, m] = month.split("-").map(Number);
    74→    74→  const d = new Date(y, (m ?? 1) - 1 + delta, 1);
    75→    75→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`;
    76→    76→}

    78→    78→function isWeekend(dateStr: string): boolean {
    79→    79→  const [y, m, d] = dateStr.split("-").map(Number);
    80→    80→  const wd = new Date(y, (m ?? 1) - 1, d ?? 1).getDay();
    81→    81→  return wd === 0 || wd === 6;
    82→    82→}
    83→    83→
    84→    84→function monthBounds(offset = 0): { from: string; to: string } {
    85→    85→  const now = new Date();
    86→    86→  const first = new Date(now.getFullYear(), now.getMonth() + offset, 1);
    87→    87→  const last = new Date(now.getFullYear(), now.getMonth() + offset + 1, 0);
    88→    88→  const f = (d: Date) => `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;
    89→    89→  return { from: f(first), to: f(last) };
    90→    90→}
