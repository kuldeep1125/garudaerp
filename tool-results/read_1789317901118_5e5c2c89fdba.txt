    50→/**
    51→ * Resolves who the money is attributed to for an expense.
    52→ * - `isCommon: true` (or spentById === "COMMON") → shared by all owners & the business;
    53→ *   `spentBy` still records who physically took/spent the money (the acting user).
    54→ * - `spentById` of a real owner → attributed to that owner.
    55→ * - otherwise → attributed to the acting user.
    56→ */
    57→export async function resolveExpenseAttribution(
    58→  body: { isCommon?: unknown; spentById?: unknown },
    59→  actingOwner: { id: string; name: string },
    60→): Promise<ExpenseAttribution> {
    61→  if (body.isCommon === true || String(body.spentById ?? "") === "COMMON") {
    62→    return { isCommon: true, spentById: actingOwner.id, spentByName: actingOwner.name };
    63→  }
    64→  const wanted = body.spentById ? String(body.spentById) : "";
    65→  if (wanted && wanted !== actingOwner.id) {
    66→    const o = await db.owner.findUnique({ where: { id: wanted } });
    67→    if (!o || !o.isActive) throw new HttpError(404, "Selected owner not found");
    68→    return { isCommon: false, spentById: o.id, spentByName: o.name };
    69→  }
    70→  return { isCommon: false, spentById: actingOwner.id, spentByName: actingOwner.name };
    71→}

    98→/** Range for reports: explicit from/to, else month param, else current month. */
    99→export function reportRange(sp: URLSearchParams): { from: Date; to: Date } {
   100→  const from = sp.get("from");
   101→  const to = sp.get("to");
   102→  if (from && to) return { from: parseDate(from), to: endOfDay(parseDate(to)) };
   103→  const month = sp.get("month");
   104→  if (month) return monthBounds(month);
   105→  const now = new Date();
   106→  return { from: new Date(now.getFullYear(), now.getMonth(), 1), to: endOfDay(now) };
   107→}

   123→export interface PropertyLedger {
   124→  billed: number;
   125→  received: number;
   126→  outstanding: number;
   127→  oldestUnpaidDate: string | null;
   128→  unpaidCount: number;
   129→  days: LedgerDay[];
   130→  monthly: { month: string; billed: number; received: number }[];
   131→}
   132→
   133→type LedgerDepRow = { date: Date; billingAmount: number };
   134→type LedgerPayRow = { date: Date; amount: number };
   135→
   136→/**
   137→ * FIFO walk: billable deployments sorted by date consume the payment pool
   138→ * chronologically. Produces per-day rows + totals + monthly aggregates.
   139→ */
   140→export function ledgerFromRows(deployments: LedgerDepRow[], payments: LedgerPayRow[]): PropertyLedger {
   141→  const billed = round2(deployments.reduce((s, d) => s + d.billingAmount, 0));
   142→  const received = round2(payments.reduce((s, p) => s + p.amount, 0));
   143→
   144→  const byDay = new Map<string, { billed: number; paid: number; sort: number }>();
   145→  const monthlyMap = new Map<string, { billed: number; received: number }>();
   146→  const sorted = [...deployments].sort((a, b) => a.date.getTime() - b.date.getTime());
   147→  let pool = received;
   148→  for (const dep of sorted) {
   149→    const dk = dayKey(dep.date);
   150→    const row = byDay.get(dk) ?? { billed: 0, paid: 0, sort: dep.date.getTime() };
   151→    const alloc = Math.max(0, Math.min(dep.billingAmount, pool));
   152→    pool = round2(pool - alloc);
   153→    row.billed = round2(row.billed + dep.billingAmount);
   154→    row.paid = round2(row.paid + alloc);
   155→    byDay.set(dk, row);
   156→    const mk = monthKey(dep.date);
   157→    const mrow = monthlyMap.get(mk) ?? { billed: 0, received: 0 };
   158→    mrow.billed = round2(mrow.billed + dep.billingAmount);
   159→    monthlyMap.set(mk, mrow);
   160→  }
   161→  for (const p of payments) {
   162→    const mk = monthKey(p.date);
   163→    const mrow = monthlyMap.get(mk) ?? { billed: 0, received: 0 };
   164→    mrow.received = round2(mrow.received + p.amount);
   165→    monthlyMap.set(mk, mrow);
   166→  }
   167→
   168→  let running = 0;
   169→  let oldestUnpaidDate: string | null = null;
   170→  let unpaidCount = 0;
   171→  const days: LedgerDay[] = [...byDay.entries()]
   172→    .sort((a, b) => a[1].sort - b[1].sort)
   173→    .map(([date, row]) => {
   174→      running = round2(running + row.billed - row.paid);
   175→      const status: LedgerDay["status"] =
   176→        row.billed - row.paid <= EPS ? "PAID" : row.paid > EPS ? "PARTIAL" : "UNPAID";
   177→      if (status !== "PAID") {
   178→        unpaidCount++;
   179→        if (!oldestUnpaidDate) oldestUnpaidDate = date;
   180→      }
   181→      return { date, billed: row.billed, paid: row.paid, outstanding: running, status };
   182→    });
   183→
   184→  const monthly = [...monthlyMap.entries()]
   185→    .sort(([a], [b]) => a.localeCompare(b))
   186→    .map(([month, r]) => ({ month, billed: r.billed, received: r.received }));
   187→
   188→  return { billed, received, outstanding: round2(billed - received), oldestUnpaidDate, unpaidCount, days, monthly };
   189→}

   191→/**
   192→ * Computes FIFO ledgers for ALL properties in three bulk queries (no N+1).
   193→ * Returns property rows plus a ledger per property id.
   194→ */
   195→export async function loadAllLedgers(): Promise<{
   196→  properties: { id: string; name: string }[];
   197→  map: Map<string, PropertyLedger>;
   198→}> {
   199→  const [properties, deps, pays] = await Promise.all([
   200→    db.property.findMany({ select: { id: true, name: true }, orderBy: { name: "asc" } }),
   201→    db.deployment.findMany({
   202→      select: { propertyId: true, date: true, billingAmount: true },
   203→      orderBy: { date: "asc" },
   204→    }),
   205→    db.propertyPayment.findMany({
   206→      select: { propertyId: true, date: true, amount: true },
   207→      orderBy: [{ date: "asc" }, { createdAt: "asc" }],
   208→    }),
   209→  ]);
   210→  const depBy = new Map<string, LedgerDepRow[]>();
   211→  for (const d of deps) {
   212→    const arr = depBy.get(d.propertyId) ?? [];
   213→    arr.push({ date: d.date, billingAmount: d.billingAmount });
   214→    depBy.set(d.propertyId, arr);
   215→  }
   216→  const payBy = new Map<string, LedgerPayRow[]>();
   217→  for (const p of pays) {
   218→    const arr = payBy.get(p.propertyId) ?? [];
   219→    arr.push({ date: p.date, amount: p.amount });
   220→    payBy.set(p.propertyId, arr);
   221→  }
   222→  const map = new Map<string, PropertyLedger>();
   223→  for (const p of properties) {
   224→    map.set(p.id, ledgerFromRows(depBy.get(p.id) ?? [], payBy.get(p.id) ?? []));
   225→  }
   226→  return { properties, map };
   227→}

   245→/**
   246→ * Recomputes per-deployment paidAmount for a property via the FIFO pool.
   247→ * Call inside a transaction after payments/deployments/amounts change.
   248→ */
   249→export async function recomputeDeploymentPaid(tx: Tx, propertyId: string): Promise<void> {
   250→  const [deps, pays] = await Promise.all([
   251→    tx.deployment.findMany({
   252→      where: { propertyId },
   253→      select: { id: true, paidAmount: true, date: true, billingAmount: true },
   254→      orderBy: [{ date: "asc" }, { createdAt: "asc" }],
   255→    }),
   256→    tx.propertyPayment.findMany({
   257→      where: { propertyId },
   258→      select: { amount: true },
   259→      orderBy: [{ date: "asc" }, { createdAt: "asc" }],
   260→    }),
   261→  ]);
   262→  let pool = round2(pays.reduce((s, p) => s + p.amount, 0));
   263→  for (const dep of deps) {
   264→    const alloc = Math.max(0, Math.min(dep.billingAmount, pool));
   265→    pool = round2(pool - alloc);
   266→    if (Math.abs(dep.paidAmount - alloc) > EPS) {
   267→      await tx.deployment.update({ where: { id: dep.id }, data: { paidAmount: alloc } });
   268→    }
   269→  }
   270→}

   272→export function derivePaidStatus(billingAmount: number, paidAmount: number): "UNPAID" | "PARTIAL" | "PAID" {

   304→export interface AdvanceBalance {
   305→  totalTaken: number;
   306→  totalDeducted: number;
   307→  balance: number;
   308→}
   309→
   310→/** Σ advances − Σ settlements.advanceDeducted (status != CANCELLED) per employee. */
   311→export async function advanceBalanceMap(employeeIds?: string[]): Promise<Map<string, AdvanceBalance>> {
   312→  const empFilter = employeeIds ? { employeeId: { in: employeeIds } } : {};
   313→  const [adv, set] = await Promise.all([
   314→    db.advance.groupBy({ by: ["employeeId"], _sum: { amount: true }, where: empFilter }),
   315→    db.settlement.groupBy({
   316→      by: ["employeeId"],
   317→      _sum: { advanceDeducted: true },
   318→      where: { status: { not: "CANCELLED" }, ...empFilter },
   319→    }),
   320→  ]);
   321→  const map = new Map<string, AdvanceBalance>();
   322→  for (const row of adv) {
   323→    map.set(row.employeeId, { totalTaken: round2(row._sum.amount ?? 0), totalDeducted: 0, balance: round2(row._sum.amount ?? 0) });
   324→  }
   325→  for (const row of set) {
   326→    const cur = map.get(row.employeeId) ?? { totalTaken: 0, totalDeducted: 0, balance: 0 };
   327→    cur.totalDeducted = round2(row._sum.advanceDeducted ?? 0);
   328→    cur.balance = round2(cur.totalTaken - cur.totalDeducted);
   329→    map.set(row.employeeId, cur);
   330→  }
   331→  return map;
   332→}

   379→/** Revenue = trip paidAmount (by startAt). Expenses exclude EMI; EMI tracked separately. */
   380→export async function vehicleStatsMap(vehicleIds: string[]): Promise<Map<string, VehicleStat>> {
   381→  const map = new Map<string, VehicleStat>();
   382→  if (!vehicleIds.length) return map;
   383→  for (const id of vehicleIds) map.set(id, emptyVehicleStat());
   384→  const now = new Date();
   385→  const mFrom = new Date(now.getFullYear(), now.getMonth(), 1);
   386→  const mTo = endOfDay(now);
   387→  const [trips, expenses] = await Promise.all([
   388→    db.trip.findMany({
   389→      where: { vehicleId: { in: vehicleIds }, status: { not: "CANCELLED" } },
   390→      select: { vehicleId: true, startAt: true, paidAmount: true },
   391→    }),
   392→    db.expense.findMany({
   393→      where: { vehicleId: { in: vehicleIds } },
   394→      select: { vehicleId: true, date: true, amount: true, categoryName: true, kind: true },
   395→    }),
   396→  ]);
   397→  for (const t of trips) {
   398→    const s = map.get(t.vehicleId);
   399→    if (!s) continue;
   400→    s.revenue = round2(s.revenue + t.paidAmount);
   401→    if (t.startAt >= mFrom && t.startAt <= mTo) s.monthRevenue = round2(s.monthRevenue + t.paidAmount);
   402→  }
   403→  for (const e of expenses) {
   404→    if (!e.vehicleId) continue; // nullable FK; filtered by query but guard anyway
   405→    const s = map.get(e.vehicleId);
   406→    if (!s) continue;
   407→    if (e.kind === "CAPITAL") continue; // owner capital movements are never vehicle opex (defensive)
   408→    const isEmi = (e.categoryName ?? "").toUpperCase() === "EMI";
   409→    if (!isEmi) s.expense = round2(s.expense + e.amount);
   410→    else s.emi = round2(s.emi + e.amount);
   411→    if (e.date >= mFrom && e.date <= mTo) {
   412→      if (!isEmi) s.monthExpense = round2(s.monthExpense + e.amount);
   413→      else s.monthEmi = round2(s.monthEmi + e.amount);
   414→    }
   415→  }
   416→  for (const s of map.values()) {
   417→    s.monthNet = round2(s.monthRevenue - s.monthExpense - s.monthEmi);
   418→    s.net = round2(s.revenue - s.expense - s.emi);
   419→  }
   420→  return map;
   421→}

   431→/** Finds (or lazily creates) the given category within a business, falling back to COMMON. */
   432→export async function ensureCategory(name: string, business: string): Promise<{ id: string | null; name: string }> {
   433→  const found = await db.expenseCategory.findFirst({ where: { name, business } });

   441→    return { id: null, name };

   458→/** Live-computed notifications, filtered by dismissals, severity-sorted. */
   459→export async function computeNotifications(): Promise<AppNotification[]> {
   460→  const now = new Date();
   461→  const today = startOfDay(now);
   462→  const in7 = endOfDay(new Date(today.getFullYear(), today.getMonth(), today.getDate() + 7));
   463→  const in30 = endOfDay(new Date(today.getFullYear(), today.getMonth(), today.getDate() + 30));
   464→  const list: AppNotification[] = [];
   465→
   466→  // 1) Property outstanding balances
   467→  const { properties, map: ledgers } = await loadAllLedgers();
   468→  for (const p of properties) {
   469→    const led = ledgers.get(p.id);
   470→    if (!led || led.outstanding <= EPS) continue;
   471→    const daysOver = led.oldestUnpaidDate
   472→      ? Math.floor((today.getTime() - parseYmd(led.oldestUnpaidDate).getTime()) / 86400000)
   473→      : 0;
   474→    list.push({
   475→      key: `prop-outstanding-${p.id}`,
   476→      severity: daysOver > 14 ? "CRITICAL" : "WARNING",
   477→      title: `${p.name} — payment pending`,
   478→      message: `₹${led.outstanding.toLocaleString("en-IN")} outstanding across ${led.unpaidCount} unpaid day(s)${led.oldestUnpaidDate ? `; oldest since ${led.oldestUnpaidDate}` : ""}.`,
   479→      view: "payments",

   487→    .filter(([, b]) => b.balance > EPS)

   525→  const emis = await db.vehicleEmiPayment.findMany({
   526→    where: { status: "PENDING", dueDate: { gte: today, lte: in7 } },
   527→    include: { vehicle: { select: { name: true } } },
   528→    orderBy: { dueDate: "asc" },
   529→  });
   530→  for (const e of emis) {

   557→      list.push({
   558→        key: `veh-fit-${v.id}`,
   559→        severity: v.fitnessExpiry < today ? "CRITICAL" : "WARNING",
   560→        title: `Fitness ${v.fitnessExpiry < today ? "expired" : "expiring"} — ${v.name}`,
   561→        message: `Fitness certificate for ${v.registrationNumber} expires on ${dayKey(v.fitnessExpiry)}.`,
   562→        view: "vehicle-detail",
   563→        params: { vehicleId: v.id },

   565→    }
   566→  }

   569→  const maintenances = await db.maintenance.findMany({
   570→    where: { status: "SCHEDULED" },
   571→    include: { vehicle: { select: { name: true } } },
   572→    orderBy: { date: "asc" },
   573→  });
   574→  for (const m of maintenances) {

   576→      key: `mnt-${m.id}`,
   577→      severity: "INFO",
   578→      title: `Maintenance scheduled — ${m.vehicle.name}`,
   579→      message: `${m.type} on ${dayKey(m.date)}${m.description ? ` — ${m.description}` : ""}.`,
   580→      view: "vehicle-detail",
   581→      params: { vehicleId: m.vehicleId },
   582→    });
   583→  }
   584→
   585→  // 7) Trips with pending payments
   586→  const trips = await db.trip.findMany({
   587→    where: { status: { not: "CANCELLED" }, paymentStatus: { in: ["PENDING", "PARTIAL"] } },
   588→    include: { vehicle: { select: { name: true } }, client: { select: { name: true } } },
   589→    orderBy: { startAt: "asc" },
   590→    take: 10,
   591→  });
   592→  for (const t of trips) {
   593→    const pending = round2(tripTarget(t) - t.paidAmount);
   594→    if (pending <= EPS) continue;
   595→    list.push({
   596→      key: `trip-pay-${t.id}`,
   597→      severity: "WARNING",
   598→      title: `Trip payment pending — ${t.vehicle.name}`,
   599→      message: `${t.client?.name ?? "Client"} owes ₹${pending.toLocaleString("en-IN")} (${t.rentalType.toLowerCase()} from ${dayKey(t.startAt)}).`,
   600→      view: "trips",
   601→      params: { tripId: t.id },
   602→    });
   603→  }
   604→
   605→  const dismissed = await db.notificationDismiss.findMany({ select: { key: true } });
   606→  const dismissedSet = new Set(dismissed.map((d) => d.key));
   607→  return list
   608→    .filter((n) => !dismissedSet.has(n.key))
   609→    .sort((a, b) => SEV_ORDER[a.severity] - SEV_ORDER[b.severity]);
   610→}

   614→/** Next sequential code like EMP-007 / VEH-003 based on existing rows. */
   615→export async function nextCode(prefix: string, kind: "employee" | "vehicle"): Promise<string> {
   616→  const rows =
   617→    kind === "employee"
   618→      ? await db.employee.findMany({ where: { code: { startsWith: `${prefix}-` } }, select: { code: true } })
   619→      : await db.vehicle.findMany({ where: { code: { startsWith: `${prefix}-` } }, select: { code: true } });
   620→  let max = 0;
   621→  for (const r of rows) {
   622→    const n = parseInt(r.code.slice(prefix.length + 1), 10);
   623→    if (Number.isFinite(n) && n > max) max = n;
   624→  }
   625→  return `${prefix}-${String(max + 1).padStart(3, "0")}`;
   626→}

   628→export function requireEnum<T extends string>(value: unknown, allowed: readonly T[], label: string): T {
   629→  const v = String(value ?? "").toUpperCase();
   630→  if (!allowed.includes(v as T)) {
   631→    throw new HttpError(400, `Invalid ${label}: must be one of ${allowed.join(", ")}`);
   632→  }
   633→  return v as T;
   634→}

   636→export function requirePositiveAmount(value: unknown, label = "Amount"): number {
   637→  const n = typeof value === "string" ? parseFloat(value) : Number(value);
   638→  if (!Number.isFinite(n) || n <= 0) throw new HttpError(400, `${label} must be greater than 0`);
   639→  return round2(n);
   640→}

   642→export function optionalAmount(value: unknown, fallback = 0): number {
   643→  if (value === undefined || value === null || value === "") return fallback;
   644→  const n = typeof value === "string" ? parseFloat(value) : Number(value);
   645→  if (!Number.isFinite(n)) throw new HttpError(400, "Invalid number value");
   646→  return round2(n);
   647→}
